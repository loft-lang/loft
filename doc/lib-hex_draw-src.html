// cellopen — DOORS AND WINDOWS ON WALLS DRAWN FROM CELLS.
//
// `housedraw::place_opening` marks N edges at `t` along a side of a `Plan`.  A structure known
// only by its cells (`cellwalls`) has no sides to count and no `t`; and an opening that must
// come through a turn of 30 degrees looking like itself cannot be "three edges", because the
// turned wall's edges lie otherwise.  So here an opening is a KIND, a POINT and a WIDTH:
//
//   in the data   a material on wall edges (`OPEN_DOOR`, `OPEN_WINDOW`, as `housedraw` writes
//                 them), never a gap: the edges stay wall, the run stays one run, and the
//                 straight wall is solved as if the opening were not there;
//   to the walls  a point on the wall's line, its centre — the mean of its edges' midpoints,
//                 taken onto the line — and a span about it of its kind's one width, the same
//                 on every wall at every angle, so one door mesh fits everywhere.
//
// `cells_place_opening` goes from a point to marked edges; `cells_openings` from marked edges
// back to centres and spans.  Extracted from tile-world-game, whose rules these keep and cite by
// name (the HO rules, its `formal/hex-houses.md`).
use hex_field::*;
use hex_form::*;
use hexsurf::*;
use cellwalls::*;
use housedraw::*;

// The measurements openings are placed and drawn with, in world units: a door's and a window's
// width, and how far an opening's edge keeps from its wall's corner.
pub struct OpeningSizes {
  pub os_door:   float,
  pub os_window: float,
  pub os_corner: float,
}

// The three measurements, in world units (one unit is 0.866 m).
pub fn opening_sizes(door: float, window: float, corner: float) -> OpeningSizes {
  OpeningSizes { os_door: door, os_window: window, os_corner: corner }
}

// A door and a window one unit wide (one hex edge, 0.87 m), an opening's edge sqrt(3)/2 unit
// from the corner.
pub fn opening_sizes_default() -> OpeningSizes { opening_sizes(1.0, 1.0, 0.8660254037844386) }

// The width an opening of `kind` is drawn at: a door's for `OPEN_DOOR`, a window's for any other.
pub fn opening_size(kind: integer, sz: OpeningSizes) -> float {
  if kind == OPEN_DOOR { sz.os_door } else { sz.os_window }
}

// A door or a window on a built structure: its kind, the wall it is in (an index into the
// structure's walls), its edges (indices into the boundary, in order along the wall), its centre
// on the wall's line and the span it is drawn over, from a to b, in world units.
pub struct CellOpening {
  pub op_kind:  integer,
  pub op_wall:  integer,
  pub op_edges: vector<integer>,
  pub op_cx:    float,
  pub op_cy:    float,
  pub op_ax:    float,
  pub op_ay:    float,
  pub op_bx:    float,
  pub op_by:    float,
}

// The opening on wall `wi` over the edges `group`: its centre where its edges' midpoints fall on
// the wall's line, and its span, its kind's width about that centre.
fn opening_over(b: const vector<WallEdge>, w: WallSurface, wi: integer, kind: integer,
                group: const vector<integer>, sz: OpeningSizes) -> CellOpening {
  px = 0.0;
  py = 0.0;
  for i in group {
    e = b[i]!;
    px += ((e.we_ak + e.we_bk) as float) * 0.5 * 0.8660254037844386;
    py += ((e.we_am + e.we_bm) as float) * 0.25;
  }
  n = len(group) as float;
  px = (px / n) ?? 0.0;
  py = (py / n) ?? 0.0;
  mx = surface_mean_x(w);
  my = surface_mean_y(w);
  (dx, dy) = surface_dir(w);
  t = (px - mx) * dx + (py - my) * dy;
  cx = mx + t * dx;
  cy = my + t * dy;
  half = opening_size(kind, sz) * 0.5;
  CellOpening { op_kind: kind, op_wall: wi, op_edges: group, op_cx: cx, op_cy: cy,
                op_ax: cx - half * dx, op_ay: cy - half * dy, op_bx: cx + half * dx, op_by: cy + half * dy }
}

// @FR-HO-Point: the walls are solved first, the openings are points on them
// The openings `marks` holds on the walls of a structure (its boundary, runs and walls, as
// `cells_built` gives them): along each wall's run, each stretch of edges marked with one kind
// (`OPEN_DOOR` or `OPEN_WINDOW`) is one opening.
pub fn cells_openings(b: const vector<WallEdge>, rs: const vector<vector<integer>>,
                      walls: const vector<WallSurface>, marks: EdgeSet, sz: OpeningSizes) -> vector<CellOpening> {
  out: vector<CellOpening> = [];
  for wi in 0..len(rs) {
    run = rs[wi]!;
    i = 0;
    while i < len(run) {
      e = b[run[i] ?? 0]!;
      kind = edge_mat(marks, e.we_qa, e.we_ra, e.we_qb, e.we_rb);
      if kind != OPEN_DOOR && kind != OPEN_WINDOW {
        i += 1;
        continue;
      }
      group: vector<integer> = [];
      while i < len(run) {
        f = b[run[i] ?? 0]!;
        if edge_mat(marks, f.we_qa, f.we_ra, f.we_qb, f.we_rb) != kind { break; }
        group += [run[i] ?? 0];
        i += 1;
      }
      out += [opening_over(b, walls[wi]!, wi, kind, group, sz)];
    }
  }
  out
}

fn open_min(a: float, b: float) -> float { if a < b { a } else { b } }
fn open_max(a: float, b: float) -> float { if a > b { a } else { b } }

// @FR-HO-Place: an opening at a point marks the nearest wall's edges within half its width
// Mark an opening of `kind` in `marks` at the point (x, y) (world units) of the built structure
// `cw`: on the wall nearest the point, its centre where the point falls along the wall, kept
// clear of the wall's ends by half the opening's width and the corner clearance; the edges of
// that wall whose midpoints, along it, lie within half the opening's width of the centre, or the
// nearest one when none does.  False when the structure has no wall.
//
// The walls do not depend on the marks, so one `cells_built` serves any number of openings.
pub fn cells_place_opening(cw: const CellWalls, marks: EdgeSet, kind: integer, x: float, y: float,
                           sz: OpeningSizes) -> boolean {
  n = len(cw.cw_walls);
  best = -1;
  best_d = 0.0;
  best_t = 0.0;
  best_lo = 0.0;
  best_hi = 0.0;
  for i in 0..n {
    w = cw.cw_walls[i]!;
    (dx, dy) = surface_dir(w);
    mx = surface_mean_x(w);
    my = surface_mean_y(w);
    (ax, ay) = cw.cw_corners[(i + n - 1) % n ?? 0] ?? (0.0, 0.0);
    (bx, by) = cw.cw_corners[i] ?? (0.0, 0.0);
    t0 = (ax - mx) * dx + (ay - my) * dy;
    t1 = (bx - mx) * dx + (by - my) * dy;
    lo = open_min(t0, t1);
    hi = open_max(t0, t1);
    t = (x - mx) * dx + (y - my) * dy;
    across = fabs((x - mx) * -dy + (y - my) * dx);
    beyond = if t < lo { lo - t } else if t > hi { t - hi } else { 0.0 };
    dist = across + beyond;
    if best < 0 || dist < best_d {
      best = i;
      best_d = dist;
      best_t = t;
      best_lo = lo;
      best_hi = hi;
    }
  }
  if best < 0 { return false; }
  half = opening_size(kind, sz) * 0.5;
  // half its width, and the corner clearance
  clear = half + sz.os_corner;
  t = best_t;
  if best_lo + clear > best_hi - clear { t = (best_lo + best_hi) * 0.5; }
  else { t = open_min(open_max(t, best_lo + clear), best_hi - clear); }
  w = cw.cw_walls[best]!;
  (dx, dy) = surface_dir(w);
  mx = surface_mean_x(w);
  my = surface_mean_y(w);
  picks: vector<integer> = [];
  nearest = -1;
  near_d = 0.0;
  for i in cw.cw_runs[best] ?? [] {
    e = cw.cw_boundary[i]!;
    ex = ((e.we_ak + e.we_bk) as float) * 0.5 * 0.8660254037844386;
    ey = ((e.we_am + e.we_bm) as float) * 0.25;
    off = fabs((ex - mx) * dx + (ey - my) * dy - t);
    if off <= half + 0.000000001 { picks += [i]; }
    if nearest < 0 || off < near_d {
      nearest = i;
      near_d = off;
    }
  }
  if len(picks) == 0 { picks = [nearest]; }
  for i in picks {
    e = cw.cw_boundary[i]!;
    edge_set_mat(marks, e.we_qa, e.we_ra, e.we_qb, e.we_rb, kind);
  }
  true
}

// cellrooms — THE ROOMS OF A STRUCTURE, AND ITS INNER WALLS AS STRAIGHT WALLS.
//
// `cellwalls` draws a structure's OUTER walls from its cells: the boundary between the cells
// inside and those outside.  A structure has walls inside it too, marked on the edges between
// two of its own cells.  They are drawn the same way, once it is said what they bound:
//
//   a ROOM is the cells an inner wall does not let you leave — walk from a cell to every
//   neighbour across an edge that is not marked, and what you reach is its room (a door is a
//   material on a wall's edges, so a door does not join two rooms);
//
//   an INNER WALL is where two rooms meet.  Each room is a structure of its own, and its
//   boundary is solved into straight walls as any structure's is (`cells_outlines`); the stretch
//   of one of its walls that borders another room is an inner wall, straight, along that wall's
//   line.
//
// So an inner wall is the boundary of a filled region, as an outer wall is: it is closed where it
// was drawn closed, it is as straight as an outer wall, and a marked edge that bounds no room — a
// stub in the middle of a floor — is no wall at all, and is counted instead of drawn.
//
// From tile-world-game, whose houses' rooms are parted this way.
use hex_field::*;
use hex_form::*;
use hexsurf::*;
use cellwalls::*;
use cellopen::*;
use housedraw::*;

// The rooms of a structure: a number per inside cell (0 .. `cr_n` - 1), -1 outside it, the rooms
// numbered as their first cells come row by row.
pub struct CellRooms {
  pub cr_of: Labels,
  pub cr_n:  integer,
}

// The rooms of `cells`, parted by the edges marked in `edges` between two of its cells.  With
// nothing marked inside it a structure of one piece is one room.
pub fn cells_rooms(cells: HexSet, edges: EdgeSet) -> CellRooms {
  of = labels_new(hexset_q0(cells), hexset_r0(cells), hexset_w(cells), hexset_h(cells));
  n = 0;
  for r in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for q in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if !hexset_get(cells, q, r) || label_get(of, q, r) >= 0 { continue; }
      label_set(of, q, r, n);
      sq: vector<integer> = [q];
      sr: vector<integer> = [r];
      at = 0;
      while at < len(sq) {
        cq = sq[at] ?? 0;
        cr = sr[at] ?? 0;
        at += 1;
        for d in 0..6 {
          nq = nb_q(cq, cr, d);
          nr = nb_r(cq, cr, d);
          if !hexset_get(cells, nq, nr) || label_get(of, nq, nr) >= 0 { continue; }
          if edge_mat(edges, cq, cr, nq, nr) != 0 { continue; }
          label_set(of, nq, nr, n);
          sq += [nq];
          sr += [nr];
        }
      }
      n += 1;
    }
  }
  CellRooms { cr_of: of, cr_n: n }
}

// The cells of room `ri` as a set of their own, over the structure's window.
pub fn room_cells(cells: HexSet, rooms: const CellRooms, ri: integer) -> HexSet {
  out = hexset_chunk(hexset_q0(cells), hexset_r0(cells), hexset_w(cells), hexset_h(cells));
  for r in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for q in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if label_get(rooms.cr_of, q, r) == ri { hexset_set(out, q, r, true); }
    }
  }
  out
}

// A structure's inner walls and the openings in them.  `in_lost` is how many marked edges had
// one room on both sides: they bound no room and are no wall.
pub struct InnerWalls {
  pub in_rooms: integer,
  pub in_walls: vector<InnerWall>,
  pub in_opens: vector<InnerOpening>,
  pub in_lost:  integer,
}

fn inner_is_opening(m: integer) -> boolean { m == OPEN_DOOR || m == OPEN_WINDOW }

// The most common of `mats` that is a wall's own; of all of them when there is no other.
fn inner_material(mats: const vector<integer>) -> integer {
  best = 0;
  best_n = 0;
  for pass in 0..2 {
    for m in mats {
      if m == 0 || (pass == 0 && inner_is_opening(m)) { continue; }
      c = 0;
      for o in mats { if o == m { c += 1; } }
      if c > best_n {
        best = m;
        best_n = c;
      }
    }
    if best_n > 0 { return best; }
  }
  best
}

// The inner walls of `cells` as straight walls, from the edges marked between its cells.  `hint`
// (a heading, -1 for none) is asked for a room whose cells show no direction; `sz` gives a door's
// and a window's width.
//
// Each room's outline is solved (`cells_outlines`), and along each of its walls every unbroken
// stretch of edges that borders ONE other room is an inner wall on that wall's line — taken from
// the room of the lower number, so each is found once.  Its extent is its edges' corners brought
// onto the line, and the wall's own corner where the stretch runs to an end of the wall.  A stretch of fewer than three edges is the turn of a corner, as on an outer
// wall, and no wall of its own.
pub fn cells_inner_walls(cells: HexSet, edges: EdgeSet, hint: integer = -1,
                         sz: OpeningSizes = opening_sizes_default()) -> InnerWalls {
  rooms = cells_rooms(cells, edges);
  walls: vector<InnerWall> = [];
  opens: vector<InnerOpening> = [];
  lost = 0;
  for r in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for q in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if !hexset_get(cells, q, r) { continue; }
      for d in 0..6 {
        nq = nb_q(q, r, d);
        nr = nb_r(q, r, d);
        if !hexset_get(cells, nq, nr) || nr < r || (nr == r && nq < q) { continue; }
        if edge_mat(edges, q, r, nq, nr) != 0 && label_get(rooms.cr_of, q, r) == label_get(rooms.cr_of, nq, nr) { lost += 1; }
      }
    }
  }
  if rooms.cr_n < 2 { return InnerWalls { in_rooms: rooms.cr_n, in_walls: walls, in_opens: opens, in_lost: lost }; }
  for ri in 0..rooms.cr_n {
    for o in cells_outlines(room_cells(cells, rooms, ri), hint) {
      b = o.co_boundary;
      for wi in 0..len(o.co_walls) {
        w = o.co_walls[wi]!;
        run = o.co_runs[wi] ?? [];
        (dx, dy) = surface_dir(w);
        mx = surface_mean_x(w);
        my = surface_mean_y(w);
        i = 0;
        while i < len(run) {
          e = b[run[i] ?? 0]!;
          other = label_get(rooms.cr_of, e.we_qb, e.we_rb);
          // a stretch against one room of a higher number
          if !hexset_get(cells, e.we_qb, e.we_rb) || other <= ri {
            i += 1;
            continue;
          }
          lo = 1000000.0;
          hi = -1000000.0;
          mats: vector<integer> = [];
          first = i;
          while i < len(run) {
            f = b[run[i] ?? 0]!;
            if !hexset_get(cells, f.we_qb, f.we_rb) || label_get(rooms.cr_of, f.we_qb, f.we_rb) != other { break; }
            mats += [edge_mat(edges, f.we_qa, f.we_ra, f.we_qb, f.we_rb)];
            for c in [(f.we_ak, f.we_am), (f.we_bk, f.we_bm)] {
              t = ((c.0 as float) * 0.8660254037844386 - mx) * dx + ((c.1 as float) * 0.5 - my) * dy;
              if t < lo { lo = t; }
              if t > hi { hi = t; }
            }
            i += 1;
          }
          // under three edges it is the turn of a corner — where this wall's run caught the end of
          // the inner wall that meets it — and not a wall of its own
          if i - first < 3 { continue; }
          // where the stretch runs to an end of this wall it runs to the wall's corner — where
          // its line meets the next wall's — whichever of the two walls the corner's own edges
          // were counted with
          nw = len(o.co_corners);
          if first == 0 && nw > 0 {
            c = o.co_corners[(wi + nw - 1) % nw ?? 0] ?? (0.0, 0.0);
            t = (c.0 - mx) * dx + (c.1 - my) * dy;
            if t < lo { lo = t; }
            if t > hi { hi = t; }
          }
          if i == len(run) && nw > 0 {
            c = o.co_corners[wi] ?? (0.0, 0.0);
            t = (c.0 - mx) * dx + (c.1 - my) * dy;
            if t < lo { lo = t; }
            if t > hi { hi = t; }
          }
          at = len(walls);
          walls += [InnerWall { iw_a: ri, iw_b: other, iw_mat: inner_material(mats), iw_edges: i - first, iw_surface: w,
                                iw_ax: mx + lo * dx, iw_ay: my + lo * dy, iw_bx: mx + hi * dx, iw_by: my + hi * dy }];
          // its doors and windows: each unbroken stretch of one kind, centred on the line
          j = first;
          while j < i {
            kind = mats[j - first] ?? 0;
            if !inner_is_opening(kind) {
              j += 1;
              continue;
            }
            sum = 0.0;
            count = 0;
            while j < i && (mats[j - first] ?? 0) == kind {
              g = b[run[j] ?? 0]!;
              sum += (((g.we_ak + g.we_bk) as float) * 0.5 * 0.8660254037844386 - mx) * dx
                   + (((g.we_am + g.we_bm) as float) * 0.25 - my) * dy;
              count += 1;
              j += 1;
            }
            t = (sum / (count as float)) ?? 0.0;
            half = opening_size(kind, sz) * 0.5;
            opens += [InnerOpening { io_wall: at, io_kind: kind, io_cx: mx + t * dx, io_cy: my + t * dy,
                                     io_ax: mx + (t - half) * dx, io_ay: my + (t - half) * dy,
                                     io_bx: mx + (t + half) * dx, io_by: my + (t + half) * dy }];
          }
        }
      }
    }
  }
  InnerWalls { in_rooms: rooms.cr_n, in_walls: walls, in_opens: opens, in_lost: lost }
}

// cellwalls — STRAIGHT WALLS FROM THE CELLS ALONE.  No `Plan`, no angle, no side numbers.
//
// `hexsurf` reads a wall's surface off the edges of one side of a `Plan`: it has to be told the
// rectangle the cells were drawn from.  Here nothing is told.  A structure is a set of cells —
// each cell inside it or outside it — and its walls follow from them: an L, a hall with wings, a
// house the editor turned by 30 degrees and kept only the cells of, a road, a stream's bed.
//
// From a set of cells, in order:
//   1. the boundary, walked (`cells_loops`, on `hex_field::trace_edges`): every edge between a
//      cell inside and one outside, the structure on the left;
//   2. the two axes its walls run along, read off the boundary (`cells_axes`: each edge votes for
//      the heading nearest the way the boundary travels round it; cells that show no direction
//      answer -1, and the caller's hint decides), and the boundary cut into runs, one per
//      straight wall (`cells_runs`);
//   3. each run's straight wall (`cells_wall`): one routine for each of the two kinds of wall
//      the lattice has, the answer a `WallSurface`, so `hexsurf`'s readers take it unchanged;
//   4. the corners, each wall mitred into the next (`surface_meet`).
// `cells_built` does the four for a structure of one loop; `cells_outlines` for any number of
// loops, with the walls a road's outline needs besides (a step, a slanted stretch).
//
// Extracted from tile-world-game, whose rules these routines keep and cite by name (the HW rules,
// its `formal/hex-houses.md`), and whose town they were checked on wall for wall.
//
// Lattice units, as `hex_field` keeps them: a cell centre is (2q + (r & 1), 3r) in doubled
// coordinates (k, m), a point is at x = k*sqrt(3)/2, y = m/2 world units.
use hex_field::*;
use hex_form::*;
use hexsurf::*;
use housedraw::*;

// One edge of a boundary: the structure's cell (qa, ra) and the outside one (qb, rb), and its
// corners from (ak, am) to (bk, bm) in doubled coordinates, the structure on the left.
pub struct WallEdge {
  pub we_qa: integer,
  pub we_ra: integer,
  pub we_qb: integer,
  pub we_rb: integer,
  pub we_ak: integer,
  pub we_am: integer,
  pub we_bk: integer,
  pub we_bm: integer,
}

// @FR-HW-Loops: every loop of an outline, a courtyard's and a block's too
// Every loop of the boundary of `cells`, each in walking order with the cells on its left: the
// outer loop of each piece and one round each hole in it (a courtyard, the block a ring of road
// goes round).  `hex_field::trace_edges`, an edge at a time: its loops, their order and starts.
pub fn cells_loops(cells: HexSet) -> vector<vector<WallEdge>> {
  // @FR-HX-Edge: the boundary is the edges between a cell inside and one outside
  em = trace_edges(cells);
  out: vector<vector<WallEdge>> = [];
  for li in 0..em.em_loops {
    a = em.em_start[li] ?? 0;
    b = em.em_start[li + 1] ?? 0;
    walk: vector<WallEdge> = [];
    for j in a..b {
      nx = if j + 1 < b { j + 1 } else { a };
      walk += [WallEdge { we_qa: em.em_qa[j] ?? 0, we_ra: em.em_ra[j] ?? 0, we_qb: em.em_qb[j] ?? 0, we_rb: em.em_rb[j] ?? 0,
                          we_ak: em.em_k[j] ?? 0, we_am: em.em_m[j] ?? 0, we_bk: em.em_k[nx] ?? 0, we_bm: em.em_m[nx] ?? 0 }];
    }
    out += [walk];
  }
  out
}

// The boundary of a structure of one piece without a courtyard: its first loop.
pub fn cells_boundary(cells: HexSet) -> vector<WallEdge> {
  ls = cells_loops(cells);
  if len(ls) == 0 { return []; }
  ls[0]!
}

// ── the boundary cut into runs ──────────────────────────────────────────────

// How many edges either side the travelling direction is read over: enough to span a period of
// the zigzag (2 edges) and of the staircase (3).
pub const CELLS_AROUND = 2;

// Which way the boundary travels at edge i, over CELLS_AROUND edges either side: 0 along +u, 1 along
// +v, 2 along -u, 3 along -v (u the house's heading, v the heading three steps on); -1 when it
// is exactly between.  Exact: the two heading steps are √3 and 3 long, one each way round, so
// the projections compare as squares, one of them times 3.
fn edge_travel(turn: integer, b: const vector<WallEdge>, i: integer) -> integer {
  n = len(b);
  a = b[((i - CELLS_AROUND) % n + n) % n ?? 0]!;
  z = b[(i + CELLS_AROUND) % n ?? 0]!;
  wk = z.we_bk - a.we_ak;
  wm = z.we_bm - a.we_am;
  (uk, um) = head_step(turn);
  (vk, vm) = head_step(turn + 3);
  u = vec_dot_4(wk, wm, uk, um);
  v = vec_dot_4(wk, wm, vk, vm);
  // |u| / L_u against |v| / L_v: with L_u = √3, L_v = 3 that is 3u² against v²; the other way
  // round, u² against 3v².
  au = if turn % 2 == 0 { 3 * u * u } else { u * u };
  av = if turn % 2 == 0 { v * v } else { 3 * v * v };
  if au > av { if u > 0 { 0 } else { 2 } } else if av > au { if v > 0 { 1 } else { 3 } } else { -1 }
}

// Every edge's travel (`edge_travel`), read once: the runs and each run's heading both go by it.
fn edge_travels(turn: integer, b: const vector<WallEdge>) -> vector<integer> {
  [for i in 0..len(b) { edge_travel(turn, b, i) }]
}

// The boundary cut into runs, one per straight wall, in order round the house.  A run shorter
// than 2 edges is the turn of a corner, not a wall: it joins the run before it.
// @FR-HW-Runs: one run per straight wall; a run under two edges joins the one before
pub fn cells_runs(turn: integer, b: const vector<WallEdge>) -> vector<vector<integer>> {
  runs_merged(b, edge_travels(turn, b))
}

// The boundary cut into runs of equal labels, in order round the loop: a label of -1 takes the
// one before it, and a run shorter than 2 edges (the turn of a corner) joins the run before it.
//
// The runs are kept as the places where the label changes, so joining one costs what the runs
// cost, not what the edges do: a road's outline has hundreds of edges and many corners.  The
// runs are taken in order from the first change after edge 0, and the first short one joins the
// run before it; then again, until none is short or one run is left.
fn runs_merged(b: const vector<WallEdge>, given: const vector<integer>) -> vector<vector<integer>> {
  n = len(b);
  label: vector<integer> = [for i in 0..n { given[i] ?? -1 }];
  for i in 0..n {
    if label[i] < 0 { label[i] = label[(i + n - 1) % n ?? 0] ?? -1; }
  }
  // where a run begins: every edge whose label is not the one before it, ascending
  cp: vector<integer> = [];
  for i in 0..n { if label[i] != label[(i + n - 1) % n ?? 0] { cp += [i]; } }
  for _ in 0..n {
    g = len(cp);
    if g < 2 { break; }
    // the runs in order: from the first change after edge 0 (a change AT edge 0 begins the last)
    first = if (cp[0] ?? 0) == 0 { 1 } else { 0 };
    short = -1;
    for j in 0..g {
      if short >= 0 { break; }
      at = (first + j) % g ?? 0;
      from = cp[at] ?? 0;
      to = cp[(at + 1) % g ?? 0] ?? 0;
      size = ((to - from) % n + n) % n ?? 0;
      if size < 2 { short = at; }
    }
    if short < 0 { break; }
    before = (short + g - 1) % g ?? 0;
    after = (short + 1) % g ?? 0;
    prev = label[cp[before] ?? 0] ?? 0;
    // the short run's edges take the label of the run before them
    from = cp[short] ?? 0;
    size = (((cp[after] ?? 0) - from) % n + n) % n ?? 0;
    for j in 0..size { label[(from + j) % n ?? 0] = prev; }
    // its beginning is no change now; nor is the next run's, when that has the same label
    gone_after = (label[cp[after] ?? 0] ?? 0) == prev && after != before;
    whole = g == 2 && (label[cp[after] ?? 0] ?? 0) == prev;
    keep: vector<integer> = [];
    for j in 0..g {
      if j == short || (j == after && gone_after) || whole { continue; }
      keep += [cp[j] ?? 0];
    }
    cp = keep;
  }
  runs_groups(label)
}

// The runs of equal labels round the loop, starting where a label changes.
fn runs_groups(label: const vector<integer>) -> vector<vector<integer>> {
  n = len(label);
  start = 0;
  for i in 0..n {
    if start == 0 && label[i] != label[(i + n - 1) % n ?? 0] { start = i; }
  }
  out: vector<vector<integer>> = [];
  cur: vector<integer> = [];
  for j in 0..n {
    i = (start + j) % n ?? 0;
    if len(cur) > 0 && label[i] != label[cur[0] ?? 0] {
      out += [cur];
      cur = [];
    }
    cur += [i];
  }
  if len(cur) > 0 { out += [cur]; }
  out
}

// ── the two kinds of straight wall, solved from the wall data ───────────────
//
// A straight wall on the hex grid is one of two kinds (one routine for each):
//
//   TOPS   (an even heading, the grid's own edge directions): the wobble across the tops of a row
//          of hexes, edges zigzagging between the row's apexes and shoulders.  Turned to run due
//          east, every edge's midpoint lies on one line, half way up the band (ROUNDTRIP §6.2,
//          band 1/2 unit) — so the line is exact from any of its edges, a whole run or a cut one.
//   SIDES  (an odd heading, the grid's corner directions): the line along the sides of hexes
//          under each other, upright sides alternating with short slanted connectors where the
//          rows shift by half a cell.  Turned to run due north, every connector's midpoint lies on
//          the line, in the middle of the band (√3/2 unit); the upright sides lie on its two
//          edges.
//
// So each routine turns its run into its own frame (a 60° turn is exact on the lattice, X1),
// reads the line off the edges that make its pattern — the most common midpoint: an edge of the
// next wall, caught in the run at a corner, has a midpoint of its own and is outvoted — and turns
// the line back.  The direction is the heading itself, exactly.  The answer is a
// `WallSurface`, so `hex_draw`'s readers and its miter (`surface_meet`) take it unchanged.  On a
// whole tops run (a side of a box) it is the line `surface_of` gives; on a sides run it
// lies 0.048 unit beside it (measured on a 5 x 4 box): the average counts the corner's edges in,
// this outvotes them.

// Turned `steps` × 60° anticlockwise.
pub fn lattice_turned(k: integer, m: integer, steps: integer) -> (integer, integer) {
  tk = k;
  tm = m;
  for _ in 0..((steps % 6) + 6) % 6 ?? 0 { (tk, tm) = lattice_rot60(tk, tm); }
  (tk, tm)
}

// A rational point (`nk / d`, `nm / d`) turned `steps` × 60°: the numerators doubled first, so
// the halving stays exact.
fn rational_turned(nk: integer, nm: integer, d: integer, steps: integer) -> (integer, integer, integer) {
  tk = nk;
  tm = nm;
  td = d;
  for _ in 0..((steps % 6) + 6) % 6 ?? 0 {
    (tk, tm) = (tk - tm, 3 * tk + tm);
    td = 2 * td;
  }
  (tk, tm, td)
}

// The most common value of `vals`, ties going to the one nearest the mean (`sum / n`), and
// among those to the one met first.  Each distinct value is counted once: a wall's edges mark
// two or three lines at most, however long it is.
fn most_common(vals: const vector<integer>, sum: integer, n: integer) -> integer {
  seen: vector<integer> = [];
  count: vector<integer> = [];
  for v in vals {
    at = -1;
    for j in 0..len(seen) { if at < 0 && (seen[j] ?? 0) == v { at = j; } }
    if at < 0 {
      seen += [v];
      count += [1];
    } else {
      count[at] += 1;
    }
  }
  best = 0;
  best_n = 0;
  for j in 0..len(seen) {
    v = seen[j] ?? 0;
    c = count[j] ?? 0;
    far = v * n - sum;
    if far < 0 { far = -far; }
    best_far = best * n - sum;
    if best_far < 0 { best_far = -best_far; }
    if c > best_n || c == best_n && far < best_far {
      best = v;
      best_n = c;
    }
  }
  best
}

// A TOPS wall (heading `h` even) through the run's edges.
// @FR-HW-Tops
pub fn cells_wall_tops(b: const vector<WallEdge>, run: const vector<integer>, h: integer) -> WallSurface {
  s = (6 - h / 2 ?? 0) % 6 ?? 0;        // turns heading h to 0, due east
  mids: vector<integer> = [];
  sk = 0;
  sm = 0;
  for i in run {
    e = b[i];
    (ak, am) = lattice_turned(e.we_ak, e.we_am, s);
    (bk, bm) = lattice_turned(e.we_bk, e.we_bm, s);
    mids += [am + bm];
    sk += ak + bk;
    sm += am + bm;
  }
  n = len(run);
  // Due east, the line is m = common / 2: the point (sk / 2n, common / 2).
  line_m = most_common(mids, sm, n);
  (nk, nm, d) = rational_turned(sk, n * line_m, 2 * n, 6 - s);
  (dk, dm) = head_step(h);
  WallSurface { ws_dk: dk, ws_dm: dm, ws_num_k: nk, ws_num_m: nm, ws_den: d, ws_n: n }
}

// A SIDES wall (heading `h` odd) through the run's edges.
// @FR-HW-Sides
pub fn cells_wall_sides(b: const vector<WallEdge>, run: const vector<integer>, h: integer) -> WallSurface {
  s = ((3 - h) / 2 ?? 0) % 6 ?? 0;      // turns heading h to 3, due north
  links: vector<integer> = [];
  sk = 0;
  sm = 0;
  lk = 0;
  for i in run {
    e = b[i];
    (ak, am) = lattice_turned(e.we_ak, e.we_am, s);
    (bk, bm) = lattice_turned(e.we_bk, e.we_bm, s);
    if ak != bk {
      links += [ak + bk];
      lk += ak + bk;
    }
    sk += ak + bk;
    sm += am + bm;
  }
  n = len(run);
  // Due north, the line is k = common / 2 of the connectors (of every edge, when the run has
  // none: one cell's upright side): the point (common / 2, sm / 2n).
  line_k = if len(links) > 0 { most_common(links, lk, len(links)) } else { most_common(links, sk, n) };
  if len(links) == 0 { line_k = sk / n ?? 0; }
  (nk, nm, d) = rational_turned(n * line_k, sm, 2 * n, 6 - s);
  (dk, dm) = head_step(h);
  WallSurface { ws_dk: dk, ws_dm: dm, ws_num_k: nk, ws_num_m: nm, ws_den: d, ws_n: n }
}

// The straight wall through a run of wall edges running along heading `h`: a tops wall on an
// even heading, a sides wall on an odd one.
// @FR-HW-Straight: a wall's direction is its heading's step, exactly
pub fn cells_wall(b: const vector<WallEdge>, run: const vector<integer>, h: integer) -> WallSurface {
  if head_norm(h) % 2 == 0 { cells_wall_tops(b, run, head_norm(h)) }
  else { cells_wall_sides(b, run, head_norm(h)) }
}

// The heading a run of the boundary travels along: the way most of its edges travel (`edge_travel`),
// as a heading of the house (`turn` for +u, three steps on for +v, and so on).
pub fn cells_run_heading(turn: integer, b: const vector<WallEdge>, run: const vector<integer>) -> integer {
  run_heading_by(turn, edge_travels(turn, b), run)
}

// `cells_run_heading` by travels already read.
fn run_heading_by(turn: integer, travel: const vector<integer>, run: const vector<integer>) -> integer {
  count = [0, 0, 0, 0];
  for i in run {
    t = travel[i] ?? -1;
    if t >= 0 { count[t] += 1; }
  }
  best = 0;
  for t in 1..4 { if count[t] > count[best] { best = t; } }
  head_norm(turn + 3 * best)
}

// The axes a boundary's walls run along, read off the boundary itself: 0 for the headings 0, 3,
// 6 and 9 (0°, 90°, …), 1 for 1, 4, 7 and 10, 2 for 2, 5, 8 and 11.  Each edge votes for the
// heading nearest the way the boundary travels round it (`CELLS_AROUND` edges either side); the axes
// with the most votes win, and -1 answers a tie: cells that show no direction (the smallest
// house, one hex and its six neighbours, is the same seven cells at every angle).  Exact: a
// heading step is √3 or 3 long, so the projections compare as squares, scaled by 9 or 3.
// @FR-HW-Axes, @FR-HX-Size: the axes read off the cells; a tie is cells that show no direction
pub fn cells_axes(b: const vector<WallEdge>) -> integer { axes_by(edge_nearest_all(b)) }

// Every edge's nearest heading (`edge_nearest_heading`), read once: the axes are voted by it and
// an outline's slanted stretches are found by it.
fn edge_nearest_all(b: const vector<WallEdge>) -> vector<integer> {
  [for i in 0..len(b) { edge_nearest_heading(b, i) }]
}

// The axes most edges vote for by their nearest headings; -1 for a tie.
fn axes_by(near: const vector<integer>) -> integer {
  votes = [0, 0, 0];
  for best in near {
    if best >= 0 { votes[best % 3] += 1; }
  }
  if votes[0] > votes[1] && votes[0] > votes[2] { 0 }
  else if votes[1] > votes[0] && votes[1] > votes[2] { 1 }
  else if votes[2] > votes[0] && votes[2] > votes[1] { 2 }
  else { -1 }
}

// ── a structure built ───────────────────────────────────────────────────────

// How a wall of a built structure runs.  Straight along one of the 12 headings is all there is
// today; the kind is kept per wall so a wall that curves, or runs between two headings, can
// stand beside these without their readers changing.
pub const WALL_STRAIGHT = 0;

// A structure as built from its cells: the cells, its thin walls (an `EdgeSet` over the cells'
// window and a cell round it, every boundary edge of material `wall_mat`), the boundary in
// walking order, the runs (each a list of indices into the boundary, in order round the
// structure), a surface and a kind per run, and the corners (corner i is where wall i meets
// wall i + 1, in world units).
pub struct CellWalls {
  pub cw_cells:    HexSet,
  pub cw_edges:    EdgeSet,
  pub cw_boundary: vector<WallEdge>,
  pub cw_runs:     vector<vector<integer>>,
  pub cw_walls:    vector<WallSurface>,
  pub cw_kinds:    vector<integer>,
  pub cw_corners:  vector<(float, float)>,
}

// An edge set over the cells' window and a cell round it, with nothing marked.
pub fn cells_no_marks(cells: HexSet) -> EdgeSet {
  edgeset_new(hexset_q0(cells) - 1, hexset_r0(cells) - 1, hexset_w(cells) + 2, hexset_h(cells) + 2)
}

// @FR-HX-Cells, @FR-HX-Editor: the walls from the cells alone, no plan and no angle
// A structure from its cells alone: the thin walls, the boundary, the axes it runs along, a run
// per straight wall, the wall solved for each and the corners.  The axes are read off the
// cells; `hint` (a heading, -1 for none) is asked only when the cells show no direction.  One
// piece without a courtyard, big enough for its cells to carry its walls (@FR-HX-Size);
// `cells_outlines` takes anything.
pub fn cells_built(cells: HexSet, wall_mat: integer, hint: integer = -1) -> CellWalls {
  edges = cells_no_marks(cells);
  // @FR-HX-Edge: the thin walls — every edge of every loop, as `draw_walls` marks them, from the
  // one walk of the boundary
  loops = cells_loops(cells);
  for loop in loops {
    for e in loop { edge_set_mat(edges, e.we_qa, e.we_ra, e.we_qb, e.we_rb, wall_mat); }
  }
  b: vector<WallEdge> = if len(loops) > 0 { loops[0]! } else { [] };
  turn = cells_axes(b);
  if turn < 0 { turn = if hint >= 0 { hint % 3 ?? 0 } else { 0 }; }
  travel = edge_travels(turn, b);
  rs = runs_merged(b, travel);
  walls: vector<WallSurface> = [for run in rs { cells_wall(b, run, run_heading_by(turn, travel, run)) }];
  corners: vector<(float, float)> = [];
  nw = len(walls);
  for i in 0..nw {
    // @FR-HW-Miter: each wall mitred into the next
    (x, y, _ok) = surface_meet(walls[i]!, walls[(i + 1) % nw ?? 0]!);
    corners += [(x, y)];
  }
  CellWalls { cw_cells: cells, cw_edges: edges, cw_boundary: b, cw_runs: rs, cw_walls: walls,
              cw_kinds: [for _ in 0..nw { WALL_STRAIGHT }], cw_corners: corners }
}

// ── outlines with any number of loops ───────────────────────────────────────

// One loop of an outline drawn as straight walls: its boundary, its walls (in order round it)
// and their corners (corner i where wall i meets wall i + 1, world units).
pub struct CellOutline {
  pub co_boundary: vector<WallEdge>,
  pub co_walls: vector<WallSurface>,
  pub co_corners: vector<(float, float)>,
  // per wall, the edges it was solved from (indices into the boundary, in order); none for a step
  pub co_runs: vector<vector<integer>> = [],
}

// @FR-HW-Step: two walls in a row that run the same way are joined by a step
// A loop's straight walls, as `cells_built` draws a house's, with one thing more: where two walls
// in a row run along the same axis (a street stepping over by a row, a lane's dead end), a short
// wall across joins them, through the midpoint of the edge where the boundary changes row, so
// every corner is a right angle and the step's line lies between two hex centres.  None (no walls) for a loop of fewer than three walls: too small
// to carry walls (HX-Size).
pub fn cells_outline(b: const vector<WallEdge>, hint: integer = -1) -> CellOutline {
  near = edge_nearest_all(b);
  turn = axes_by(near);
  if turn < 0 { turn = if hint >= 0 { hint % 3 ?? 0 } else { 0 }; }
  // the runs, each along the nearest of the twelve headings (a road's stretches run every way),
  // each cut where its line steps over (`split_by_line`), each part with its heading
  rs: vector<vector<integer>> = [];
  hs: vector<integer> = [];
  heads = edge_headings(b, turn, near);
  for run in runs_merged(b, heads) {
    h = heads[run[0] ?? 0] ?? turn;
    for part in cells_split_by_line(b, run, h) {
      rs += [part];
      hs += [h];
    }
  }
  solved: vector<WallSurface> = [for i in 0..len(rs) { cells_wall(b, rs[i] ?? [], hs[i] ?? 0) }];
  walls: vector<WallSurface> = [];
  wall_runs: vector<vector<integer>> = [];
  n = len(solved);
  for i in 0..n {
    w = solved[i];
    walls += [w];
    wall_runs += [rs[i] ?? []];
    nx = solved[(i + 1) % n ?? 0]!;
    if n > 1 && w.ws_dk * nx.ws_dm - w.ws_dm * nx.ws_dk == 0 {
      // the step: a short wall across the axis, of the other family, through the edge of the
      // step that marks that family's line (`step_edge`)
      h = head_norm(surface_heading(w) + 3);
      e = runs_step_edge(b, rs[i] ?? [], rs[(i + 1) % n ?? 0] ?? [], w, nx, h);
      (dk, dm) = head_step(h);
      walls += [WallSurface { ws_dk: dk, ws_dm: dm, ws_num_k: e.we_ak + e.we_bk, ws_num_m: e.we_am + e.we_bm, ws_den: 2, ws_n: 0 }];
      wall_runs += [[]];
    }
  }
  corners: vector<(float, float)> = [];
  nw = len(walls);
  if nw < 3 { return CellOutline { co_boundary: b, co_walls: [], co_corners: [] }; }
  for i in 0..nw {
    (x, y, _ok) = surface_meet(walls[i]!, walls[(i + 1) % nw ?? 0]!);
    corners += [(x, y)];
  }
  CellOutline { co_boundary: b, co_walls: walls, co_corners: corners, co_runs: wall_runs }
}

// @FR-HW-Step: a run is cut where its line steps over by a row
// A run cut where the line it runs along steps over: along a tops wall each edge's midpoint is on
// the line, along a sides wall each slanted connector's is, so where that midpoint moves to
// another row (and stays there for two edges or more), a new part begins.  A single edge off the
// line (the next wall's, caught at a corner) stays with its part, and so does a part of fewer
// than three edges at either end of the run.
pub fn cells_split_by_line(b: const vector<WallEdge>, run: const vector<integer>, h: integer) -> vector<vector<integer>> {
  tops = head_norm(h) % 2 == 0;
  s = if tops { (6 - head_norm(h) / 2 ?? 0) % 6 ?? 0 } else { ((3 - head_norm(h)) / 2 ?? 0) % 6 ?? 0 };
  // each edge's line, -1000000 where the edge marks none (an upright side of a sides wall)
  at: vector<integer> = [];
  for i in run {
    e = b[i];
    (ak, am) = lattice_turned(e.we_ak, e.we_am, s);
    (bk, bm) = lattice_turned(e.we_bk, e.we_bm, s);
    if tops { at += [am + bm]; } else if ak != bk { at += [ak + bk]; } else { at += [-1000000]; }
  }
  parts: vector<vector<integer>> = [];
  cur: vector<integer> = [];
  line = -1000000;
  for j in 0..len(run) {
    v = at[j] ?? -1000000;
    // a new line, held by this edge and the next one that marks a line
    if v != -1000000 && line != -1000000 && v != line {
      nxt = -1000000;
      for k in j + 1..len(run) { if nxt == -1000000 && (at[k] ?? -1000000) != -1000000 { nxt = at[k] ?? -1000000; } }
      if nxt == v && len(cur) > 0 {
        parts += [cur];
        cur = [];
      }
    }
    if v != -1000000 && (line == -1000000 || len(cur) == 0 || v == line) { line = v; }
    if v != -1000000 && len(cur) == 0 { line = v; }
    cur += [run[j] ?? 0];
  }
  if len(cur) > 0 { parts += [cur]; }
  // a part of fewer than three edges at either end of the run is a corner's edges caught there,
  // not a step: it joins the part beside it
  n = len(parts);
  if n < 2 { return parts; }
  out: vector<vector<integer>> = [];
  for i in 0..n {
    pt = parts[i] ?? [];
    if i == 1 && len(parts[0] ?? []) < 3 { out = [(parts[0] ?? []) + pt]; continue; }
    if i == 0 && len(pt) < 3 { continue; }
    if i == n - 1 && len(pt) < 3 && len(out) > 0 {
      last = out[len(out) - 1] ?? [];
      out[len(out) - 1] = last + pt;
      continue;
    }
    out += [pt];
  }
  out
}

// How many edges a heading off the loop's two axes must hold for to make a wall of its own: fewer
// are the bevel of a step, which stays on the axes.
pub const CELLS_SLANT = 6;

// @FR-HW-Slant: a stretch off the loop's axes keeps its heading when it holds it long enough
// Each edge's heading: along the loop's axes as a house's walls run (`travel`), except where the
// boundary keeps to another of the twelve headings for CELLS_SLANT edges or more (a road's slanted
// stretch).
fn edge_headings(b: const vector<WallEdge>, turn: integer, free: const vector<integer>) -> vector<integer> {
  n = len(b);
  out: vector<integer> = [];
  for i in 0..n {
    t = edge_travel(turn, b, i);
    out += [if t < 0 { -1 } else { head_norm(turn + 3 * t) }];
  }
  i = 0;
  while i < n {
    h = free[i] ?? -1;
    j = i;
    while j < n && (free[j] ?? -1) == h { j += 1; }
    if h >= 0 && ((h - turn) % 3 + 3) % 3 != 0 && j - i >= CELLS_SLANT {
      for k in i..j { out[k] = h; }
    }
    i = if j > i { j } else { i + 1 };
  }
  out
}

// The heading nearest the way the boundary travels at edge i, over CELLS_AROUND edges either side
// (as `axes_of` votes), -1 when it does not travel.
fn edge_nearest_heading(b: const vector<WallEdge>, i: integer) -> integer {
  n = len(b);
  a = b[((i - CELLS_AROUND) % n + n) % n ?? 0]!;
  z = b[(i + CELLS_AROUND) % n ?? 0]!;
  wk = z.we_bk - a.we_ak;
  wm = z.we_bm - a.we_am;
  best = -1;
  best_d = 0;
  best_l = 1;
  for h in 0..12 {
    (sk, sm) = head_step(h);
    d = vec_dot_4(wk, wm, sk, sm);
    if d <= 0 { continue; }
    l = if h % 2 == 0 { 3 } else { 9 };
    if best < 0 || d * d * best_l > best_d * best_d * l {
      best = h;
      best_d = d;
      best_l = l;
    }
  }
  best
}

// The edge a step between two parallel walls `wa` and `wb` (the end of run `ra`, the start of
// `rb`) runs through, the step along heading `h`: of the edges lying between the two walls'
// lines, one that marks a line of the step's family — any edge for a tops wall (every edge's
// midpoint is on its line), a slanted one for a sides wall (an upright side lies in line with
// the next row's centres).  The run's last edge when none lies between.
fn runs_step_edge(b: const vector<WallEdge>, ra: const vector<integer>, rb: const vector<integer>,
             wa: WallSurface, wb: WallSurface, h: integer) -> WallEdge {
  // the walls' common normal, and their lines' places along it (doubled coordinates, scaled)
  nk = -wa.ws_dm;
  nm = wa.ws_dk;
  oa = ((3 * wa.ws_num_k * nk + wa.ws_num_m * nm) as float) / (wa.ws_den as float) ?? 0.0;
  ob = ((3 * wb.ws_num_k * nk + wb.ws_num_m * nm) as float) / (wb.ws_den as float) ?? 0.0;
  lo = if oa < ob { oa } else { ob };
  hi = if oa < ob { ob } else { oa };
  sides = head_norm(h) % 2 == 1;
  s = if sides { ((3 - head_norm(h)) / 2 ?? 0) % 6 ?? 0 } else { (6 - head_norm(h) / 2 ?? 0) % 6 ?? 0 };
  cand: vector<integer> = [];
  for j in 0..4 { if len(ra) - 1 - j >= 0 { cand += [ra[len(ra) - 1 - j] ?? 0]; } }
  for j in 0..4 { if j < len(rb) { cand += [rb[j] ?? 0]; } }
  best = -1;
  for i in cand {
    e = b[i];
    p = ((3 * (e.we_ak + e.we_bk) * nk + (e.we_am + e.we_bm) * nm) as float) * 0.5;
    if p <= lo + 0.001 || p >= hi - 0.001 { continue; }
    if sides {
      (ak, _am) = lattice_turned(e.we_ak, e.we_am, s);
      (bk, _bm) = lattice_turned(e.we_bk, e.we_bm, s);
      if ak == bk { continue; }
    }
    if best < 0 { best = i; }
  }
  if best < 0 { best = ra[len(ra) - 1] ?? 0; }
  b[best]!
}

  [for l in cells_loops(cells) { cells_outline(l, hint) }]
}

// hexsurf — THE WALL SURFACE, BY AVERAGING.  Nothing here is fitted, so nothing has a tolerance.
//
// `ROUNDTRIP` §6.1: the straight surface a wall run approximates is DERIVED by averaging the edges
// that are already there, never fitted to them.
//
//   direction   the SUM of the edge vectors — an exact INTEGER vector in doubled coordinates
//   position    the mean of the edge MIDPOINTS.  A corner is an integer in (k,m), so a midpoint is
//               a half-integer and the mean is an exact RATIONAL — kept here as numerator and
//               denominator, never as a float
//   band        the exact perpendicular extent of the strip about that mean.  NOT an error term:
//               it is the geometry of the strip, and an exact algebraic constant per family
//
// This is what makes domain-A straight-wall recovery **R1**.  Least squares would introduce a
// residual to threshold; an average of exact rationals IS the answer.  `P4` therefore holds all
// the way through the presentation layer — there is no `ε` anywhere between `𝕋` and the drawn wall.
use hex_field::*;
use hex_grid::*;
use hex_form::*;

// ── the exact band constants (`ROUNDTRIP` §6.2), in world units ────────────
//
// 1 u = one hex edge = the circumradius; 1 hex step = sqrt(3) u = 1.5 m.  These are the CORNER
// bands — the full extent of the strip — which is what §6.2's table names.
pub const BAND_TOPS  = 0.5;                    // east family, dm = 0
pub const BAND_SIDES = 0.8660254037844386;     // north family, dk = 0 — sqrt(3)/2
pub const BAND_RATIO = 1.7320508075688772;     // sqrt(3), exactly

// The widening rule: both walls are presented straight and equally thick, at the LARGER band.
// Only the tops widen, and the amount is a closed form in Q(sqrt3) — nothing measured or tuned.
pub const WIDEN_TOTAL = 0.36602540378443865;   // (sqrt3 - 1)/2  u
pub const WIDEN_FACE  = 0.18301270189221933;   // (sqrt3 - 1)/4  u, applied to each face

// The averaged surface of one wall run.  Direction is exact integer; position is an exact
// rational held as `(num_k, num_m) / den`.
pub struct WallSurface {
  pub ws_dk:    integer,   // summed edge vector, doubled coordinates — exact
  pub ws_dm:    integer,
  pub ws_num_k: integer, // mean midpoint = (num_k / den, num_m / den) in doubled coordinates
  pub ws_num_m: integer,
  pub ws_den:   integer, // 2 * n
  pub ws_n:     integer, // edges averaged
}

// The two corners bounding the boundary edge between cell (qa,ra) and its outside neighbour.
fn edge_corners_of(qa: integer, ra: integer, qb: integer, rb: integer) -> (integer, integer) {
  hex_edge_corners(hex_neighbor_dir(qa, ra, qb, rb))
}

// THE DERIVATION.  Sum the edge vectors; sum the doubled midpoints.  Both are integer sums, so the
// surface is exact before any float is involved.
//
// Example: @HXD-001
pub fn surface_of(p: Plan, cells: HexSet, side: integer) -> WallSurface {
  sr = side_edges(p, cells, side);
  n = side_run_len(sr);
  sdk = 0;
  sdm = 0;
  smk = 0;
  smm = 0;
  for si in 0..n {
    qa = sr.sr_qa[si] ?? 0;
    ra = sr.sr_ra[si] ?? 0;
    (c1, c2) = edge_corners_of(qa, ra, sr.sr_qb[si] ?? 0, sr.sr_rb[si] ?? 0);
    sdk = sdk + corner_k(c2) - corner_k(c1);
    sdm = sdm + corner_m(c2) - corner_m(c1);
    smk = smk + 2 * lattice_k(qa, ra) + corner_k(c1) + corner_k(c2);
    smm = smm + 2 * lattice_m(qa, ra) + corner_m(c1) + corner_m(c2);
  }
  WallSurface { ws_dk: sdk, ws_dm: sdm, ws_num_k: smk, ws_num_m: smm,
                ws_den: 2 * n, ws_n: n }
}

// Which of the 12 HEADINGS is the mean direction an exact positive multiple of?  -1 if none.
//
// This is the exactness claim: the averaged direction equals the nominal side direction EXACTLY,
// as an integer multiple of a heading step.  Any other value would mean the average had drifted
// off the lattice, which is what a fit would do and an average must not.
//
// PARALLEL, not an integer multiple, and not "one component is zero".  Both of those tests
// reject geometry that is exact:
//
//   "one component is zero"   only 4 of the 12 headings have one — a side running NE is `(1,3)`,
//                             both components non-zero and perfectly exact.  Failed 16 of 24.
//   "an integer multiple"     the summed vector is `(0,14)` where the heading step is `(0,6)`;
//                             the SUM over a run is not a whole number of steps.  Failed 12 of 24.
//
// What §6.1 actually claims is that the DIRECTION is exact — no residual perpendicular component —
// which is a zero CROSS PRODUCT against the heading step, in exact integers.  The magnitude is the
// run's length and carries no claim.
//
// Example: @HXD-002
pub fn surface_heading(w: WallSurface) -> integer {
  for sh in 0..HEAD_N {
    st = head_step(sh);
    if w.ws_dk * st.1 - w.ws_dm * st.0 == 0 {
      if w.ws_dk * st.0 + w.ws_dm * st.1 > 0 { return sh; }
    }
  }
  -1
}

// Does this surface run along one of the exact headings?  An inexact one is still drawable
// but cannot be reasoned about on the lattice, so ask before treating it as integer.
pub fn surface_is_exact(w: WallSurface) -> boolean { surface_heading(w) >= 0 }

// World position of the mean, and the unit perpendicular to the mean direction.
pub fn surface_mean_x(w: WallSurface) -> float {
  (((w.ws_num_k as float) / (w.ws_den as float)) ?? 0.0) * 0.8660254037844386
}

// The surface's mean point in world y — the companion of `surface_mean_x`, and where a
// label or a normal is anchored.
pub fn surface_mean_y(w: WallSurface) -> float {
  (((w.ws_num_m as float) / (w.ws_den as float)) ?? 0.0) * 0.5
}

// The surface's direction as a unit vector in world units ((0, 0) for a surface of no edges).
pub fn surface_dir(w: WallSurface) -> (float, float) {
  wx = (w.ws_dk as float) * 0.8660254037844386;
  wy = (w.ws_dm as float) * 0.5;
  ln = sqrt(wx * wx + wy * wy);
  ((wx / ln) ?? 0.0, (wy / ln) ?? 0.0)
}

fn perp_x(w: WallSurface) -> float {
  wx = (w.ws_dk as float) * 0.8660254037844386;
  wy = (w.ws_dm as float) * 0.5;
  ln = sqrt(wx * wx + wy * wy) ?? 1.0;
  (0.0 - wy / ln) ?? 0.0
}

fn perp_y(w: WallSurface) -> float {
  wx = (w.ws_dk as float) * 0.8660254037844386;
  wy = (w.ws_dm as float) * 0.5;
  ln = sqrt(wx * wx + wy * wy) ?? 1.0;
  (wx / ln) ?? 0.0
}

// The perpendicular extent of the run's CORNERS about the mean line — the band of §6.2.
//
// Example: @HXD-001
pub fn surface_band_corners(p: Plan, cells: HexSet, side: integer) -> float {
  w = surface_of(p, cells, side);
  sr = side_edges(p, cells, side);
  px = perp_x(w);
  py = perp_y(w);
  cx = surface_mean_x(w);
  cy = surface_mean_y(w);
  lo = 1.0e9;
  hi = -1.0e9;
  for bi in 0..side_run_len(sr) {
    qa = sr.sr_qa[bi] ?? 0;
    ra = sr.sr_ra[bi] ?? 0;
    (c1, c2) = edge_corners_of(qa, ra, sr.sr_qb[bi] ?? 0, sr.sr_rb[bi] ?? 0);
    for wi in 0..2 {
      cc = c1;
      if wi == 1 { cc = c2; }
      ex = ((lattice_k(qa, ra) + corner_k(cc)) as float) * 0.8660254037844386;
      ey = ((lattice_m(qa, ra) + corner_m(cc)) as float) * 0.5;
      ev = (ex - cx) * px + (ey - cy) * py;
      if ev < lo { lo = ev; }
      if ev > hi { hi = ev; }
    }
  }
  hi - lo
}

// ── G2: THE WALL AS *ONE* QUAD ─────────────────────────────────────────────
//
// `G2` asks that every wall render as **one flat quad — its analytic surface — not the strip**.
// Today the shot draws the wall *as it is stored*: one strip per boundary edge, so a side that is
// nine edges long is nine quads that zigzag by the band `X47` measured.  The surface is already
// exact (direction exactly a heading, position an exact rational); what was missing is its EXTENT,
// so a renderer can emit the segment instead of the strip.
//
// The span is the run's corners **projected onto the mean line**, taking the extremes.  The two
// points returned lie ON the line by construction, so the quad the renderer builds from them has
// perpendicular spread exactly **0** — which is `G2`'s check — while the strip it replaces spreads
// by `surface_band_corners`.
//
// Example: @HXD-003
pub fn surface_span(p: Plan, cells: HexSet, side: integer) -> (float, float, float, float) {
  w = surface_of(p, cells, side);
  sr = side_edges(p, cells, side);
  dx = (w.ws_dk as float) * 0.8660254037844386;
  dy = (w.ws_dm as float) * 0.5;
  dl = sqrt(dx * dx + dy * dy) ?? 1.0;
  ux = (dx / dl) ?? 1.0;
  uy = (dy / dl) ?? 0.0;
  cx = surface_mean_x(w);
  cy = surface_mean_y(w);
  lo = 1.0e9;
  hi = -1.0e9;
  for si in 0..side_run_len(sr) {
    qa = sr.sr_qa[si] ?? 0;
    ra = sr.sr_ra[si] ?? 0;
    (c1, c2) = edge_corners_of(qa, ra, sr.sr_qb[si] ?? 0, sr.sr_rb[si] ?? 0);
    for sj in 0..2 {
      cc = c1;
      if sj == 1 { cc = c2; }
      ex = ((lattice_k(qa, ra) + corner_k(cc)) as float) * 0.8660254037844386;
      ey = ((lattice_m(qa, ra) + corner_m(cc)) as float) * 0.5;
      tv = (ex - cx) * ux + (ey - cy) * uy;
      if tv < lo { lo = tv; }
      if tv > hi { hi = tv; }
    }
  }
  (cx + lo * ux, cy + lo * uy, cx + hi * ux, cy + hi * uy)
}

// The perpendicular spread of the FITTED quad's own corners about its line — `G2`'s check.  Both
// endpoints come off the line, so this is the number `eave_spread(fitted) = 0` names.  Measured
// rather than asserted, so that if the span ever stops lying on the line the gate says so.
//
// ⚠ *EXACTLY* 0 IS A CLAIM ABOUT FLOATS, AND IT IS ORIENTATION-DEPENDENT.  The derivation above is
// exact — integer sums and rationals — but the span is projected through world coordinates, and on
// the orientations where that arithmetic does not cancel the round trip costs an ulp.  Measured
// over 12 orientations x 4 sides: 32 read exactly 0 and 16 read 2.22e-16.  So `== 0.0` passes only
// on the orientations it was tried on; what holds everywhere is the comparison that means
// something — flat against the STRIP's band, which is fifteen orders of magnitude larger.
//
// Example: @HXD-001
pub fn surface_fitted_spread(p: Plan, cells: HexSet, side: integer) -> float {
  w = surface_of(p, cells, side);
  (sx0, sy0, sx1, sy1) = surface_span(p, cells, side);
  px = perp_x(w);
  py = perp_y(w);
  cx = surface_mean_x(w);
  cy = surface_mean_y(w);
  e0 = (sx0 - cx) * px + (sy0 - cy) * py;
  e1 = (sx1 - cx) * px + (sy1 - cy) * py;
  lo = e0;
  hi = e0;
  if e1 < lo { lo = e1; }
  if e1 > hi { hi = e1; }
  hi - lo
}

// THE MITER POINT (`I-CORNER` part 4): where two adjacent fitted surfaces intersect.  This is the
// corner the RENDER should use — each quad running miter-to-miter, so consecutive sides share an
// endpoint exactly and the outline closes with no gap.  That closure is the invariant, and it is
// exact.
//
// ⚠ IT IS NOT THE `Plan`'S CORNER, and it never can be.  An earlier note here claimed the
// intersection *"must land on the exact model corner, or the surfaces have drifted off the shape
// they were fitted to"* — a check nobody ran.  Measured, it misses by 0.47 world units on the 5x4
// cottage, and swept over plan sizes the miss is neither constant (0.465 .. 0.827) nor even
// consistently outward.  The reason is structural rather than numerical: what these surfaces
// recover is the face of the wall AS DRAWN, and the drawn footprint is the plan quantised to
// cells, so the recovered corner is an exact rational of the lattice while the plan's half-depth
// is an irrational multiple of `sqrt(3)/2`.  Two number systems that meet only at the origin.
//
// So a caller must never diff a recovered outline against plan geometry, and must not try to
// correct for the offset either.  Use the recovered corner for anything that has to line up with
// what was drawn.
//
// `ok` is false when the two lines are parallel, which for adjacent sides of a closed form cannot
// happen — so a false here is a real defect, not a degenerate input to be shrugged at.
//
// Example: @HXD-003
pub fn surface_miter(p: Plan, cells: HexSet, sa: integer, sb: integer)
                     -> (float, float, boolean) {
  surface_meet(surface_of(p, cells, sa), surface_of(p, cells, sb))
}

// Where two surfaces' lines meet, in world units: `surface_miter` for surfaces that are not two
// sides of one `Plan` — the walls of a structure known only by its cells, an L's, a road's.
// `ok` is false when the lines are parallel.
pub fn surface_meet(wa: WallSurface, wb: WallSurface) -> (float, float, boolean) {
  ax = surface_mean_x(wa);
  ay = surface_mean_y(wa);
  bx = surface_mean_x(wb);
  by = surface_mean_y(wb);
  aux = (wa.ws_dk as float) * 0.8660254037844386;
  auy = (wa.ws_dm as float) * 0.5;
  bux = (wb.ws_dk as float) * 0.8660254037844386;
  buy = (wb.ws_dm as float) * 0.5;
  den = aux * buy - auy * bux;
  if den < 0.000000001 && den > 0.0 - 0.000000001 { return (0.0, 0.0, false); }
  tt = (((bx - ax) * buy - (by - ay) * bux) / den) ?? 0.0;
  (ax + tt * aux, ay + tt * auy, true)
}

// THE MITERED QUAD (`G2` + `I-CORNER` part 4): the side's segment running MITER TO MITER, so
// consecutive sides share an endpoint exactly and the outline closes with no corner gap.  This is
// what a renderer should draw; `surface_span` (the projected extent) leaves the gaps the fitted
// render made visible.
//
// Example: @HXD-003
pub fn surface_quad(p: Plan, cells: HexSet, side: integer, nsides: integer)
                    -> (float, float, float, float) {
  // `%` is FALLIBLE in loft, so `nsides = 0` types `prev`/`next` as `integer?` — and a null index
  // arriving at `surface_miter` would read as a valid-looking side **0** rather than a refusal: a
  // silently wrong corner instead of a caught one.  A miter needs `prev != side != next`, so it
  // needs at least three sides; below that there is no corner to cut and the honest answer is the
  // un-mitered span, which is already this function's defined escape when a miter fails.
  //
  // Every caller passes 4, so this cannot fire today — but *"this cannot fire"* is a claim like
  // any other (`X57`: `wall_separates`' comment said exactly that, was wrong, and was
  // load-bearing under a green suite).  The `?? 0` below is unreachable PAST this guard; the
  // guard is what refuses, the `??` only discharges the type.
  if nsides < 3 { return surface_span(p, cells, side); }
  prev = ((side - 1) % nsides + nsides) % nsides ?? 0;
  next = (side + 1) % nsides ?? 0;
  (ax, ay, aok) = surface_miter(p, cells, prev, side);
  (bx, by, bok) = surface_miter(p, cells, side, next);
  if !aok || !bok { return surface_span(p, cells, side); }
  (ax, ay, bx, by)
}

// One stored edge's INTERVAL along the fitted line — where edge `i` of the run projects onto the
// surface.  This is what makes a feature renderable as `FEATURES` describes it: *"a door is
// `[s0,s1]` along an exact straight or arc"*, an interval on the analytic surface rather than a
// stored edge.  The renderer draws the wall as one quad (`surface_span`) and each feature as a
// sub-interval of it, so the zigzag never reaches the picture.
//
// Example: @HXD-006
pub fn surface_edge_span(p: Plan, cells: HexSet, side: integer, i: integer)
                         -> (float, float, float, float) {
  w = surface_of(p, cells, side);
  sr = side_edges(p, cells, side);
  dx = (w.ws_dk as float) * 0.8660254037844386;
  dy = (w.ws_dm as float) * 0.5;
  dl = sqrt(dx * dx + dy * dy) ?? 1.0;
  ux = (dx / dl) ?? 1.0;
  uy = (dy / dl) ?? 0.0;
  cx = surface_mean_x(w);
  cy = surface_mean_y(w);
  qa = sr.sr_qa[i] ?? 0;
  ra = sr.sr_ra[i] ?? 0;
  (c1, c2) = edge_corners_of(qa, ra, sr.sr_qb[i] ?? 0, sr.sr_rb[i] ?? 0);
  lo = 1.0e9;
  hi = -1.0e9;
  for ej in 0..2 {
    cc = c1;
    if ej == 1 { cc = c2; }
    ex = ((lattice_k(qa, ra) + corner_k(cc)) as float) * 0.8660254037844386;
    ey = ((lattice_m(qa, ra) + corner_m(cc)) as float) * 0.5;
    tv = (ex - cx) * ux + (ey - cy) * uy;
    if tv < lo { lo = tv; }
    if tv > hi { hi = tv; }
  }
  (cx + lo * ux, cy + lo * uy, cx + hi * ux, cy + hi * uy)
}

// The material on stored edge `i` of a side — so the renderer can find the features on a run
// without re-deriving the run's order.
pub fn surface_edge_mat(p: Plan, cells: HexSet, edges: EdgeSet, side: integer,
                        i: integer) -> integer {
  sr = side_edges(p, cells, side);
  edge_mat(edges, sr.sr_qa[i] ?? 0, sr.sr_ra[i] ?? 0, sr.sr_qb[i] ?? 0, sr.sr_rb[i] ?? 0)
}

// The perpendicular extent of the MIDPOINTS about the mean line.  Not in §6.2, and it is not the
// same as the corner band: on the east family the midpoints are exactly COLINEAR (0), while on the
// north family they span the full band — the row stagger, which the corners hide.
//
// Example: @HXD-001
pub fn surface_band_mids(p: Plan, cells: HexSet, side: integer) -> float {
  w = surface_of(p, cells, side);
  sr = side_edges(p, cells, side);
  px = perp_x(w);
  py = perp_y(w);
  cx = surface_mean_x(w);
  cy = surface_mean_y(w);
  lo = 1.0e9;
  hi = -1.0e9;
  for mi in 0..side_run_len(sr) {
    qa = sr.sr_qa[mi] ?? 0;
    ra = sr.sr_ra[mi] ?? 0;
    (c1, c2) = edge_corners_of(qa, ra, sr.sr_qb[mi] ?? 0, sr.sr_rb[mi] ?? 0);
    mx = ((lattice_k(qa, ra) as float)
          + ((corner_k(c1) + corner_k(c2)) as float) / 2.0) * 0.8660254037844386;
    my = ((lattice_m(qa, ra) as float)
          + ((corner_m(c1) + corner_m(c2)) as float) / 2.0) * 0.5;
    mv = (mx - cx) * px + (my - cy) * py;
    if mv < lo { lo = mv; }
    if mv > hi { hi = mv; }
  }
  hi - lo
}

// The scatter a LEAST-SQUARES formulation would have to threshold: the sum of squared
// perpendicular deviations of the midpoints from the derived mean line.  Averaging names the same
// geometry as an exact band constant and has nothing to tune; this is the control that the
// distinction is real rather than rhetorical.
//
// Example: @HXD-001
pub fn surface_lsq_residual(p: Plan, cells: HexSet, side: integer) -> float {
  w = surface_of(p, cells, side);
  sr = side_edges(p, cells, side);
  px = perp_x(w);
  py = perp_y(w);
  cx = surface_mean_x(w);
  cy = surface_mean_y(w);
  acc = 0.0;
  for ri in 0..side_run_len(sr) {
    qa = sr.sr_qa[ri] ?? 0;
    ra = sr.sr_ra[ri] ?? 0;
    (c1, c2) = edge_corners_of(qa, ra, sr.sr_qb[ri] ?? 0, sr.sr_rb[ri] ?? 0);
    mx = ((lattice_k(qa, ra) as float)
          + ((corner_k(c1) + corner_k(c2)) as float) / 2.0) * 0.8660254037844386;
    my = ((lattice_m(qa, ra) as float)
          + ((corner_m(c1) + corner_m(c2)) as float) / 2.0) * 0.5;
    dv = (mx - cx) * px + (my - cy) * py;
    acc = acc + dv * dv;
  }
  acc
}

// ── A5 — FEATURES ON A STRAIGHT SIDE, AND WHAT OF THEM SURVIVES ────────────
//
// `OD-9` is closed: a door or window IS the material on the wall slot, not an annotation beside
// one (`ROUNDTRIP` §2.4.1).  The edge is never removed, so a run cannot fragment — the doored-tower
// defect is structurally impossible rather than merely avoided.  A5 asks the next question: does
// the feature survive the ROUND TRIP?
//
// THE MODEL STORES `(side, t)` (`I2`, affine-invariant).  THE FIELD STORES A MATERIAL ON EDGES.
// So recovery has to get `t` back out of a set of marked edges — and it cannot get an arbitrary
// `t` back, because every `t` in an interval of width `1/n` selects the same edge.  What IS exact:
//
//     the edge centres along a run of n edges sit at   t = (2i + 1) / 2n
//
// exact rationals, odd numerator over `2n`.  `place_opening` snaps to the nearest, so a feature
// authored at one of those `t` values round-trips exactly and one authored between them does not.
// That is the same shape as the line-endpoint rule (§10.10): the doorstep quantises the model to
// what the storage can distinguish, rather than letting a value in that cannot come back.
//
// The run is NOT stored in `t` order — `side_edges` fills it in cell-scan order — so a feature is
// indexed by its exact `t` numerator, never by its position in the run.

// The exact `t` of stored edge `i`, as the numerator over `2n`: an odd integer in `1 .. 2n-1`.
// `sr_t` is a float only because `SideRun` holds it that way; the value is an exact rational and
// this recovers the numerator without a tolerance being involved in the answer.
//
// Example: @HXD-006
pub fn feature_t_num(sr: SideRun, i: integer, n: integer) -> integer {
  (round((sr.sr_t[i] ?? 0.0) * 2.0 * (n as float)) ?? 0.0) as integer
}

// Is `num/2n` a legal feature position — an edge centre, i.e. an ODD numerator in range?
pub fn feature_t_admissible(num: integer, n: integer) -> boolean {
  num > 0 && num < 2 * n && ((num % 2) + 2) % 2 == 1
}

// Recover a feature from the field: the `t` numerators of every edge on `side` carrying `kind`,
// in ascending `t` order (C2's ordering, not the storage order).
//
// Example: @HXD-006
pub fn feature_find(p: Plan, cells: HexSet, edges: EdgeSet,
                    side: integer, kind: integer) -> vector<integer> {
  sr = side_edges(p, cells, side);
  n = side_run_len(sr);
  out: vector<integer> = [];
  for fi in 0..n {
    if edge_mat(edges, sr.sr_qa[fi] ?? 0, sr.sr_ra[fi] ?? 0,
                       sr.sr_qb[fi] ?? 0, sr.sr_rb[fi] ?? 0) == kind {
      out += [feature_t_num(sr, fi, n)];
    }
  }
  // ascending t — insertion sort, the run is tens of edges at most
  for si in 1..len(out) {
    ps = si;
    while ps > 0 && (out[ps - 1] ?? 0) > (out[ps] ?? 0) {
      tmp = out[ps - 1] ?? 0;
      out[ps - 1] = out[ps] ?? 0;
      out[ps] = tmp;
      ps = ps - 1;
    }
  }
  out
}

// How many boundary edges carry a wall material at all — the run's length, which a re-materialled
// feature must leave UNCHANGED.  Deleting an edge instead is what fragments a run.
//
// Example: @HXD-005
pub fn wall_edge_count(cells: HexSet, edges: EdgeSet) -> integer {
  n = 0;
  for wr in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for wq in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if hexset_get(cells, wq, wr) {
        // ALL SIX directions, as `draw_walls` does.  A boundary edge is visited exactly once
        // because only the inside cell is in the set — iterating just the three OWNED directions
        // misses every edge the outside cell owns and undercounts by about half (38 read as 19).
        for wd in 0..6 {
          (nq, nr) = hex_neighbor(wq, wr, wd);
          if !hexset_get(cells, nq, nr) {
            if edge_mat(edges, wq, wr, nq, nr) != 0 { n = n + 1; }
          }
        }
      }
    }
  }
  n
}

// Copyright (c) 2026 Jurjen Stellingwerff

// housedraw — DRAW A BUILDING IN ANY OF THE 12 ORIENTATIONS.
//
// The field primitives are hex_field's (HexSet/Labels/EdgeSet/Heights) and the roof
// profiles hex_roof's.  This is the layer above them: a LOCAL FRAME (hex_form's `Plan`) so
// the same building can be drawn at 6 rotations x 2 chiralities, and openings placed on its
// walls by (side, t) rather than by cell coordinates.
//
// FOUR ROUTINES, and each writes exactly one layer:
//
//   draw_floor    the massing  -> HexSet + Labels     (the floor you stand on)
//   draw_walls    its boundary -> EdgeSet             (thin walls: no floor is lost)
//   place_opening n edges at t -> EdgeSet material    (doors and windows)
//   draw_roof     a ridge      -> Heights             (a thin caller of hexroof)
//
// THE TWO RULES THAT MAKE 12 ORIENTATIONS WORK:
//
//  1. THE MIRROR IS APPLIED ONCE, IN THE FRAME.  Local -> world negates local u when
//     mirrored; a side's endpoints then only EXCHANGE.  Applying it in both places turns
//     every outward normal INWARD (measured: -1.000 at all 12 orientations).
//
//  2. A WALL IS THE BOUNDARY OF A FILLED REGION, never a band of cells buffered along a
//     line.  The boundary of a region is a closed loop BY CONSTRUCTION at any size,
//     orientation or tie; the buffer rule leaks (measured: 2 components, 0 cells enclosed
//     — a wall with a hole — while still passing an equivariance check 12/12).
//
// A DOOR IS A MATERIAL ON ITS EDGES, AND NEVER DELETES THEM.  Deleting fragments the wall
// run, which is the defect DRAWING.md names as a doored tower fitting 3 arcs instead of 1.
// The foxel stores ONE material per wall slot (ROUNDTRIP.md §2.4), so an opening REPLACES
// the wall material rather than sitting in a channel beside it.  Composition therefore
// belongs in the material vocabulary — "a door in a cottage wall" is a material — which is
// what the consumer reads to decide passability.
use hex_field::*;
use hex_form::*;
use hex_way::*;
use hex_roof::*;
use hex_grid::*;

const EPS = 0.000000001;

// Opening materials — values in the SAME vocabulary as the wall materials, because they
// share the one slot.  A door overrides `solid`; a window does not (it overrides opacity,
// which is an elevation property and never touches the footprint).
//
// ⚠ `OPEN_NONE` IS NOT "no opening" — it is the wall-material value **0, which means NO WALL**.
// Writing it DELETES the edge and breaks the run in two (`X51`: 37 edges, 2 dangling ends, and on
// a tower 3 arcs instead of 1).  It is kept because 0 is the slot's empty value, but it is never
// what an author means by an opening.
//
// THE TAXONOMY.  A door is not a gap in the wall; it is something that can be a collider or
// not.  A real gap is allowed too — the door itself is missing, but the wall continues as normal.
// Three distinct things live on a wall edge, and **all three keep the wall one continuous run**:
//
//   OPEN_DOOR    an opening fitted with a door leaf — a collider WHEN CLOSED, not when open
//   OPEN_WINDOW  an opening at elevation — never a floor-level collider, wall continues
//   OPEN_GAP     a real gap: the leaf is missing entirely, never a collider, wall continues
//
// The difference between `OPEN_GAP` and `OPEN_NONE` is the whole point: a gap is an **opening in a
// wall that is still there**; `OPEN_NONE` is the **absence of the wall**.
//
// THE RATIONALE, and it is what makes this not a matter of taste: a gap or a door is never
// rendered as a missing wall — there is a door frame or a ragged stone opening there.  An
// opening **has geometry of its own** — a frame, a lintel, a ragged
// jamb.  Absence has none.  So an opening must name a material that carries that geometry, which
// is precisely a wall id into the palette; writing 0 would ask the renderer to draw nothing and the
// collision layer to find no wall, both wrong.
pub const OPEN_NONE   = 0;
pub const OPEN_DOOR   = 1;
pub const OPEN_WINDOW = 2;
pub const OPEN_GAP    = 3;


// ── 1. THE FLOOR ────────────────────────────────────────────────────────────
//
// The massing is filled FIRST and walled second.  Comparisons are inclusive (<= bound
// + eps) so the fill never loses a column of cells to the structural tie: along u every
// cell-to-side distance is an exact multiple of sqrt(3)/2, which is exactly the default
// halfwidth, so a strict comparison sits on the last bit of a float for a whole column.
pub fn draw_floor(p: Plan, cells: HexSet, types: Labels, floor_type: integer) -> integer {
  hw = plan_hw(p);
  hd = plan_hd(p);
  n = 0;
  for fr in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for fq in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      (lu, lv) = plan_to_local(p, cell_x(fq, fr), cell_y(fq, fr));
      if fabs(lu) <= hw + EPS && fabs(lv) <= hd + EPS {
        hexset_set(cells, fq, fr, true);
        label_set(types, fq, fr, floor_type);
        n = n + 1;
      }
    }
  }
  n
}

// ── 2. THE WALLS ────────────────────────────────────────────────────────────
//
// Every edge between a cell of the region and one outside it.  Closed by construction,
// and it costs no floor at all: a 7.5 x 6.0 m cottage keeps all 27 of its cells, where
// a wall drawn as a band of cells would eat 16 of them — 59% of the house.
//
// Example: @HXD-005
pub fn draw_walls(cells: HexSet, edges: EdgeSet, wall_mat: integer) -> integer {
  n = 0;
  for wr in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for wq in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if hexset_get(cells, wq, wr) {
        for d in 0..6 {
          nq = nb_q(wq, wr, d);
          nr = nb_r(wq, wr, d);
          if !hexset_get(cells, nq, nr) {
            edge_set_mat(edges, wq, wr, nq, nr, wall_mat);
            n = n + 1;
          }
        }
      }
    }
  }
  n
}


// ── 3. OPENINGS ─────────────────────────────────────────────────────────────
//
// A door is N EDGES centred at t, not a t-interval erased with a segment: erasing
// over-erases by ~halfwidth at EACH end (a 3.00 m interval opens 4.50 m of wall) and
// shrinking the segment over-corrects.  One hex edge is 1.0 world unit = 0.87 m of
// clear width — a real domestic door, which a cell-thick wall could not express at all.
//
// The opening material REPLACES the wall material on those edges (the slot holds one).
// With any kind but `OPEN_NONE` nothing is deleted, so the wall still fits as one run.
// Returns how many edges it wrote: fewer than `nedges` when the side is shorter.
//
// Example: @HXD-004
pub fn place_opening(p: Plan, cells: HexSet, edges: EdgeSet,
                     side: integer, t: float, nedges: integer, kind: integer) -> integer {
  sr = side_edges(p, cells, side);
  total = len(sr.sr_t);
  if total == 0 { return 0; }
  taken: vector<boolean> = [for _ in 0..total { false }];
  placed = 0;
  for _pick in 0..nedges {
    best = -1;
    bestd = 0.0;
    for i in 0..total {
      if !(taken[i] ?? true) {
        d = fabs((sr.sr_t[i] ?? 0.0) - t);
        if best < 0 || d < bestd { best = i; bestd = d; }
      }
    }
    if best >= 0 {
      taken[best] = true;
      edge_set_mat(edges, sr.sr_qa[best] ?? 0, sr.sr_ra[best] ?? 0,
                          sr.sr_qb[best] ?? 0, sr.sr_rb[best] ?? 0, kind);
      placed = placed + 1;
    }
  }
  placed
}

// ── 4. THE ROOF ─────────────────────────────────────────────────────────────
//
// One ring of halo around the massing, because that ring is what makes the EAVE
// interpolable: without it 38 of the 76 wall-edge samples face a cell with no stored
// height, and the wall top has nothing to meet.
//
// Example: @HXD-007
pub fn grow_ring(src: HexSet, dst: HexSet) {
  for gr in hexset_r0(src)..hexset_r0(src) + hexset_h(src) {
    for gq in hexset_q0(src)..hexset_q0(src) + hexset_w(src) {
      if hexset_get(src, gq, gr) {
        hexset_set(dst, gq, gr, true);
        for gd in 0..6 { hexset_set(dst, nb_q(gq, gr, gd), nb_r(gq, gr, gd), true); }
      }
    }
  }
}

// A gable: the ridge runs along local u THROUGH the anchor and PAST both gable ends.
// Stopping it at the wall makes the distance go radial past the tip, so the gable end
// rolls over like a cone quadrant — 0.43 units, 0.37 m of bulge.  A hip is the same
// call with the ridge shortened; then the conical ends are the feature, not a defect.
//
// Example: @HXD-007
pub fn draw_roof(p: Plan, cells: HexSet, hz: Heights,
                 eave: float, pitch: float, hip_steps: integer) {
  hw = plan_hw(p);
  hd = plan_hd(p);
  half = hw + hd + HEX_LEN;                       // gable: run well past the ends
  if hip_steps > 0 { half = hw - (hip_steps as float) * HEX_LEN * 0.5; }
  (x0, y0) = plan_to_world(p, -half, 0.0);
  (x1, y1) = plan_to_world(p, half, 0.0);
  tr = track_new();
  track_straight(tr, x0, y0, x1, y1);
  roof_ridge(cells, hz, tr, eave + pitch * hd, pitch, 1000000.0);
}

// Against `surface_of`, which is told the `Plan`: a TOPS wall (an even heading) is the same line,
// exactly — the two rationals name one point across it.  A SIDES wall (an odd heading) is not:
// `surface_of` averages every edge its side holds, the corner's with them, where `cells_wall`
// reads the line off the slanted connectors and outvotes the corner's.  On the 5 x 4 box they
// lie 0.0481 world units (4 cm) apart.  Both are exact; they answer two questions.
fn test_a_wall_from_cells_against_the_plans_surface() {
  for rot in 0..6 {
    p = plan_new(0, 0, 5, 4, rot, false);
    cells = box_cells(rot, false);
    b = cells_built(cells, 9);
    tops = 0;
    sides = 0;
    for side in 0..4 {
      s = surface_of(p, cells, side);
      for w in b.cw_walls {
        if surface_heading(w) != surface_heading(s) { continue; }
        // the offset of s's mean from w's line, x 4 x both denominators x |w's step|: exact
        ek = s.ws_num_k * w.ws_den - w.ws_num_k * s.ws_den;
        em = s.ws_num_m * w.ws_den - w.ws_num_m * s.ws_den;
        cross = 3 * ek * perp_k(w.ws_dk, w.ws_dm) + em * perp_m(w.ws_dk, w.ws_dm);
        if head_is_edge(surface_heading(w)) {
          assert(cross == 0, "rot {rot} side {side}: a tops wall is surface_of's line");
          tops += 1;
        } else {
          off = abs(left_of(w, surface_mean_x(s), surface_mean_y(s)));
          assert(cross != 0 && off > 0.0481 && off < 0.0482, "rot {rot} side {side}: a sides wall lies 0.0481 beside surface_of's ({off})");
          sides += 1;
        }
      }
    }
    assert(tops == 2 && sides == 2, "rot {rot}: two walls of each kind, got {tops} and {sides}");
  }
}

// Is `p` (a coordinate along heading `head`, scaled) between the lines `lo` and `hi` half steps out?
fn between(head: integer, p: integer, lo: integer, hi: integer) -> boolean {
  if head_is_edge(head) { return p >= 12 * lo && p <= 12 * hi; }
  sign_root3(p, -12 * lo) >= 0 && sign_root3(p, -12 * hi) <= 0
}

// An L along heading `h`, by exact tests: a body 5 hex steps by 3, a wing 2 by 2 on one end of
// its far side (in half steps: u -5 .. 5, v -3 .. 3 and u -5 .. -1, v 3 .. 7).
fn ell_cells(h: integer) -> HexSet {
  cells = hexset_chunk(-20, -20, 41, 41);
  us = head_step(h);
  vs = head_step(h + 3);
  for r in -20..21 {
    for q in -20..21 {
      u = vec_dot_4(2 * lattice_k(q, r), 2 * lattice_m(q, r), us.0, us.1);
      v = vec_dot_4(2 * lattice_k(q, r), 2 * lattice_m(q, r), vs.0, vs.1);
      body = between(h, u, -5, 5) && between(h + 3, v, -3, 3);
      wing = between(h, u, -5, -1) && between(h + 3, v, 3, 7);
      if body || wing { hexset_set(cells, q, r, true); }
    }
  }
  cells
}

// How many floor cells touch the outside — the ring a wall drawn as a BAND OF CELLS

fn ring_cells(cells: HexSet) -> integer {
  n = 0;
  for r in R0..R0 + H {
    for q in Q0..Q0 + W {
      if hexset_get(cells, q, r) {
        out = false;
        for d in 0..6 { if !hexset_get(cells, nb_q(q, r, d), nb_r(q, r, d)) { out = true; } }
        if out { n = n + 1; }
      }
    }
  }
  n
}

// The wall-edge count written the tempting way: only the three directions a cell
// "owns", so every edge the OUTSIDE cell owns is missed.  Parameterised by which of
// the two neighbour conventions is used, because they disagree about which three.
fn count_three_dirs(cells: HexSet, edges: EdgeSet, hexgrid_order: boolean) -> integer {
  n = 0;
  for r in R0..R0 + H {
    for q in Q0..Q0 + W {
      if hexset_get(cells, q, r) {
        for d in 0..3 {
          nq = nb_q(q, r, d);
          nr = nb_r(q, r, d);
          if hexgrid_order { (nq, nr) = hex_neighbor(q, r, d); }
          if !hexset_get(cells, nq, nr) && edge_mat(edges, q, r, nq, nr) != 0 { n = n + 1; }
        }
      }
    }
  }
  n
}

// The distance between the end of side `a`'s drawn segment and the start of side `b`'s.
fn span_gap(p: Plan, cells: HexSet, a: integer, b: integer) -> float {
  (_x0, _y0, x1, y1) = surface_span(p, cells, a);
  (u0, v0, _u1, _v1) = surface_span(p, cells, b);
  sqrt((x1 - u0) * (x1 - u0) + (y1 - v0) * (y1 - v0)) ?? 0.0
}

fn quad_gap(p: Plan, cells: HexSet, a: integer, b: integer) -> float {
  (_x0, _y0, x1, y1) = surface_quad(p, cells, a, 4);
  (u0, v0, _u1, _v1) = surface_quad(p, cells, b, 4);
  sqrt((x1 - u0) * (x1 - u0) + (y1 - v0) * (y1 - v0)) ?? 0.0
}

// A ridge along the plan's local u, extending `half` either side of the anchor.

// @HXD-001 — A WALL IS ONE FLAT QUAD, NOT THE STRIP IT IS STORED AS.  The field
// stores a wall as N separate boundary edges, and those edges ZIGZAG: they are
// the sides of hexagons, so consecutive ones step sideways by an exact amount.
// Draw them as they are stored and a nine-edge wall is nine quads with a visible
// comb; draw the analytic surface and it is one.
//
// The surface is DERIVED by averaging, never fitted — the direction is the sum of
// the edge vectors and the position the mean of their midpoints, both exact
// integer sums — so there is no residual anywhere to threshold.  What the two
// bands below measure is not error: it is the GEOMETRY of the strip, and it is an
// exact algebraic constant per family, `sqrt(3)` apart.
fn test_a_wall_is_one_quad_and_the_strip_it_replaces_has_a_width() {
  p = cottage();
  cells = cottage_cells();

  // The two families, and their exact bands.
  b0 = surface_band_corners(p, cells, 0);
  b1 = surface_band_corners(p, cells, 1);
  assert(abs(b0 - BAND_SIDES) < 1e-12, "a north-family side's strip is sqrt(3)/2 wide, got {b0}");
  assert(abs(b1 - BAND_TOPS) < 1e-12, "an east-family side's is 1/2, got {b1}");
  assert(abs((b0 / b1 ?? 0.0) - BAND_RATIO) < 1e-12, "and the ratio is exactly sqrt(3)");

  // The quad that replaces the strip has no width at all: both its endpoints come
  // off the derived line, so its perpendicular spread is zero.
  assert(surface_fitted_spread(p, cells, 0) == 0.0, "the fitted quad is flat");
  assert(surface_fitted_spread(p, cells, 1) == 0.0, "on both families");

  // …to within the float it is presented in, which is NOT the same as "exactly 0"
  // and is worth knowing before you write `== 0.0` into a gate.  The derivation is
  // exact (integers and rationals); the world coordinates are f64, and projecting
  // a point onto a line and back costs an ulp on the orientations where the
  // arithmetic does not cancel.  Over the 12 orientations x 4 sides, 32 read
  // exactly 0 and 16 read 2.22e-16 — fifteen orders of magnitude below the band
  // they are being distinguished from, which is the comparison that means something.
  worst = 0.0;
  nexact = 0;
  ninexact = 0;
  for rot in 0..6 {
    for mi in 0..2 {
      q = plan_new(0, 0, 5, 4, rot, mi == 1);
      c = patch();
      lb = labels_new(Q0, R0, W, H);
      draw_floor(q, c, lb, 1);
      for side in 0..4 {
        sp = surface_fitted_spread(q, c, side);
        if sp == 0.0 { nexact = nexact + 1; } else { ninexact = ninexact + 1; }
        if sp > worst { worst = sp; }
      }
    }
  }
  assert(nexact + ninexact == 48, "12 orientations x 4 sides");
  assert(ninexact > 0, "an `== 0.0` assertion passes only on the orientations you tried");
  assert(worst < 1e-15, "and the worst spread is still an ulp, got {worst}");
  assert((worst / BAND_TOPS ?? 1.0) < 1e-14, "which is the flat-vs-strip comparison that holds");

  // What the strip's zigzag actually is: on the east family the edge MIDPOINTS are
  // exactly colinear and the corners alternate; on the north family the midpoints
  // themselves stagger by the full band.  The corners hide that difference.
  assert(surface_band_mids(p, cells, 1) == 0.0, "east-family midpoints are colinear");
  assert(abs(surface_band_mids(p, cells, 0) - BAND_SIDES) < 1e-12,
         "north-family midpoints span the whole band — the row stagger");

  // And the control that "averaging, not fitting" is a real distinction: a least
  // squares formulation would have a residual here to threshold, and the average
  // has an exact answer instead.
  assert(surface_lsq_residual(p, cells, 1) == 0.0, "nothing to fit on the east family");
  assert(abs(surface_lsq_residual(p, cells, 0) - (11.0 / 12.0 ?? 0.0)) < 1e-12,
         "and exactly 11/12 to threshold on the north one, got {surface_lsq_residual(p, cells, 0)}");
}

// @HXD-002 — THE EXACTNESS CHECK, AND THE TWO OBVIOUS WAYS TO WRITE IT WRONG.
// The claim `surface_heading` makes is that the AVERAGED direction is exactly one
// of the twelve headings — no residual perpendicular component — which in integers
// is a zero CROSS PRODUCT against that heading's step.
//
// Both simpler-looking tests fail on geometry nobody doubts, and they fail in a
// way that reads like a real defect:
//
//   "one component is zero"   only 4 of the 12 headings have one; a side running
//                             NE is (1,3), both components non-zero and exact
//   "an integer multiple"     the summed vector over a nine-edge run is (0,14)
//                             where the heading step is (0,6) — the SUM is not a
//                             whole number of steps, and never claimed to be
//
// The magnitude carries no claim at all: it is the run's length.
fn test_exactness_is_a_cross_product_not_a_multiple() {
  p = cottage();
  cells = cottage_cells();

  w = surface_of(p, cells, 0);
  assert(surface_heading(w) == 3, "side 0 runs on heading 3");
  assert(w.ws_dk == 0 && w.ws_dm == 14, "its summed vector is (0,14), got ({w.ws_dk},{w.ws_dm})");
  st = head_step(3);
  assert(st.0 == 0 && st.1 == 6, "and heading 3's step is (0,6)");
  assert(w.ws_dk * st.1 - w.ws_dm * st.0 == 0, "the cross product is exactly zero — that is the claim");
  assert((14 % 6 ?? 0) != 0, "while 14 is not a multiple of 6 — the sum is not a whole number of steps");

  // Both wrong tests, counted over the six rotations x four sides.
  nexact = 0;
  fail_component = 0;
  fail_multiple = 0;
  for rot in 0..6 {
    q = plan_new(0, 0, 5, 4, rot, false);
    c = patch();
    lb = labels_new(Q0, R0, W, H);
    draw_floor(q, c, lb, 1);
    for side in 0..4 {
      v = surface_of(q, c, side);
      hd = surface_heading(v);
      if hd >= 0 { nexact = nexact + 1; }
      if !(v.ws_dk == 0 || v.ws_dm == 0) { fail_component = fail_component + 1; }
      sp = head_step(if hd >= 0 { hd } else { 0 });
      ismult = false;
      if sp.0 != 0 { ismult = (v.ws_dk % sp.0 ?? 1) == 0 && v.ws_dm * sp.0 == v.ws_dk * sp.1; }
      if sp.0 == 0 { ismult = (v.ws_dm % sp.1 ?? 1) == 0 && v.ws_dk == 0; }
      if !ismult { fail_multiple = fail_multiple + 1; }
    }
  }
  assert(nexact == 24, "every one of the 24 surfaces is exact by the cross product, got {nexact}");
  assert(fail_component == 16, "'one component is zero' rejects 16 of them, got {fail_component}");
  assert(fail_multiple == 12, "'an integer multiple' rejects 12 of them, got {fail_multiple}");
}

// @HXD-003 — THE MITER CLOSES THE OUTLINE, AND ITS CORNER IS NOT THE PLAN'S.
// Two adjacent surfaces are two lines; where they cross is the corner a renderer
// should cut to, and `surface_quad` returns each side miter to miter.  Then
// consecutive quads share an endpoint EXACTLY and the outline closes.  The
// un-mitered `surface_span` — each side's own extent projected onto its own line —
// leaves a real gap at every corner, and it is not a small one.
//
// What that corner is NOT is the corner of the `Plan`.  The recovered outline is
// the face of the wall as DRAWN, and the drawn footprint is the plan quantised to
// cells.  The two live in different number systems: the recovered corner is an
// exact rational of the lattice (a multiple of 3/4 across the rows), while the
// plan's half-depth is an irrational multiple of `sqrt(3)/2`.  They agree only at
// the origin — so this is not an offset a caller can correct for, and swept over
// plan sizes the difference is neither constant nor even consistently outward.
fn test_the_miter_closes_the_outline_but_is_not_the_plans_corner() {
  p = cottage();
  cells = cottage_cells();

  for side in 0..4 {
    nx = (side + 1) % 4 ?? 0;
    assert(quad_gap(p, cells, side, nx) == 0.0, "mitered sides {side}->{nx} share their endpoint");
    assert(abs(span_gap(p, cells, side, nx) - 0.4589642122738423) < 1e-12,
           "the un-mitered spans leave a corner gap, got {span_gap(p, cells, side, nx)}");
  }

  // Swept over plan sizes: the recovered corner never lands on the plan's, it is
  // sometimes outside the plan rectangle and sometimes inside it, and the distance
  // wanders with how the fill quantises.
  noutside = 0;
  ninside = 0;
  nearest = 9.0;
  for wid in 2..9 {
    q = plan_new(0, 0, wid, wid - 1, 0, false);
    c = patch();
    lb = labels_new(Q0, R0, W, H);
    draw_floor(q, c, lb, 1);
    (mx, my, mok) = surface_miter(q, c, 3, 0);
    assert(mok, "adjacent sides of a closed form always cross");
    (cx, cy) = plan_side_point(q, 0, 0.0);
    d = sqrt((mx - cx) * (mx - cx) + (my - cy) * (my - cy)) ?? 0.0;
    if d < nearest { nearest = d; }
    (_lu, lv) = plan_to_local(q, mx, my);
    // the recovered corner sits on the row lattice: an exact multiple of 3/4
    rows = (abs(lv) / 0.75) ?? 0.0;
    assert(abs(rows - ((round(rows) ?? 0.0))) < 1e-12,
           "the recovered corner is a rational of the lattice, got {abs(lv)}");
    // the plan's half-depth is not — it is (dep * sqrt(3) / 2)
    steps = (plan_hd(q) / 0.75) ?? 0.0;
    assert(abs(steps - ((round(steps) ?? 0.0))) > 1e-6, "and the plan's half-depth is not");
    if abs(lv) > plan_hd(q) { noutside = noutside + 1; } else { ninside = ninside + 1; }
  }
  assert(nearest > 0.4, "no plan size brings the two corners closer than {nearest} world units");
  assert(noutside > 0 && ninside > 0,
         "and the recovered corner is outside the plan on some sizes and inside on others");
}

// @HXD-004 — A DOOR IS A MATERIAL, AND NEVER A HOLE.  The field stores ONE
// material per wall slot, so an opening REPLACES the wall material rather than
// sitting in a channel beside it.  That leaves exactly one way to get it wrong,
// and it is the way that looks simplest: write 0.
//
// `OPEN_NONE` is not "no opening" — it is the wall material 0, which means NO
// WALL.  Writing it deletes the edge, and the wall stops being one closed run:
// the boundary that had no loose ends grows two.  Everything downstream that
// treats a wall as one run — recovery, collision, the surface above — is reading
// a different shape from then on.
//
// `OPEN_GAP` is what "a hole in the wall" actually means here: the leaf is missing
// entirely, nothing collides, and THE WALL IS STILL THERE.  An opening has
// geometry of its own — a frame, a lintel, a ragged jamb — and absence has none.
fn test_a_door_is_a_material_and_never_a_hole() {
  p = cottage();
  cells = cottage_cells();

  for kind in [OPEN_DOOR, OPEN_WINDOW, OPEN_GAP, OPEN_NONE] {
    edges = edgeset_new(Q0, R0, W, H);
    ne = draw_walls(cells, edges, WALLMAT);
    assert(ne == 38 && wall_chain_ends(edges, Q0, R0, W, H) == 0,
           "the drawn wall is a closed loop before anything is placed");
    placed = place_opening(p, cells, edges, 0, 0.5, 1, kind);
    assert(placed == 1, "one edge takes the opening");
    left = wall_edge_count(cells, edges);
    ends = wall_chain_ends(edges, Q0, R0, W, H);
    if kind == OPEN_NONE {
      assert(left == 37, "material 0 DELETES the edge: 38 -> {left}");
      assert(ends == 2, "and the closed loop grows two loose ends, got {ends}");
    } else {
      assert(left == 38, "an opening keeps its edge, got {left}");
      assert(ends == 0, "so the wall is still one closed run, got {ends}");
    }
  }
}

// @HXD-005 — A WALL IS THE BOUNDARY OF A FILLED REGION, WHICH IS WHY IT COSTS NO
// FLOOR.  Walls are edges between a cell of the region and a cell outside it, so
// the boundary is a closed loop BY CONSTRUCTION at any size, orientation or tie —
// and every floor cell survives.  Drawn instead as a band of cells along the
// outline, this cottage would lose the 16 of its 27 cells that touch the outside:
// 59% of the house, and the remaining room is not the room that was designed.
//
// Counting those edges has one trap, and it comes from the field's own symmetry:
// a boundary edge belongs to a PAIR of cells, and the enumeration a caller writes
// first — the three directions each cell "owns" — misses every edge whose owner is
// the outside cell.  Both neighbour conventions in this family have such an
// enumeration, and they are not the same three, so the same mistake reads 19 in one
// and 17 in the other while the answer is 38.
fn test_the_wall_costs_no_floor_and_the_count_needs_all_six() {
  cells = cottage_cells();
  assert(hexset_count(cells) == 27, "the cottage keeps all 27 floor cells");
  edges = edgeset_new(Q0, R0, W, H);
  ne = draw_walls(cells, edges, WALLMAT);
  assert(ne == 38, "with 38 wall edges around them");
  assert(ring_cells(cells) == 16, "16 of the 27 touch the outside — what a band wall would eat");

  assert(wall_edge_count(cells, edges) == 38, "all six directions from every cell: 38");
  assert(count_three_dirs(cells, edges, true) == 19, "the three a cell owns, hex_grid order: 19");
  assert(count_three_dirs(cells, edges, false) == 17, "the same shortcut, hex_field order: 17");
}

// @HXD-006 — A FEATURE IS INDEXED BY ITS EXACT `t`, NEVER BY ITS POSITION IN THE
// RUN.  The model puts a door at `(side, t)`; the field marks edges.  Along a run
// of `n` edges the recoverable positions are the edge centres `(2i+1)/2n` — exact
// rationals, odd numerator over `2n` — and `place_opening` snaps to the nearest,
// which is what makes a feature survive the round trip (`hex_fit`'s @HXI-005 is
// the doorstep that refuses the positions this cannot get back).
//
// The trap is the run's ORDER.  `side_edges` fills it in cell-scan order, which is
// not `t` order: the numerators of this side come out `1 5 3 7 9 13 11 15 17`.  So
// two edges that are ADJACENT along the wall can sit two apart in the run with a
// plain wall edge between them — which is exactly what a two-edge window does here.
// Read features with `feature_find`, which returns numerators in ascending `t`.
fn test_a_feature_is_indexed_by_its_t_not_its_position() {
  p = cottage();
  cells = cottage_cells();
  edges = edgeset_new(Q0, R0, W, H);
  draw_walls(cells, edges, WALLMAT);

  sr = side_edges(p, cells, 0);
  n = side_run_len(sr);
  assert(n == 9, "side 0 carries 9 edges, got {n}");
  order = "";
  for i in 0..n { order += "{feature_t_num(sr, i, n)} "; }
  assert(order == "1 5 3 7 9 13 11 15 17 ", "the STORAGE order of the t numerators is [{order}]");

  // A door in the middle, and a two-edge window near the far end.
  assert(place_opening(p, cells, edges, 0, 0.5, 1, OPEN_DOOR) == 1, "one door edge");
  assert(place_opening(p, cells, edges, 0, 0.1, 2, OPEN_WINDOW) == 2, "two window edges");

  door = feature_find(p, cells, edges, 0, OPEN_DOOR);
  window = feature_find(p, cells, edges, 0, OPEN_WINDOW);
  assert(len(door) == 1 && (door[0] ?? 0) == 9, "the door is at t = 9/18, exactly the middle");
  assert(len(window) == 2 && (window[0] ?? 0) == 1 && (window[1] ?? 0) == 3,
         "the window took the two nearest centres to 0.1, t = 1/18 and 3/18");
  assert(feature_t_admissible(9, 9) && !feature_t_admissible(8, 9),
         "a legal t has an ODD numerator over 2n — an even one is a corner, not a centre");

  // …and those two adjacent window edges are NOT adjacent in the run.
  assert(surface_edge_mat(p, cells, edges, 0, 0) == OPEN_WINDOW, "run position 0 is window");
  assert(surface_edge_mat(p, cells, edges, 0, 1) == WALLMAT, "position 1 is plain wall");
  assert(surface_edge_mat(p, cells, edges, 0, 2) == OPEN_WINDOW, "position 2 is the window again");

  // What a renderer draws is not the marked EDGE but its interval ON the surface:
  // the wall is one quad and each feature is a sub-interval of it, so the zigzag
  // never reaches the picture.  One edge is exactly 1.0 world unit of clear
  // width — 0.87 m, a real domestic door, which a cell-thick wall could not express.
  (ex0, ey0, ex1, ey1) = surface_edge_span(p, cells, 0, 0);
  wide = sqrt((ex1 - ex0) * (ex1 - ex0) + (ey1 - ey0) * (ey1 - ey0)) ?? 0.0;
  assert(abs(wide - 1.0) < 1e-12, "one edge opens 1.0 world unit of wall, got {wide}");
  assert(abs(wide * 0.8660254037844386 - 0.8660254037844386) < 1e-12, "which is 0.87 m");
  // and both its ends lie on the wall's own line, not on the zigzag
  (sx0, sy0, sx1, sy1) = surface_span(p, cells, 0);
  along = sqrt((sx1 - sx0) * (sx1 - sx0) + (sy1 - sy0) * (sy1 - sy0)) ?? 0.0;
  assert(abs(ex0 - sx0) < 1e-12 && abs(ex1 - sx1) < 1e-12,
         "the interval runs along the same line as the wall quad");
  assert(along > wide, "and is a sub-interval of it: {wide} of {along}");
}

// @HXD-007 — THE RIDGE HAS TO RUN PAST THE GABLE ENDS.  A roof height is the
// distance from a ridge LINE, so where a cell's nearest point on that line is an
// ENDPOINT rather than a perpendicular foot, the distance goes radial and the roof
// rolls over like a quarter cone.  Stopping the ridge at the wall — the obvious
// thing to do — does exactly that, and the gable end drops by sqrt(3)/4 world
// units, 0.375 m of bulge on a real building.
//
// It is also a lesson about the measurement.  Compare the two rules over the
// MASSING alone and they are identical to the last bit, because no cell of the
// massing is past the ridge's end.  The difference lives entirely in the halo ring
// `grow_ring` adds — which is there because the eave has to interpolate against
// cells outside the wall — so an instrument that skips the halo reports a perfect
// result over the empty set.
fn test_the_ridge_runs_past_the_gable_ends() {
  p = cottage();
  massing = cottage_cells();
  hw = plan_hw(p);
  hd = plan_hd(p);

  // On the massing alone the two rules agree — the comparison has nowhere to bite.
  past_m = ridge_heights(p, massing, hw + hd + HEX_LEN);
  stop_m = ridge_heights(p, massing, hw);
  worst_m = 0.0;
  for r in R0..R0 + H {
    for q in Q0..Q0 + W {
      if hexset_get(massing, q, r) {
        d = abs(height_get(past_m, q, r) - height_get(stop_m, q, r));
        if d > worst_m { worst_m = d; }
      }
    }
  }
  assert(worst_m < 1e-12, "over the massing the two ridges agree — an empty measurement");

  // With the halo the difference is the bulge.  `draw_roof` is the shipped rule —
  // it runs the ridge past both ends — and the hand-built ridge below is the
  // tempting one that stops at the wall.
  grown = patch();
  grow_ring(massing, grown);
  assert(hexset_count(grown) == 49, "the halo takes 27 cells to 49, got {hexset_count(grown)}");
  past = heights_new(Q0, R0, W, H);
  draw_roof(p, grown, past, 2.0, 0.5, 0);
  stop = ridge_heights(p, grown, hw);
  worst = 0.0;
  ndiff = 0;
  for r in R0..R0 + H {
    for q in Q0..Q0 + W {
      if hexset_get(grown, q, r) {
        d = abs(height_get(past, q, r) - height_get(stop, q, r));
        if d > 1e-9 { ndiff = ndiff + 1; }
        if d > worst { worst = d; }
      }
    }
  }
  assert(ndiff == 10, "10 of the 49 cells sit past the shortened ridge, got {ndiff}");
  assert(abs(worst - 0.4330127018922193) < 1e-12,
         "and the worst of them rolls over by sqrt(3)/4 units, got {worst}");
  assert(abs(worst * 0.8660254037844386 - 0.375) < 1e-12, "which is 0.375 m of bulge");

  // The rolling over is not a defect in `roof_ridge` — it is what a SHORT ridge
  // means, and on a hip roof it is the shape you are asking for.  Same call, same
  // mechanism, and the only difference is whether the ridge was meant to end there.
  hip = heights_new(Q0, R0, W, H);
  draw_roof(p, grown, hip, 2.0, 0.5, 2);
  nlower = 0;
  for r in R0..R0 + H {
    for q in Q0..Q0 + W {
      if hexset_get(grown, q, r) && height_get(hip, q, r) < height_get(past, q, r) - 1e-9 {
        nlower = nlower + 1;
      }
    }
  }
  assert(nlower > 0, "a hip's shortened ridge lowers the ends deliberately, {nlower} cells");
}
