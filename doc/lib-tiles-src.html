// Copyright (c) 2026 Jurjen Stellingwerff
// SPDX-License-Identifier: LGPL-3.0-or-later
//
// tiles — tile sheets, map layers and sprites on a small 16-bit surface, scaled up once per
// frame.  Built for a handheld (the Tanmatsu: a 32-bit core, 768 KB of fast SRAM, an 800x480
// panel whose framebuffer is in slower PSRAM), and the same on every target.
//
// The shape of a frame: compose it at 1x on a `Surface` (200x120 for a 4x scale on 800x480 is
// 48 KB), drawing map layers and sprites, then `present` it scaled — and on the device, rotated
// a quarter turn into the portrait panel's framebuffer.  Only that last step touches the large
// buffer, once per pixel.
//
// A sheet is converted once, when it is made: one byte per pixel (index 0 transparent, 1..255
// RGB565 colours), tile after tile.  A Kenney 16 px pack of 132 tiles is 34 KB that way, where
// RGBA would be 135 KB.  Colours are RGB565 — `rgb(r, g, b)` makes one — and a sheet with more
// than 255 of them, once in RGB565, is refused.
//
// Orientation is Tiled's: a tile is transposed (the diagonal flip) first, then mirrored
// horizontally, then vertically, so the eight combinations are the eight ways a square can lie.
//
// Handles: a sheet, a surface or a layer is a positive integer, null when it could not be made.
use imaging;

fn handle(h: integer) -> integer? {
  if h <= 0 { return null; }
  h
}

fn flags(flip_h: boolean, flip_v: boolean, flip_d: boolean) -> integer {
  (if flip_h { 1 } else { 0 }) + (if flip_v { 2 } else { 0 }) + (if flip_d { 4 } else { 0 })
}

/// An RGB565 colour from 8-bit channels: 5 bits of red, 6 of green, 5 of blue.
pub fn rgb(r: integer, g: integer, b: integer) -> integer {
  ((r & 255) / 8) * 2048 + ((g & 255) / 4) * 32 + (b & 255) / 8
}

// --- Sheets ---

/// A sheet from `width * height` RGBA bytes, cut into `tile_w * tile_h` tiles with `margin`
/// pixels round the sheet and `spacing` between tiles (Tiled's names).  A pixel with alpha
/// below 128 is transparent.  Tiles are numbered from 0, row by row.  Null when the sizes do not
/// fit or the sheet has more than 255 colours once in RGB565.
pub fn sheet(rgba: const vector<u8>, width: integer, height: integer, tile_w: integer,
             tile_h: integer, spacing: integer = 0, margin: integer = 0) -> integer? {
  if len(rgba) < width * height * 4 { return null; }
  handle(tl_set_new(rgba, width, height, tile_w, tile_h, spacing, margin))
}

/// A sheet from a decoded image (`imaging::png`).
pub fn sheet_from_image(img: const imaging::Image, tile_w: integer, tile_h: integer,
                        spacing: integer = 0, margin: integer = 0) -> integer? {
  bytes: vector<u8> = [];
  for p in img.data { bytes += [p.r as u8, p.g as u8, p.b as u8, p.a as u8]; }
  sheet(bytes, img.width, img.height, tile_w, tile_h, spacing, margin)
}

/// A sheet from a PNG file.
pub fn sheet_from_png(path: text, tile_w: integer, tile_h: integer, spacing: integer = 0,
                      margin: integer = 0) -> integer? {
  img = file(path).png();
  if img == null { return null; }
  sheet_from_image(img ?? imaging::Image { }, tile_w, tile_h, spacing, margin)
}

/// How many tiles a sheet holds, and how many colours (at most 255).
pub fn tile_count(sheet: integer) -> integer {

}

pub fn colour_count(sheet: integer) -> integer {
  tl_set_colours(sheet)
}

pub fn free_sheet(sheet: integer) {
  tl_set_free(sheet);
}

/// A 1x RGB565 surface to compose a frame on, cleared to black.
pub fn surface(width: integer, height: integer) -> integer? {
  handle(tl_surf_new(width, height))
}

pub fn free_surface(surface: integer) {
  tl_surf_free(surface);
}

/// Fill a surface with one RGB565 colour.
pub fn clear(surface: integer, colour: integer) {
  tl_clear(surface, colour);
}

/// Fill the `w * h` rectangle with its top-left at (x, y) with one RGB565 colour.  What falls
/// off the surface is skipped, and a rectangle with no width or height draws nothing.
pub fn rect(surface: integer, x: integer, y: integer, w: integer, h: integer, colour: integer) {
  tl_rect(surface, x, y, w, h, colour);
}

/// The one-pixel outline of that rectangle, inside it: a cursor, a selection, the frame round
/// a palette.  Clipped like `rect`.
pub fn frame(surface: integer, x: integer, y: integer, w: integer, h: integer, colour: integer) {
  if w <= 0 || h <= 0 { return; }
  rect(surface, x, y, w, 1, colour);
  rect(surface, x, y + h - 1, w, 1, colour);
  rect(surface, x, y + 1, 1, h - 2, colour);
  rect(surface, x + w - 1, y + 1, 1, h - 2, colour);
}

/// How many shades a tile can be drawn at: 0 is the tile as it is, `SHADES - 1` the darkest.
pub const SHADES = 8;

/// Draw tile `tile` of `sheet` with its top-left at (x, y), in any of the eight orientations.
/// A sprite is a tile drawn this way.  Transparent pixels and whatever falls off the surface
/// are skipped.  `shade` darkens it: 0 is the tile as it is, and each shade up to
/// `SHADES - 1` is darker, down to about a quarter of the light, so the darkest still shows
/// what is there.  A shade costs nothing: each sheet keeps its palette at every shade.
pub fn draw(surface: integer, sheet: integer, tile: integer, x: integer, y: integer,
            flip_h: boolean = false, flip_v: boolean = false, flip_d: boolean = false,
            shade: integer = 0) {
  s = if shade < 0 { 0 } else if shade >= SHADES { SHADES - 1 } else { shade };
  tl_draw(surface, sheet, tile, x, y, flags(flip_h, flip_v, flip_d) + s * 256);
}

/// Draw tile `tile` of `sheet` turned `degrees` clockwise about its pixel `pivot` (x, y within
/// the tile): placed as `draw` places it at (x, y), oriented by the flips first, then turned —
/// a character's slight lean as it walks, a weapon held at an angle or swung.  Every pixel of
/// the tile lands on exactly one pixel of the surface (the turn is whole quarter turns and three
/// shears of whole rows and columns), so the pixel art keeps its pixels and colours at any angle;
/// 0 degrees is `draw`, and 90, 180 and 270 are its quarter turns.  The shade, transparency
/// and clipping are `draw`'s.
pub fn draw_turned(surface: integer, sheet: integer, tile: integer, x: integer, y: integer,
                   degrees: integer, pivot: (integer, integer), flip_h: boolean = false,
                   flip_v: boolean = false, flip_d: boolean = false, shade: integer = 0) {
  s = if shade < 0 { 0 } else if shade >= SHADES { SHADES - 1 } else { shade };
  tl_draw_turned(surface, sheet, tile, x, y, degrees, pivot.0, pivot.1, flags(flip_h, flip_v, flip_d) + s * 256);
}

/// How finely `draw_ground` shades: its darkness is counted in sixteenths of a shade.
pub const SHADE_PARTS = 16;

/// Draw tile `tile` of `sheet` as a piece of ground that slopes and is lit gradually: its place
/// is the plain `draw`'s at (x, y), with its corners — top-left, top-right, bottom-left,
/// bottom-right — lifted `lift` pixels (up is positive, so (4, 4, 0, 0) leans its far edge up
/// a step) and darkened `shade` sixteenths of a shade (`SHADE_PARTS` is one whole shade darker;
/// past `SHADES - 1` whole shades it stays the darkest).
///
/// Each column of the tile stays a column of pixels, stretched or squeezed between its top and
/// bottom edges, whose lifts are interpolated from the corners at the column's centre.  So two
/// tiles that give their shared corners the same lifts meet exactly, with no gap — a slope, a
/// mound, a ramp onto a higher floor; and a drop between them shows as the gap a wall tile (or a
/// darker one) fills, drawn the same way.  The darkness is interpolated across the tile and
/// shared between the two nearest shades by a fixed 4x4 dither, so light fades smoothly within
/// a tile and the pattern stays with the ground as the view moves.  Over a plain `draw` it costs
/// a little per column and a table lookup per pixel; a tile with every lift and shade 0 draws
/// exactly what `draw` draws.  Orientation, transparency and clipping are `draw`'s.
pub fn draw_ground(surface: integer, sheet: integer, tile: integer, x: integer, y: integer,
                   lift: (integer, integer, integer, integer),
                   shade: (integer, integer, integer, integer) = (0, 0, 0, 0),
                   flip_h: boolean = false, flip_v: boolean = false, flip_d: boolean = false) {
  tl_draw_ground(surface, sheet, tile, x, y, lift.0, lift.1, lift.2, lift.3,
                 shade.0, shade.1, shade.2, shade.3, flags(flip_h, flip_v, flip_d));
}

/// The surface's pixels, row by row, each an RGB565 value.
pub fn pixels(surface: integer, width: integer, height: integer) -> vector<i32> {
  out: vector<i32> = [0 as i32; width * height];
  tl_pixels(surface, out);
  out
}

/// The surface scaled by the integer `scale`, as `present` hands it to a display: row by row,
/// (width * scale) x (height * scale) RGB565 values — or, `rotate`d a quarter turn clockwise
/// for a portrait panel, (height * scale) x (width * scale), the surface's top-left at the
/// top-right.
pub fn present(surface: integer, width: integer, height: integer, scale: integer,
               rotate: boolean = false) -> vector<i32> {
  out: vector<i32> = [0 as i32; width * height * scale * scale];
  tl_present(surface, out, scale, if rotate { 1 } else { 0 });
  out
}

// --- Map layers ---

/// A layer cell: tile `tile` (from 0) in one of the eight orientations.  0 is an empty cell.
pub fn cell(tile: integer, flip_h: boolean = false, flip_v: boolean = false,
            flip_d: boolean = false) -> integer {
  (tile + 1) + flags(flip_h, flip_v, flip_d) * 16777216
}

/// A Tiled global tile id (with its flip bits: 31 horizontal, 30 vertical, 29 diagonal) as a
/// cell of a sheet whose first id is `firstgid`; 0 for an empty one or a tile of another sheet
/// (an id below `firstgid`).
pub fn from_tiled(gid: integer, firstgid: integer = 1) -> integer {
  id = gid % 536870912;
  if id < firstgid || id == 0 { return 0; }
  bits = gid / 536870912;
  cell(id - firstgid, bits / 4 % 2 == 1, bits / 2 % 2 == 1, bits % 2 == 1)
}

/// A map layer of `cols * rows` cells, row by row.  Null when the cell count does not match.
pub fn layer(cells: const vector<integer>, cols: integer, rows: integer) -> integer? {
  if len(cells) != cols * rows { return null; }
  packed: vector<i32> = [];
  for c in cells { packed += [c & 2147483647]; }
  handle(tl_map_new(packed, cols, rows))
}

/// A layer from a Tiled layer's CSV data (`<data encoding="csv">`), its gids read by `from_tiled`.
pub fn layer_from_tiled_csv(csv: text, cols: integer, rows: integer, firstgid: integer = 1) -> integer? {
  cells: vector<integer> = [];
  for part in csv.split(',') {
    t = part.trim();
    if len(t) > 0 { cells += [from_tiled(t as integer? ?? 0, firstgid)]; }
  }
  layer(cells, cols, rows)
}

pub fn free_layer(layer: integer) {
  tl_map_free(layer);
}

/// Set or read one cell; a position off the map is ignored, and reads 0.
pub fn set_cell(layer: integer, x: integer, y: integer, cell: integer) {
  tl_map_set(layer, x, y, cell & 2147483647);
}

pub fn get_cell(layer: integer, x: integer, y: integer) -> integer {
  tl_map_get(layer, x, y)
}

/// Draw every cell of `layer` that shows on `surface` when the layer's pixel (cam_x, cam_y) is
/// the surface's top-left — one call per layer, scrolling to the pixel.  Draw layers back to
/// front.
pub fn draw_layer(surface: integer, layer: integer, sheet: integer, cam_x: integer = 0,
                  cam_y: integer = 0) {
  tl_draw_map(surface, layer, sheet, cam_x, cam_y);
}

// Copyright (c) 2026 Jurjen Stellingwerff
// SPDX-License-Identifier: LGPL-3.0-or-later
//
// The tile layer, read back pixel by pixel; every expected value is computed by hand in its
// cell.  The fixture sheet is 4x2 RGBA pixels, two 2x2 tiles side by side:
//   tile 0:  R G      tile 1:  W W
//            B .               W W      (. transparent)
// In RGB565, R = 63488 (31 << 11), G = 2016 (63 << 5), B = 31, W = 65535; a fresh or cleared
// surface is 0.  A tile drawn at (0, 0) on a 2x2 surface fills it, so a 2x2 cell reads as
// `[top-left, top-right, bottom-left, bottom-right]`.
use tiles;

const R = 63488;
const G = 2016;
const B = 31;
const W = 65535;

fn px(r: integer, g: integer, b: integer, a: integer) -> vector<u8> {
  [(r & 255) as u8, (g & 255) as u8, (b & 255) as u8, (a & 255) as u8]
}

// Row 0: R G W W; row 1: B . W W.
fn fixture() -> integer {
  bytes: vector<u8> = [];
  bytes += px(255, 0, 0, 255);
  bytes += px(0, 255, 0, 255);
  bytes += px(255, 255, 255, 255);
  bytes += px(255, 255, 255, 255);
  bytes += px(0, 0, 255, 255);
  bytes += px(0, 0, 0, 0);
  bytes += px(255, 255, 255, 255);
  bytes += px(255, 255, 255, 255);
  tiles::sheet(bytes, 4, 2, 2, 2) ?? 0
}

fn show(v: const vector<i32>) -> text {
  t = "";
  for p in v { t += "{p},"; }
  t
}

// Tile 0 drawn on a 2x2 surface with the given flips.
fn oriented(flip_h: boolean, flip_v: boolean, flip_d: boolean) -> text {
  s = tiles::surface(2, 2) ?? 0;
  tiles::draw(s, fixture(), 0, 0, 0, flip_h, flip_v, flip_d);
  show(tiles::pixels(s, 2, 2))
}

fn main() {
  assert(tiles::rgb(255, 0, 0) == R && tiles::rgb(0, 255, 0) == G, "RGB565 red and green");
  assert(tiles::rgb(0, 0, 255) == B && tiles::rgb(255, 255, 255) == W, "RGB565 blue and white");
}

fn test_sheet_counts() {
  s = fixture();
  assert(s > 0, "the sheet is made");
  assert(tiles::tile_count(s) == 2, "two tiles, got {tiles::tile_count(s)}");
  assert(tiles::colour_count(s) == 4, "R, G, B, W, got {tiles::colour_count(s)}");
}

// At (1, 1) on a 4x3 surface; the transparent pixel leaves the background.
fn test_draw_places_and_keeps_transparency() {
  s = tiles::surface(4, 3) ?? 0;
  tiles::draw(s, fixture(), 0, 1, 1);
  got = show(tiles::pixels(s, 4, 3));
  assert(got == "0,0,0,0,0,{R},{G},0,0,{B},0,0,", "got {got}");
}

fn test_the_eight_orientations() {
  assert(oriented(false, false, false) == "{R},{G},{B},0,", "as is: {oriented(false, false, false)}");
  assert(oriented(true, false, false) == "{G},{R},0,{B},", "mirrored: {oriented(true, false, false)}");
  assert(oriented(false, true, false) == "{B},0,{R},{G},", "flipped: {oriented(false, true, false)}");
  assert(oriented(false, false, true) == "{R},{B},{G},0,", "transposed: {oriented(false, false, true)}");
  assert(oriented(true, false, true) == "{B},{R},0,{G},", "a quarter turn clockwise: {oriented(true, false, true)}");
  assert(oriented(true, true, false) == "0,{B},{G},{R},", "a half turn: {oriented(true, true, false)}");
  assert(oriented(false, true, true) == "{G},0,{R},{B},", "a quarter turn back: {oriented(false, true, true)}");
  assert(oriented(true, true, true) == "0,{G},{B},{R},", "the anti-transpose: {oriented(true, true, true)}");
}

// Tile 1 (white) half off the surface at either corner: only the pixel inside is drawn.
fn test_clipping() {
  s = tiles::surface(2, 2) ?? 0;
  tiles::draw(s, fixture(), 1, -1, -1);
  assert(show(tiles::pixels(s, 2, 2)) == "{W},0,0,0,", "top-left: {show(tiles::pixels(s, 2, 2))}");
  t = tiles::surface(2, 2) ?? 0;
  tiles::draw(t, fixture(), 1, 1, 1);
  assert(show(tiles::pixels(t, 2, 2)) == "0,0,0,{W},", "bottom-right: {show(tiles::pixels(t, 2, 2))}");
  tiles::draw(t, fixture(), 7, 0, 0);
  assert(show(tiles::pixels(t, 2, 2)) == "0,0,0,{W},", "a tile the sheet has not draws nothing");
  // Over the left and the right edge, whole rows: a column that is not clipped would land in
  // the neighbouring row (x -1 of row 1 is x 1 of row 0, x 2 of row 0 is x 0 of row 1).
  l = tiles::surface(2, 2) ?? 0;
  tiles::draw(l, fixture(), 1, -1, 0);
  assert(show(tiles::pixels(l, 2, 2)) == "{W},0,{W},0,", "left edge: {show(tiles::pixels(l, 2, 2))}");
  r = tiles::surface(2, 2) ?? 0;
  tiles::draw(r, fixture(), 1, 1, 0);
  assert(show(tiles::pixels(r, 2, 2)) == "0,{W},0,{W},", "right edge: {show(tiles::pixels(r, 2, 2))}");
}

// A 7x4 sheet, margin 1, spacing 1: two 2x2 tiles at x 1..2 and 4..5, y 1..2, the rest magenta.
// Tile 0 is red, tile 1 blue; neither may show magenta.
fn test_margin_and_spacing() {
  bytes: vector<u8> = [];
  for y in 0..4 {
    for x in 0..7 {
      if y >= 1 && y <= 2 && x >= 1 && x <= 2 {
        bytes += px(255, 0, 0, 255);
      } else if y >= 1 && y <= 2 && x >= 4 && x <= 5 {
        bytes += px(0, 0, 255, 255);
      } else {
        bytes += px(255, 0, 255, 255);
      }
    }
  }
  sh = tiles::sheet(bytes, 7, 4, 2, 2, 1, 1) ?? 0;
  assert(tiles::tile_count(sh) == 2, "two tiles, got {tiles::tile_count(sh)}");
  s = tiles::surface(4, 2) ?? 0;
  tiles::draw(s, sh, 0, 0, 0);
  tiles::draw(s, sh, 1, 2, 0);
  got = show(tiles::pixels(s, 4, 2));
  assert(got == "{R},{R},{B},{B},{R},{R},{B},{B},", "got {got}");
}

// A 2x1 layer [tile 0, tile 1] on a 4x2 surface scrolled one pixel right: tile 0's right column
// at x 0 (G over transparent), tile 1 at x 1..2, and nothing past the map's end at x 3.
fn test_layer_scrolls_to_the_pixel() {
  m = tiles::layer([tiles::cell(0), tiles::cell(1)], 2, 1) ?? 0;
  s = tiles::surface(4, 2) ?? 0;
  tiles::draw_layer(s, m, fixture(), 1, 0);
  got = show(tiles::pixels(s, 4, 2));

// Tiled ids: bit 31 horizontal, 30 vertical, 29 diagonal; a cell is (tile + 1) | flags << 24.
fn test_tiled_ids() {
  assert(tiles::from_tiled(2) == 2, "gid 2 is tile 1");
  assert(tiles::from_tiled(2147483650) == 16777218, "0x80000002: tile 1 mirrored");
  assert(tiles::from_tiled(2684354561) == 83886081, "0xA0000001: tile 0 mirrored and transposed");
  assert(tiles::from_tiled(0) == 0, "an empty cell");
  assert(tiles::from_tiled(5, 10) == 0, "a tile of an earlier sheet");

  assert(tiles::get_cell(m, 0, 0) == 1 && tiles::get_cell(m, 1, 0) == 2, "the first row");
  assert(tiles::get_cell(m, 0, 1) == 0 && tiles::get_cell(m, 1, 1) == 16777218, "the second row");
  tiles::set_cell(m, 0, 1, tiles::cell(1, true));
  assert(tiles::get_cell(m, 0, 1) == 16777218, "a cell set");
  assert(tiles::get_cell(m, 9, 9) == 0, "off the map reads 0");
}

// --- Ground: draw_ground ---
//
// A second fixture: one 2x4 tile, a colour per row — R, G, B, W from the top — so a column read
// top to bottom shows which tile row each pixel took.  `column(s, x, h)` reads column x of a
// surface h high, one letter a pixel (. for nothing drawn).

fn rows_fixture() -> integer {
  bytes: vector<u8> = [];
  for c in [(255, 0, 0), (0, 255, 0), (0, 0, 255), (255, 255, 255)] {
    bytes += px(c.0, c.1, c.2, 255);
    bytes += px(c.0, c.1, c.2, 255);
  }
  tiles::sheet(bytes, 2, 4, 2, 4) ?? 0
}

// Shade 1 (x 208 / 256 per channel, rounded down) as lower case: r = 25 << 11, g = 51 << 5,
// b = 25, w = all three; shade 2 (x 168 / 256) as digits: 1 = 20 << 11, 2 = 41 << 5, 3 = 20,
// 4 = all three.
fn letter(p: integer) -> text {
  if p == R { "R" } else if p == G { "G" } else if p == B { "B" } else if p == W { "W" }
  else if p == 51200 { "r" } else if p == 1632 { "g" } else if p == 25 { "b" } else if p == 52857 { "w" }
  else if p == 40960 { "1" } else if p == 1312 { "2" } else if p == 20 { "3" } else if p == 42292 { "4" }
  else if p == 0 { "." } else { "?" }
}

fn column(s: integer, x: integer, w: integer, h: integer) -> text {
  v = tiles::pixels(s, w, h);
  t = "";
  for y in 0..h { t += letter(v[y * w + x] ?? -1); }
  t
}

fn test_flat_ground_is_draw() {

  }
}

// --- Turned: draw_turned ---

// Opaque pixels of a w x h surface, and how many of each fixture colour.
fn census(s: integer, w: integer, h: integer) -> text {
  n = 0;
  r = 0;
  g = 0;
  b = 0;
  wh = 0;
  for p in tiles::pixels(s, w, h) {
    if p != 0 { n += 1; }
    if p == R { r += 1; }
    if p == G { g += 1; }
    if p == B { b += 1; }
    if p == W { wh += 1; }
  }
  "{n}: {r} {g} {b} {wh}"
}

// No turn is draw, in every orientation and about any pivot.
fn test_no_turn_is_draw() {
  for o in 0..8 {
    a = tiles::surface(4, 3) ?? 0;
    tiles::draw(a, fixture(), 0, 1, 1, (o & 1) != 0, (o & 2) != 0, (o & 4) != 0);
    b = tiles::surface(4, 3) ?? 0;
    tiles::draw_turned(b, fixture(), 0, 1, 1, 0, (1, 0), (o & 1) != 0, (o & 2) != 0, (o & 4) != 0);
    assert(show(tiles::pixels(a, 4, 3)) == show(tiles::pixels(b, 4, 3)), "orientation {o}: {show(tiles::pixels(b, 4, 3))}");
  }
}

// Quarter turns about the top-left pixel are draw's quarter turns, moved so that pixel stays:
// 90 (a quarter turn clockwise) lands at x - 1, 180 (a half turn) at (x - 1, y - 1), 270 (a
// quarter turn back) at y - 1; and -90 is 270, 450 is 90.
fn test_quarter_turns_are_draw() {
  for c in [(90, -1, 0, true, false, true), (180, -1, -1, true, true, false), (270, 0, -1, false, true, true),
            (-90, 0, -1, false, true, true), (450, -1, 0, true, false, true)] {
    a = tiles::surface(4, 4) ?? 0;
    tiles::draw(a, fixture(), 0, 1 + c.1, 1 + c.2, c.3, c.4, c.5);
    b = tiles::surface(4, 4) ?? 0;
    tiles::draw_turned(b, fixture(), 0, 1, 1, c.0, (0, 0));
    assert(show(tiles::pixels(a, 4, 4)) == show(tiles::pixels(b, 4, 4)), "{c.0}: {show(tiles::pixels(b, 4, 4))} against {show(tiles::pixels(a, 4, 4))}");
  }
}

// 45 degrees about the bottom-left pixel of the R G B W column tile, drawn at (2, 1) so the

// (0, -2) to (1, -1), (1, -2) to (2, -1); B (0, -1) stays, (1, -1) to (1, 0); W (0, 0) stays,

fn test_forty_five_degrees() {
  s = tiles::surface(6, 6) ?? 0;

  t = "";

}

fn column_row(s: integer, y: integer, w: integer) -> text {
  v = tiles::pixels(s, w, 6);
  t = "";
  for x in 0..w { t += letter(v[y * w + x] ?? -1); }
  t
}

// Every turn keeps every pixel: at each whole degree of a full turn, the column tile turned about
