// --- Ed25519 signatures (RFC 8032) ---
//
// All key / message / signature bytes are standard base64 `text`.  A secret
// key is the RFC 8032 32-byte seed (base64); the public key (32B) and
// signature (64B) are the canonical encodings.  Malformed input never halts:
// `ed25519_public_key` / `ed25519_sign` return "" and `ed25519_verify`
// returns false.

// Derive the 32-byte public key (base64) from a 32-byte secret seed (base64).
pub fn ed25519_public_key(secret_key_b64: text) -> text {
  ed25519_public_key_native(secret_key_b64)
}

// Sign the base64 `message` bytes with the 32-byte secret seed (base64).
// Returns the 64-byte signature, base64.
// Example: @CRY-002 — sign `base64_encode(msg)`; the bare text signs other bytes, silently.
pub fn ed25519_sign(secret_key_b64: text, message_b64: text) -> text {
  ed25519_sign_native(secret_key_b64, message_b64)
}

// Verify a signature over the base64 `message` bytes under a public key.
// Returns true iff the signature is valid.
// Example: @CRY-002 — why a wrongly-spelled message still verifies against itself.
pub fn ed25519_verify(public_key_b64: text, message_b64: text, signature_b64: text) -> boolean {
  ed25519_verify_native(public_key_b64, message_b64, signature_b64)
}

// --- ES256 signatures (ECDSA P-256 / SHA-256, JOSE `alg: ES256`) ---
//
// The signature RFC 8555 ACME requires for account keys and CSRs — Let's Encrypt
// rejects Ed25519 account keys, so ES256 cannot be replaced by the ed25519 path.
// All bytes are standard base64 `text`.  A secret key is the 32-byte P-256 scalar
// `d`; a public key is the 64-byte affine point `x || y` (the JWK coordinates,
// no 0x04 SEC1 prefix); a signature is the 64-byte raw `r || s` (the JOSE form —
// the DER encoding is for the X.509/PKCS#10 layer, never for a JWS signature).
// Signing is deterministic (RFC 6979), so the same key + message always signs to
// the same bytes; only `keygen` draws entropy.  Malformed input never halts:
// `ecdsa_p256_keygen` / `_public_key` / `_sign` return "" and `_verify` false.

// Generate a fresh 32-byte P-256 secret scalar (base64) from the OS CSPRNG.
pub fn ecdsa_p256_keygen() -> text {
  ecdsa_p256_keygen_native()
}

// Derive the 64-byte public key `x || y` (base64) from a 32-byte secret scalar (base64).
// Example: @CRY-003 — the three JOSE lengths (32 / 64 / 64) a caller should check.
pub fn ecdsa_p256_public_key(secret_key_b64: text) -> text {
  ecdsa_p256_public_key_native(secret_key_b64)
}

// Sign the base64 `message` bytes with the 32-byte secret scalar (base64).
// Returns the 64-byte `r || s` signature (base64), deterministic per RFC 6979.
// Example: @CRY-003 — signing twice answers identical bytes, and `r || s` is not DER.
pub fn ecdsa_p256_sign(secret_key_b64: text, message_b64: text) -> text {
  ecdsa_p256_sign_native(secret_key_b64, message_b64)
}

// Verify an ES256 signature (64-byte `r || s`) over the base64 `message` bytes
// under a 64-byte `x || y` public key.  Returns true iff the signature is valid.
pub fn ecdsa_p256_verify(public_key_b64: text, message_b64: text, signature_b64: text) -> boolean {
  ecdsa_p256_verify_native(public_key_b64, message_b64, signature_b64)
}

// --- Raw bytes <-> base64 ---
//
// loft `text` is UTF-8-validated, so a non-UTF-8 byte string cannot ride as
// `text` (`text_from_bytes` returns "").  These two inverse helpers move raw
// bytes across the base64 boundary instead.
//
// CAUTION — a `vector<u8>` returned by `base64_to_bytes` is sound to READ
// (index, iterate, pass by value, or concatenate INTO a fresh loft vector with
// `fresh += decoded`), but do NOT use it as the left-hand accumulator of `+=`
// (`decoded += more`): in-place growth of a store-allocated byte vector after a
// later store allocation currently corrupts it (a loft FFI-vector bug).  To
// concatenate base64 byte strings, prefer composing in base64 throughout (see
// how the HPKE code below uses `bytes_concat_b64_native`).

// Standard base64 of a raw byte vector.  Empty vector -> "".

  bytes_to_base64_native(bytes)
}

// ====================================================================
// HPKE — Hybrid Public Key Encryption, base mode (RFC 9180)
// ====================================================================
//
// Single-shot Seal/Open for the ciphersuite
//   DHKEM(X25519, HKDF-SHA256) / HKDF-SHA256 / AES-256-GCM
//   kem_id = 0x0020, kdf_id = 0x0001, aead_id = 0x0002, mode = 0x00 (base).
//
// This is a PURE-LOFT composition over the native primitives above (X25519,
// HKDF Extract/Expand, AES-256-GCM), so it runs identically on every backend
// the primitives reach.  Correctness is gated on an RFC 9180 known-answer test
// vector (see tests/hpke.loft).
//
// Byte composition stays entirely in base64 `text`: every byte chunk (labels,
// suite ids, keys, DH outputs) is a base64 string, and `bytes_concat_b64`
// concatenates the underlying raw bytes.  ASCII labels are base64-encoded with
// the existing `base64_encode` (ASCII is valid UTF-8).  This avoids composing
// in a store-allocated `vector<u8>`, whose in-place `+=` growth currently
// corrupts after an intervening store allocation (a loft FFI-vector bug — see
// the public `bytes_to_base64` / `base64_to_bytes` note below; they are sound
// for read-only use, just not as a `+=` accumulator).

// base64(bytes(a) || bytes(b)) — concatenate two base64 byte strings.
fn concat(a_b64: text, b_b64: text) -> text {
  bytes_concat_b64_native(a_b64, b_b64)
}

// An ASCII label as a base64 byte string (ASCII is valid UTF-8, so this is the
// plain text base64 encoder).
fn label_b64(s: text) -> text {
  base64_encode_native(s)
}

// I2OSP(n, 2) as base64: the 2-byte big-endian encoding of `n` (RFC 8017).
// Only the small fixed set HPKE needs (suite-id fields and the L lengths 32/12)
// is required here, so each is a precomputed base64 constant — no `vector<u8>`.
fn i2osp_2_b64(n: integer) -> text {
  if n == 0x0020 { return "ACA="; }   // 0x00 0x20  (kem_id, and L = 32)
  if n == 0x0001 { return "AAE="; }   // 0x00 0x01  (kdf_id)
  if n == 0x0002 { return "AAI="; }   // 0x00 0x02  (aead_id)
  if n == 12     { return "AAw="; }   // 0x00 0x0c  (L = 12, base nonce)
  ""
}

// The main HPKE suite_id (base64): "HPKE" || I2OSP(kem,2) || I2OSP(kdf,2) || I2OSP(aead,2).
fn hpke_suite_id_b64() -> text {
  s = concat(label_b64("HPKE"), i2osp_2_b64(0x0020));  // + kem_id
  s = concat(s, i2osp_2_b64(0x0001));                  // + kdf_id
  concat(s, i2osp_2_b64(0x0002))                       // + aead_id
}

// The DHKEM suite_id (base64): "KEM" || I2OSP(kem_id,2).
fn kem_suite_id_b64() -> text {
  concat(label_b64("KEM"), i2osp_2_b64(0x0020))
}

// LabeledExtract(salt, label, ikm) [RFC 9180 §4]:
//   labeled_ikm = "HPKE-v1" || suite_id || label || ikm
//   return HKDF-Extract(salt, labeled_ikm)
// All byte args are base64.  Returns the 32-byte PRK (base64).
fn labeled_extract(suite_id_b64: text, salt_b64: text, label: text, ikm_b64: text) -> text {
  li = concat(label_b64("HPKE-v1"), suite_id_b64);
  li = concat(li, label_b64(label));
  li = concat(li, ikm_b64);
  hkdf_extract(salt_b64, li)
}

// LabeledExpand(prk, label, info, L) [RFC 9180 §4]:
//   labeled_info = I2OSP(L,2) || "HPKE-v1" || suite_id || label || info
//   return HKDF-Expand(prk, labeled_info, L)
// Returns L bytes (base64).
fn labeled_expand(suite_id_b64: text, prk_b64: text, label: text, info_b64: text, length: integer) -> text {
  li = concat(i2osp_2_b64(length), label_b64("HPKE-v1"));
  li = concat(li, suite_id_b64);
  li = concat(li, label_b64(label));
  li = concat(li, info_b64);
  hkdf_expand(prk_b64, li, length)
}

// DHKEM ExtractAndExpand(dh, kem_context) [RFC 9180 §4.1], KEM suite_id:
//   eae_prk       = LabeledExtract("", "eae_prk", dh)
//   shared_secret = LabeledExpand(eae_prk, "shared_secret", kem_context, Nsecret=32)
// Returns the 32-byte shared secret (base64).
fn dhkem_extract_and_expand(dh_b64: text, kem_context_b64: text) -> text {
  ks = kem_suite_id_b64();
  eae_prk = labeled_extract(ks, "", "eae_prk", dh_b64);
  labeled_expand(ks, eae_prk, "shared_secret", kem_context_b64, 32)
}

// The X25519 base point (base64): 0x09 followed by 31 zero bytes (RFC 7748).
// `x25519_dh(sk, base_point)` yields sk's public key.
const X25519_BASE_POINT = "CQAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA=";

// The X25519 public key for secret scalar `sk_b64`: X25519(sk, base_point).
fn x25519_public_key(sk_b64: text) -> text {
  x25519_dh(sk_b64, X25519_BASE_POINT)
}

// DHKEM Decap(enc, skR) [RFC 9180 §4.1]:
//   dh          = DH(skR, enc);  pkRm = pk(skR)
//   kem_context = enc || pkRm
//   return ExtractAndExpand(dh, kem_context)
// Returns the 32-byte shared secret (base64), or "" if enc/skR are invalid.
fn dhkem_decap(enc_b64: text, sk_r_b64: text) -> text {
  dh_b64 = x25519_dh(sk_r_b64, enc_b64);
  if dh_b64 == "" { return ""; }
  pk_r_b64 = x25519_public_key(sk_r_b64);
  kem_context = concat(enc_b64, pk_r_b64);
  dhkem_extract_and_expand(dh_b64, kem_context)
}

// DHKEM Encap(pkR, skE) — the deterministic variant [RFC 9180 §4.1].  Takes a
// fixed ephemeral secret `sk_e_b64` (the KAT path); `hpke_seal_base` generates
// a random one.
//   pkEm = pk(skE);  dh = DH(skE, pkR);  enc = pkEm
//   kem_context = enc || pkR
//   return (ExtractAndExpand(dh, kem_context), enc)
// Returns (shared_secret_b64, enc_b64); ("","") on invalid input.
fn dhkem_encap_deterministic(pk_r_b64: text, sk_e_b64: text) -> (text, text) {
  enc_b64 = x25519_public_key(sk_e_b64);
  if enc_b64 == "" { return ("", ""); }
  dh_b64 = x25519_dh(sk_e_b64, pk_r_b64);
  if dh_b64 == "" { return ("", ""); }
  kem_context = concat(enc_b64, pk_r_b64);
  shared_secret = dhkem_extract_and_expand(dh_b64, kem_context);
  (shared_secret, enc_b64)
}

// KeySchedule for base mode (RFC 9180 §5.1), MAIN suite_id.  Returns the AEAD
// key (32 bytes) and base nonce (12 bytes) as (key_b64, nonce_b64).
//   psk_id_hash          = LabeledExtract("", "psk_id_hash", "")
//   info_hash            = LabeledExtract("", "info_hash", info)
//   key_schedule_context = 0x00 || psk_id_hash || info_hash
//   secret               = LabeledExtract(shared_secret, "secret", "")
//   key        = LabeledExpand(secret, "key", key_schedule_context, Nk=32)
//   base_nonce = LabeledExpand(secret, "base_nonce", key_schedule_context, Nn=12)
fn key_schedule_base(shared_secret_b64: text, info_b64: text) -> (text, text) {
  s = hpke_suite_id_b64();

  psk_id_hash = labeled_extract(s, "", "psk_id_hash", "");
  info_hash = labeled_extract(s, "", "info_hash", info_b64);

  // key_schedule_context = mode (0x00) || psk_id_hash || info_hash.
  // The single 0x00 mode byte as base64 is "AA==".
  ks_context = concat("AA==", psk_id_hash);
  ks_context = concat(ks_context, info_hash);

  // secret = LabeledExtract(shared_secret, "secret", "")  — shared_secret is the salt.
  secret = labeled_extract(s, shared_secret_b64, "secret", "");

  key = labeled_expand(s, secret, "key", ks_context, 32);
  base_nonce = labeled_expand(s, secret, "base_nonce", ks_context, 12);
  (key, base_nonce)
}

// The result of `hpke_seal_base`: the KEM-encapsulated key `enc` and the AEAD
// `ciphertext` (ciphertext||16-byte-tag), both base64.
pub struct HpkeSealed {
  const enc: text,
  const ciphertext: text,
}

// Internal: seal with an already-encapsulated (shared_secret, enc).  seq = 0,
// so the AEAD nonce is exactly base_nonce (RFC 9180 §5.2).
fn seal_with_context(shared_secret_b64: text, enc_b64: text, info_b64: text, aad_b64: text, pt_b64: text) -> HpkeSealed {
  (key, base_nonce) = key_schedule_base(shared_secret_b64, info_b64);
  if key == "" || base_nonce == "" { return HpkeSealed { enc: "", ciphertext: "" }; }
  ct = aes256gcm_seal(key, base_nonce, aad_b64, pt_b64);
  HpkeSealed { enc: enc_b64, ciphertext: ct }
}

// --- Public HPKE API ---

// HPKE SealBase (RFC 9180 §6.1): encrypt `pt` to recipient public key `pk_r`,
// binding `info` (key-schedule context) and `aad` (per-message AAD).  Generates
// a fresh ephemeral key, so each call's output differs.  All args/fields base64.
// Returns HpkeSealed { enc, ciphertext }; both "" on failure.
// Example: @CRY-006 — `enc` is fresh per call and must travel with the ciphertext.
pub fn hpke_seal_base(pk_r_b64: text, info_b64: text, aad_b64: text, pt_b64: text) -> HpkeSealed {
  sk_e_b64 = random_bytes(32);
  (shared_secret, enc) = dhkem_encap_deterministic(pk_r_b64, sk_e_b64);
  if shared_secret == "" { return HpkeSealed { enc: "", ciphertext: "" }; }
  seal_with_context(shared_secret, enc, info_b64, aad_b64, pt_b64)
}

// Deterministic SealBase — TEST ONLY.  Uses the caller-supplied ephemeral
// secret `sk_e_b64` instead of a random one, so the output is reproducible
// against an RFC 9180 known-answer vector.  Never use in production.
pub fn hpke_seal_base_deterministic(pk_r_b64: text, sk_e_b64: text, info_b64: text, aad_b64: text, pt_b64: text) -> HpkeSealed {
  (shared_secret, enc) = dhkem_encap_deterministic(pk_r_b64, sk_e_b64);
  if shared_secret == "" { return HpkeSealed { enc: "", ciphertext: "" }; }
  seal_with_context(shared_secret, enc, info_b64, aad_b64, pt_b64)
}

// HPKE OpenBase (RFC 9180 §6.1): decrypt `ct` for recipient secret key `sk_r`
// using the sender's `enc`, binding `info` and `aad`.  All args base64.  Returns
// the base64 plaintext, or "" on ANY failure (bad enc/key or AEAD auth failure).
// Example: @CRY-006 — `info` AND `aad` must both match the sender's, or open gives "".
pub fn hpke_open_base(sk_r_b64: text, enc_b64: text, info_b64: text, aad_b64: text, ct_b64: text) -> text {
  shared_secret = dhkem_decap(enc_b64, sk_r_b64);
  if shared_secret == "" { return ""; }
  (key, base_nonce) = key_schedule_base(shared_secret, info_b64);
  if key == "" || base_nonce == "" { return ""; }
  aes256gcm_open(key, base_nonce, aad_b64, ct_b64)
}

// Tests for the `crypto` library.  Each test pins a known-correct
// vector (computed offline via Python's `hashlib` / `hmac` / `base64`
// stdlib) so a regression in the native impl surfaces immediately.
// SHA-256 vectors target the FIPS 180-4 padding seam (55-, 56-,
// 64-byte boundaries).  HMAC-SHA-256 covers empty inputs and the
// >64-byte key path that triggers HMAC's hash-the-key fallback.
// Base64 covers all three padding shapes (0/1/2 leftover bytes) per
// RFC 4648 §10, and base64url verifies the `+`/`/` → `-`/`_`
// substitution via a UTF-8 input that produces a `/` in standard
// base64.

use crypto;

// Helper: build a string of N copies of a single character.
fn repeat(c: text, n: integer) -> text {
  out = "";
  for _ in 0..n { out += c; }
  out
}

//
// ES256 four-target parity driver (the crypto_simple.loft convention: a `fn main`
// that `loft test` runs on interpret + --native, and that you also BUILD to the
// two wasm targets to prove the deterministic bytes are identical there):
//
//   loft tests/es256_parity.loft                 # interpret
//   loft tests/es256_parity.loft --native
//   loft tests/es256_parity.loft --native-wasm   # then: wasmtime tests/.loft/es256_parity.wasm
//   loft tests/es256_parity.loft --html          # then: serve tests/.loft/ + headless chromium
//
// The deterministic ops (public_key / sign / verify over the RFC 6979 A.2.5
// vectors) must be byte-identical on all four; keygen is random, so it is only
// checked structurally (a fresh key must sign-and-verify, and be 44 base64 chars
// = a 32-byte scalar).  Values are bound to LOCALS before each native call — a
// bare literal handed straight in as a temporary hits a native store-lifetime
// bug (arg pointer arrives null).
use crypto;

fn main() {
  sk = "ya+p2EW6dRZrXCFXZ7HWk05Qw9s26JsSe4piKxIPZyE=";
  exp_pk = "YP7UuiVanTHJYet0xjVtaMBJuJI7Yfps5mliLmDyn7Z5A/4QCLi8maQa6elWKLxk8vGyDC1+n1F3o8KU1EYimQ==";

// Copyright (c) 2026 Jurjen Stellingwerff
// SPDX-License-Identifier: LGPL-3.0-or-later
//
// Worked examples for the `crypto` library (@PLN141).  Each `test_*` below is a REAL,
// run-every-CI demonstration of a function whose correct use is not obvious from its
// signature alone; the function documents itself by CITING the tag with
// `// Example: @CRY-00N`, and `check_doc_drift.sh examples` fails if a citation goes
// dangling (no test carries the tag).  These are teaching examples, distinct from the
// known-answer suites in crypto.loft / ed25519.loft / es256.loft / aead_kex.loft /
// hpke.loft, which pin the RFC vectors instead.
//
// Why this library owes them: every parameter and return here is `text`.  The type
// checker cannot tell a message from the base64 of a message, a hex digest from a
// base64 one, a 32-byte seed from a 64-byte key, or a "" that means "it failed" from a
// "" that means "it was empty".  So every example below is a contract a caller can get
// wrong while the program type-checks perfectly.
//
// Every value is bound to a LOCAL before it reaches a native call: a bare literal or a
// function result handed straight through as a TEMPORARY hits a native store-lifetime
// bug (the argument's pointer arrives null while its length survives).  That is the
// convention in ed25519.loft and es256.loft too.
use crypto;

// @CRY-001 — two hashing doors, and why there are two.  `sha256` takes loft `text` and
// answers a 64-character lowercase HEX digest; `sha256_b64` takes the BASE64 of a raw
// byte string and answers the BASE64 of the same 32-byte digest.  Both hash bytes the
// same way — they differ only in how the input arrives and how the output is spelled,
// and nothing in `text -> text` says which one you are holding.  Reach for `sha256_b64`
// whenever what you are hashing is not valid UTF-8 (a ciphertext, a canonical CBOR
// record): such a byte string cannot ride as `text` at all, so `sha256` cannot see it.
// `bytes_to_base64` / `base64_to_bytes` are the bridge to a `vector<u8>` — READ the
// decoded vector (index it, iterate it, pass it on); never make it the left side of
// `+=`, which corrupts after a later store allocation.
fn test_hash_text_versus_bytes() {
  // The same input through both doors.
  text_in = "hello";
  hex = crypto::sha256(text_in);
  assert(hex == "2cf24dba5fb0a30e26e83b2ac5b9e29e1b161e5c1fa7425e73043362938b9824",
         "sha256 answers HEX: {hex}");
  bytes_in = crypto::base64_encode(text_in);
  b64 = crypto::sha256_b64(bytes_in);
  assert(b64 == "LPJNul+wow4m6DsqxbninhsWHlwfp0JecwQzYpOLmCQ=",
         "sha256_b64 answers BASE64 of the same 32 bytes: {b64}");
  // The difference a caller trips over: 64 hex characters against a 32-byte digest.
  // Comparing one spelling to the other can never match.
  digest_bytes = crypto::base64_to_bytes(b64);
  assert(len(hex) == 64, "the hex digest is 64 characters");
  assert(len(digest_bytes) == 32, "the base64 digest decodes to 32 bytes");
  assert(hex != b64, "two spellings of one digest are not equal");
  // A byte string that is NOT valid UTF-8 (here the single byte 0xff) can only be
  // hashed through the base64 door: `text` is UTF-8-validated, so it cannot carry it.
  raw = crypto::bytes_to_base64([255 as u8]);
  assert(raw == "/w==", "raw bytes cross the boundary as base64: {raw}");
  raw_digest = crypto::sha256_b64(raw);
  assert(raw_digest == "qBAK5qoZQNC2Y7sxzUZhQuu9vVGHExuS2TgYmHgy64k=",
         "sha256 of the byte 0xff: {raw_digest}");
}

// @CRY-002 — sign the BASE64 of your message, never the message.  Every Ed25519
// argument is base64 of bytes: the secret key is the RFC 8032 32-byte SEED (not a
// 64-byte expanded key), the public key comes from `ed25519_public_key`, and the
// message goes in as `base64_encode(msg)`.  Handing the message text straight in
// type-checks and signs whatever bytes that text happens to decode to as base64 — and
// it stays silent, because verifying with the SAME wrong spelling still succeeds.  Only
// a peer that spelled it correctly ever sees the mismatch.  Malformed input never
// halts: both producers answer "" and `ed25519_verify` answers false, so an unchecked
// "" is a signature you never made.
fn test_ed25519_sign_the_base64_of_the_message() {
  seed = "TM0Imyj/ltqdtsNG7BFOD1uKMZ81q6Yk2oz27U+4pvs=";   // RFC 8032 §7.1 TEST 2 seed
  pk = crypto::ed25519_public_key(seed);
  assert(pk == "PUAXw+hDiVqStwqnTRt+vJyYLM8uxJaMwM1V8Sr0Zgw=",
         "the 32-byte seed derives the public key: {pk}");
  // Correct: the message "test" carried as base64.
  msg = crypto::base64_encode("test");
  assert(msg == "dGVzdA==", "base64_encode is the bridge: {msg}");
  sig = crypto::ed25519_sign(seed, msg);
  ok = crypto::ed25519_verify(pk, msg, sig);
  assert(ok, "the signature verifies over the message bytes");
  // The trap: "test" is itself valid base64 (four characters), so passing it as the
  // message signs the three bytes b5 eb 2d instead of the four bytes of "test".
  wrong = "test";
  sig_wrong = crypto::ed25519_sign(seed, wrong);
  assert(sig_wrong != "", "the wrong call does not fail — it signs different bytes");
  assert(sig_wrong != sig, "so it is a different signature: {sig_wrong}");
  self_ok = crypto::ed25519_verify(pk, wrong, sig_wrong);
  assert(self_ok, "and it verifies against itself — invisible inside one program");
  peer_ok = crypto::ed25519_verify(pk, msg, sig_wrong);
  assert(!peer_ok, "only a peer that spelled the message correctly sees the mismatch");
  // Soft failure: a wrong-length seed answers "" rather than halting.
  short = "dG9vc2hvcnQ=";
  none = crypto::ed25519_sign(short, msg);
  assert(none == "", "a bad key answers an empty signature — test for it");
}

// @CRY-003 — ES256 is DETERMINISTIC, and its encodings are the flat JOSE ones.  Signing
// follows RFC 6979, so one key and one message always answer the same bytes: two
// signatures differing is a bug, not fresh randomness, and `ecdsa_p256_keygen` is the
// only call here that draws entropy.  Nothing in `text -> text` says which encoding it
// wants, and ES256 wants the JOSE forms throughout: the secret key is the bare 32-byte
// scalar `d`, the public key is the 64-byte affine point `x || y` with NO 0x04 SEC1
// prefix, and a signature is the 64-byte raw `r || s` — not DER, which belongs to the
// X.509 / PKCS#10 layer.  Feed a DER signature or an 0x04-prefixed key and you get
// `false` / "" with nothing naming the reason, so check the three lengths.
fn test_es256_deterministic_and_jose_shaped() {
  sk = crypto::ecdsa_p256_keygen();
  sk_bytes = crypto::base64_to_bytes(sk);
  assert(len(sk_bytes) == 32, "the secret key is the bare 32-byte scalar d");
  pk = crypto::ecdsa_p256_public_key(sk);
  pk_bytes = crypto::base64_to_bytes(pk);
  assert(len(pk_bytes) == 64, "the public key is 64 bytes x||y — no 0x04 prefix");
  msg = crypto::base64_encode("acme newOrder");
  sig = crypto::ecdsa_p256_sign(sk, msg);
  sig_bytes = crypto::base64_to_bytes(sig);
  assert(len(sig_bytes) == 64, "the signature is 64 bytes r||s — not DER");
  ok = crypto::ecdsa_p256_verify(pk, msg, sig);
  assert(ok, "a fresh ES256 signature verifies");
  // Deterministic: the same key and message sign to the identical bytes every time.
  again = crypto::ecdsa_p256_sign(sk, msg);
  assert(again == sig, "RFC 6979 — signing twice must answer the identical bytes");
  // Only keygen draws entropy.
  other = crypto::ecdsa_p256_keygen();
  assert(other != sk, "keygen draws a fresh scalar per call");
}

// @CRY-004 — a Diffie-Hellman output is NOT a key.  `x25519_dh(mine, theirs)` answers
// the raw 32-byte shared point and both sides compute the same one, but RFC 7748 §6.1
// says never use it directly: run it through `hkdf_sha256` so the key depends on a salt
// and a context `info`.  That `info` is what separates two keys derived from ONE shared
// secret — a send key from a receive key, one protocol version from the next — and it
// is the argument callers leave empty.  Use the split `hkdf_extract` / `hkdf_expand`
// when one PRK feeds several Expands, or when the PRK is itself the value you want
// (HPKE does both); the one-shot is exactly those two composed.  An EMPTY base64 salt
// does not mean "no salt": it selects the RFC 5869 all-zero salt.
fn test_x25519_then_hkdf_makes_the_key() {
  // RFC 7748 §6.1, the Alice / Bob example.
  alice_sk = "dwdtCnMYpX08FsFyUbJmRd9ML4frwJkqsXf7pR25LCo=";
  alice_pk = "hSDwCYkwp1R0i33ctD73Wg2/Og0mOBr066SpjqqbTmo=";
  bob_sk = "XasIfmJKikt54X+Lg4AO5m87sSkmGLb9HC+LJ/+I4Os=";
  bob_pk = "3p7bfXt9wbTTW2HC7OQ1Nz+DQ8hbeGdNrfx+FG+IK08=";
  mine = crypto::x25519_dh(alice_sk, bob_pk);
  theirs = crypto::x25519_dh(bob_sk, alice_pk);
  assert(mine == theirs, "both sides derive the same shared point");
  // The DH output is input keying material, never the key itself.
  salt = crypto::random_bytes(32);
  info = crypto::base64_encode("loft session key v1");
  key = crypto::hkdf_sha256(salt, mine, info, 32);
  key_bytes = crypto::base64_to_bytes(key);
  assert(len(key_bytes) == 32, "HKDF answers the 32-byte AEAD key");
  assert(key != mine, "the key is derived from the shared point, never equal to it");
  // A different `info` over the SAME shared secret is a different key: this is how one
  // handshake yields several keys without a second exchange.
  other_info = crypto::base64_encode("loft session key v2");
  other = crypto::hkdf_sha256(salt, mine, other_info, 32);
  assert(other != key, "info separates two keys from one shared secret");
  // The one-shot IS Extract then Expand — reach for the halves when a PRK is reused.
  prk = crypto::hkdf_extract(salt, mine);
  prk_bytes = crypto::base64_to_bytes(prk);
  assert(len(prk_bytes) == 32, "Extract answers the 32-byte PRK");
  composed = crypto::hkdf_expand(prk, info, 32);
  assert(composed == key, "hkdf_sha256 is hkdf_extract followed by hkdf_expand");
}

// @CRY-005 — AES-256-GCM: the tag rides inside the ciphertext, and "" from `open` means
// AUTHENTICATION FAILED.  `aes256gcm_seal` answers base64 of the ciphertext WITH its
// 16-byte tag appended, so the sealed blob is always 16 bytes longer than the plaintext
// and `open` takes that whole blob — there is no separate tag argument to carry.  The
// key is 32 bytes and the nonce is 12; a nonce MUST NOT repeat under one key, so draw a
// fresh `random_bytes(12)` per message and send it alongside.  The `aad` is
// authenticated but not encrypted, and must match EXACTLY on open.  The trap: `open`
// answers "" for every failure — bad key, wrong nonce, altered aad, tampered tag — and
// "" is also what an authentic EMPTY plaintext answers.  Treat "" as failure and carry
// emptiness some other way.
fn test_aes256gcm_seal_open_and_the_empty_answer() {
  key = crypto::random_bytes(32);
  nonce = crypto::random_bytes(12);
  aad = crypto::base64_encode("v1|user-42");
  pt = crypto::base64_encode("attack at dawn");
  sealed = crypto::aes256gcm_seal(key, nonce, aad, pt);
  // ciphertext || 16-byte tag — one blob, nothing separate to keep track of.
  pt_bytes = crypto::base64_to_bytes(pt);
  sealed_bytes = crypto::base64_to_bytes(sealed);
  assert(len(sealed_bytes) == len(pt_bytes) + 16,
         "sealed is the plaintext plus a 16-byte tag");
  opened = crypto::aes256gcm_open(key, nonce, aad, sealed);
  assert(opened == pt, "open recovers the plaintext");
  // The aad is authenticated: change it and open refuses.
  other_aad = crypto::base64_encode("v1|user-43");
  denied = crypto::aes256gcm_open(key, nonce, other_aad, sealed);
  assert(denied == "", "a changed aad fails authentication");
  // A fresh nonce per message: the SAME plaintext seals to a different blob, which is
  // what makes reusing one nonce under one key detectable as the mistake it is.
  nonce2 = crypto::random_bytes(12);
  sealed2 = crypto::aes256gcm_seal(key, nonce2, aad, pt);
  assert(sealed2 != sealed, "a fresh nonce gives a fresh ciphertext for one plaintext");
  opened2 = crypto::aes256gcm_open(key, nonce2, aad, sealed2);
  assert(opened2 == pt, "and its own nonce opens it");
  // The ambiguity: an authentic EMPTY plaintext opens to "" as well — the very answer a
  // forgery gets.  "" is a failure signal, never data.
  nonce3 = crypto::random_bytes(12);
  empty_sealed = crypto::aes256gcm_seal(key, nonce3, aad, "");
  empty_bytes = crypto::base64_to_bytes(empty_sealed);
  assert(len(empty_bytes) == 16, "sealing nothing yields just the tag");
  empty_opened = crypto::aes256gcm_open(key, nonce3, aad, empty_sealed);
  assert(empty_opened == "", "an authentic empty plaintext is indistinguishable from a failure");
}

// @CRY-006 — HPKE seals to a public key alone, but the recipient needs THREE things
// back: `enc`, the `ciphertext`, and the very same `info` and `aad` the sender used.
// `hpke_seal_base` draws a fresh ephemeral key on every call, so `enc` differs each
// time and MUST travel with the ciphertext — it cannot be cached, and it is not
// derivable from the recipient key.  `info` binds the key schedule (a protocol or
// version context) while `aad` binds this one message; both are authenticated, and
// either one differing makes `hpke_open_base` answer "" with nothing saying which.
// X25519 has no separate keygen, so a recipient public key is the secret scalar taken
// against the RFC 7748 base point — the recipe below.  The `_deterministic` /
// `_for_test` entry points exist only to pin the RFC 9180 vectors: they take the
// ephemeral secret from the caller, so never reach for them in real code.
fn test_hpke_transmit_enc_and_match_the_context() {
  // The CFRG RFC 9180 vector's recipient key pair.
  sk_r = "SXtFAmZM/qXVrws5k02sciQqdPhIBFHhrufWpTMgMz0=";
  // Where a public key comes from: X25519 of the secret against the base point
  // (0x09 followed by 31 zero bytes).
  base_point = "CQAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA=";
  pk_r = crypto::x25519_dh(sk_r, base_point);
  assert(pk_r == "Qw9LmFlmUUWmsbonQCRIe9ZvA6LdV313U8aNfX0AwAw=",
         "the secret scalar against the base point is the public key: {pk_r}");
  info = crypto::base64_encode("loft/hpke demo v1");
  aad = crypto::base64_encode("message 1");
  pt = crypto::base64_encode("Beauty is truth, truth beauty");
  sealed = crypto::hpke_seal_base(pk_r, info, aad, pt);
  assert(sealed.enc != "", "seal answers the encapsulated key");
  assert(sealed.ciphertext != "", "and the ciphertext");
  // Opening needs both halves, plus the identical info and aad.
  opened = crypto::hpke_open_base(sk_r, sealed.enc, info, aad, sealed.ciphertext);
  assert(opened == pt, "the recipient recovers the plaintext");
  // A fresh ephemeral key per call: `enc` differs every time, so it cannot be
  // recomputed later — it travels with the ciphertext or the message is lost.
  again = crypto::hpke_seal_base(pk_r, info, aad, pt);
  assert(again.enc != sealed.enc, "each seal encapsulates a fresh ephemeral key");
  assert(again.ciphertext != sealed.ciphertext, "so one plaintext seals differently");
  // Mismatch either bound context and open answers "" — the same "" for both.
  wrong_info = crypto::base64_encode("loft/hpke demo v2");
  bad_info = crypto::hpke_open_base(sk_r, sealed.enc, wrong_info, aad, sealed.ciphertext);
  assert(bad_info == "", "a different info fails the key schedule");
  wrong_aad = crypto::base64_encode("message 2");
  bad_aad = crypto::hpke_open_base(sk_r, sealed.enc, info, wrong_aad, sealed.ciphertext);
  assert(bad_aad == "", "a different aad fails the AEAD");
}
