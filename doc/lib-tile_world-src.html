// Copyright (c) 2026 Jurjen Stellingwerff
// SPDX-License-Identifier: LGPL-3.0-or-later
//
// tile_world — an unbounded, layered square-grid world, and the routines that answer
// questions about it: what is on a cell, how far it is to walk somewhere, what can be seen
// from where, and how much light falls on each cell.  What the pieces of a game DO —
// pushing, climbing, picking up — is the game's: this library holds the world and answers.
//
// ── The memory model ─────────────────────────────────────────────────
//
// The world is a stack of LAYERS — `z` is the floor level, negative below the ground,
// positive above — and each layer is an unbounded 2-D grid.  Like hex_world and moros's map,
// a layer is held in 32x32 CHUNKS that exist only where something was set, found through a
// hash keyed on (cx, cy, z), so a lookup costs the same in a world of ten chunks or ten
// million.  Chunk coordinates are the floor of a coordinate over 32, so the world grows west
// and north of the origin as it does east and south.
//
// A cell's TERRAIN is two bytes: a kind (0..15) in the top four bits and a tile number (the
// picture, 0..4095) in the low twelve.  A chunk is 2 KB.  A cell nobody set reads as its
// layer's GROUND (`set_ground`: open land that costs no memory), or void when the layer has
// none; a cell set to void stays void (a pit in the ground), and `clear` makes it unset again.
//
// A cell can also carry its two EDGES — the north and the west one; the east and the south
// edge are its neighbours' west and north — and a RISE, its own height in steps (a hillside).
// They share one 32-bit word a cell, in an array a chunk gets only when one of its cells first
// has an edge or a rise.
//
// THINGS move, so they are not in the terrain: each cell that holds any has a STACK (bottom
// first) in a second hash keyed on (x, y, z), removed when it empties.  A thing is a number
// the game chooses.
//
// ── Kinds ────────────────────────────────────────────────────────────
//
// The game declares what each terrain kind, each thing and each edge kind is, and the routines
// read only that: whether it can be walked on (or, for a thing, walked through; for an edge,
// walked across), whether it is opaque (stops sight and light), whether it gives light, how
// high a thing stands, and a glyph for `view`.  An undeclared terrain kind other than void is
// a wall: not walkable, opaque.  An undeclared thing is 1 high, blocks, and is not opaque.  An
// undeclared edge kind is a wall: not crossed, opaque.

/// Void: a cell nobody set.  Nothing stands there; sight and light pass.
pub const VOID = 0;

/// The darkest shade `shades` gives: a cell this many steps or more from every light.
pub const DARKEST = 7;

/// The four sides of a cell, for its edges.
pub const NORTH = 0;
pub const EAST = 1;
pub const SOUTH = 2;
pub const WEST = 3;

// A cell's raw terrain: 0 is unset (the layer's ground), 1 is void set on purpose.
const UNSET = 0;
const SET_VOID = 1;

const CHUNK = 32;

struct Chunk {
  const cx: integer,
  const cy: integer,
  const z: integer,
  cells: vector<u16>,
  // One word a cell — north edge (bits 0..7), west edge (8..15), rise (16..19) — or empty
  // while no cell of the chunk has any.
  extra: vector<u32>
}

// The things on one cell, bottom first.
struct Stack {
  const x: integer,
  const y: integer,
  const z: integer,
  things: vector<integer>
}

// What the game declared about a terrain kind, and about a thing.  Two record types, not one:
// two keyed collections of one record type in a struct share their records (loft#2216).
struct TerrainKind {
  const kd: integer,
  walk: boolean,
  opaque: boolean,
  light: boolean,
  glyph: text
}

struct EdgeKind {
  const kd: integer,
  walk: boolean,
  opaque: boolean,
  glyph: text
}

// What an unset cell of layer `gz` reads as.
struct Ground {
  const gz: integer,
  kind: integer,
  tile: integer
}

struct ThingKind {
  const kd: integer,
  walk: boolean,
  opaque: boolean,
  light: boolean,
  height: integer,
  glyph: text
}

/// The world: its chunks, its stacks, and the kinds the game declared.
pub struct World {
  chunks: hash<Chunk[cx, cy, z]>,
  stacks: hash<Stack[x, y, z]>,
  terrains: hash<TerrainKind[kd]>,
  objects: hash<ThingKind[kd]>,
  edges: hash<EdgeKind[kd]>,
  grounds: hash<Ground[gz]>
}

/// An empty world: every cell void, no kinds declared.
pub fn world() -> World {
  World { chunks: [], stacks: [], terrains: [], objects: [], edges: [], grounds: [] }
}

// --- Kinds ---

/// Declare terrain kind `kind` (1..15): whether it can be walked on, whether it stops sight
/// and light, whether it gives light (a lava floor, a window), and its glyph in `view`.
pub fn terrain_kind(w: World, kind: integer, walk: boolean, opaque: boolean,
                    light: boolean = false, glyph: text = "?") {
  w.terrains[kind] = null;
  w.terrains += [TerrainKind { kd: kind, walk: walk, opaque: opaque, light: light, glyph: glyph }];
}

/// Declare thing `thing`: how many height steps it adds to its stack, whether it blocks a cell
/// for walking (`passable`, `shortest_path`), whether it stops sight and light, whether it
/// gives light (a torch), and its glyph in `view`.
pub fn thing_kind(w: World, thing: integer, height: integer, blocks: boolean,
                  opaque: boolean = false, light: boolean = false, glyph: text = "?") {
  w.objects[thing] = null;
  w.objects += [ThingKind { kd: thing, walk: !blocks, opaque: opaque, light: light, height: height, glyph: glyph }];
}

/// Declare edge kind `kind` (1..255): whether it can be walked across (an open doorway) and
/// whether it stops sight and light (a wall, a closed door; not a fence or a window), and its
/// glyph.
pub fn edge_kind(w: World, kind: integer, walk: boolean, opaque: boolean, glyph: text = "|") {
  w.edges[kind] = null;
  w.edges += [EdgeKind { kd: kind, walk: walk, opaque: opaque, glyph: glyph }];
}

/// Can terrain kind `kind` be walked on: as declared; void and an undeclared kind cannot.
pub fn walkable_kind(w: const World, kind: integer) -> boolean {
  k = w.terrains[kind];
  if !k { return false; }
  k.walk
}

/// How many height steps thing `thing` adds: as declared, 1 for an undeclared thing.
pub fn thing_height(w: const World, thing: integer) -> integer {
  k = w.objects[thing];
  if !k { return 1; }
  k.height
}

// --- Addressing ---

/// floor(v / 32): the chunk a coordinate is in, west and north of the origin included.
pub fn chunk_of(v: integer) -> integer {
  if v >= 0 { v / CHUNK } else { -((CHUNK - 1 - v) / CHUNK) }
}

/// v mod 32 in [0, 31]: the coordinate inside its chunk.
pub fn inside(v: integer) -> integer {
  v - chunk_of(v) * CHUNK
}

/// How many chunks the world holds — the measure of its memory, 2 KB each.
pub fn chunk_count(w: const World) -> integer {
  len(w.chunks)
}

/// The layers that hold any terrain, lowest first — what saving the world walks.
pub fn layers(w: const World) -> vector<integer> {
  out: vector<integer> = [];
  for c in w.chunks {
    at = 0;
    while at < len(out) && (out[at] ?? 0) < c.z { at += 1; }
    if at == len(out) || (out[at] ?? 0) != c.z { out.insert(at, c.z); }
  }
  out
}

/// The chunks layer z holds, as (cx, cy), in reading order (row by row, west to east) — each
/// one 32x32 cells from (cx * 32, cy * 32).
pub fn chunks(w: const World, z: integer) -> vector<(integer, integer)> {
  out: vector<(integer, integer)> = [];
  for c in w.chunks {
    if c.z != z { continue; }
    at = 0;
    while at < len(out) && before(out[at].1, out[at].0, c.cy, c.cx) { at += 1; }
    out.insert(at, (c.cx, c.cy));
  }
  out
}

/// Every cell that holds things, as (x, y, z): layer by layer, lowest first, each in reading
/// order.
pub fn stacks(w: const World) -> vector<(integer, integer, integer)> {
  out: vector<(integer, integer, integer)> = [];
  for s in w.stacks {
    at = 0;
    while at < len(out) && (out[at].2 < s.z || (out[at].2 == s.z && before(out[at].1, out[at].0, s.y, s.x))) {
      at += 1;
    }
    out.insert(at, (s.x, s.y, s.z));
  }
  out
}

// Does row-major position (ay, ax) come before (by, bx)?
fn before(ay: integer, ax: integer, by: integer, bx: integer) -> boolean {
  ay < by || (ay == by && ax < bx)
}

// --- Terrain ---

// Make the chunk holding (x, y, z) when it does not exist yet.
fn ensure_chunk(w: World, x: integer, y: integer, z: integer) {
  cx = chunk_of(x);
  cy = chunk_of(y);
  if !w.chunks[cx, cy, z] {
    w.chunks += [Chunk { cx: cx, cy: cy, z: z, cells: [0 as u16; CHUNK * CHUNK], extra: [] }];
  }
}

/// Set a cell's terrain: a kind (0..15) and, for the picture, a tile number (0..4095).  VOID
/// makes the cell void even on a layer with a ground — a pit, a hole; `clear` unsets it.
pub fn set_terrain(w: World, x: integer, y: integer, z: integer, kind: integer, tile: integer = 0) {
  ensure_chunk(w, x, y, z);
  c = w.chunks[chunk_of(x), chunk_of(y), z];
  v = if kind == VOID { SET_VOID } else { (kind & 15) * 4096 + (tile & 4095) };
  c.cells[inside(y) * CHUNK + inside(x)] = (v & 65535) as u16;
}

/// Make a cell unset again: it reads as its layer's ground, or as void.
pub fn clear(w: World, x: integer, y: integer, z: integer) {
  c = w.chunks[chunk_of(x), chunk_of(y), z];
  if !c { return; }
  c.cells[inside(y) * CHUNK + inside(x)] = UNSET as u16;
}

/// Was this cell set (to anything, void included), rather than read from the ground?
pub fn is_set(w: const World, x: integer, y: integer, z: integer) -> boolean {
  raw(w, x, y, z) != UNSET
}

/// Give layer z a ground: every cell of it nobody set reads as terrain `kind` with picture
/// `tile` — open land that costs no memory.  Kind VOID takes the ground away.
pub fn set_ground(w: World, z: integer, kind: integer, tile: integer = 0) {
  w.grounds[z] = null;
  if kind != VOID { w.grounds += [Ground { gz: z, kind: kind & 15, tile: tile & 4095 }]; }
}

/// The layers that have a ground, lowest first — with `layers`, what saving a world walks.
pub fn ground_layers(w: const World) -> vector<integer> {
  out: vector<integer> = [];
  for g in w.grounds {
    at = 0;
    while at < len(out) && (out[at] ?? 0) < g.gz { at += 1; }
    out.insert(at, g.gz);
  }
  out
}

/// Layer z's ground kind, VOID when it has none.
pub fn ground(w: const World, z: integer) -> integer {
  g = w.grounds[z];
  if !g { return VOID; }
  g.kind
}

/// Layer z's ground tile, 0 when it has none.
pub fn ground_tile(w: const World, z: integer) -> integer {
  g = w.grounds[z];
  if !g { return 0; }
  g.tile
}

/// Set a rectangle of terrain, corners included.
pub fn fill(w: World, x0: integer, y0: integer, x1: integer, y1: integer, z: integer,
            kind: integer, tile: integer = 0) {
  for y in y0..=y1 {
    for x in x0..=x1 { set_terrain(w, x, y, z, kind, tile); }
  }
}

fn raw(w: const World, x: integer, y: integer, z: integer) -> integer {
  c = w.chunks[chunk_of(x), chunk_of(y), z];
  if !c { return 0; }
  (c.cells[inside(y) * CHUNK + inside(x)] ?? 0) as integer
}

/// A cell's terrain kind: what it was set to, or its layer's ground when it was not.
pub fn terrain(w: const World, x: integer, y: integer, z: integer) -> integer {
  r = raw(w, x, y, z);
  if r == UNSET { return ground(w, z); }
  if r == SET_VOID { return VOID; }
  r / 4096
}

/// A cell's tile number, its picture: what it was set to, or its layer's ground tile.
pub fn tile(w: const World, x: integer, y: integer, z: integer) -> integer {
  r = raw(w, x, y, z);
  if r == UNSET { return ground_tile(w, z); }
  if r == SET_VOID { return 0; }
  r % 4096
}

// --- Edges and rise ---

fn word(w: const World, x: integer, y: integer, z: integer) -> integer {
  c = w.chunks[chunk_of(x), chunk_of(y), z];
  if !c { return 0; }
  if len(c.extra) == 0 { return 0; }
  (c.extra[inside(y) * CHUNK + inside(x)] ?? 0) as integer
}

fn set_word(w: World, x: integer, y: integer, z: integer, v: integer) {
  ensure_chunk(w, x, y, z);
  c = w.chunks[chunk_of(x), chunk_of(y), z];
  if !c { return; }
  if len(c.extra) == 0 { c.extra = [0 as u32; CHUNK * CHUNK]; }
  c.extra[inside(y) * CHUNK + inside(x)] = (v & 2147483647) as u32;
}

// Which cell holds the edge on `side` of (x, y), and in which byte: (x, y, shift).
fn edge_home(x: integer, y: integer, side: integer) -> (integer, integer, integer) {
  if side == NORTH { return (x, y, 0); }
  if side == SOUTH { return (x, y + 1, 0); }
  if side == WEST { return (x, y, 8); }
  (x + 1, y, 8)
}

/// Put edge kind `kind` (1..255) on `side` (NORTH, EAST, SOUTH, WEST) of cell (x, y, z); 0
/// takes the edge away.  The east edge of a cell is its east neighbour's west edge, and the
/// same edge: setting one sets both.
pub fn set_edge(w: World, x: integer, y: integer, z: integer, side: integer, kind: integer) {
  h = edge_home(x, y, side);
  v = word(w, h.0, h.1, z);
  set_word(w, h.0, h.1, z, (v & ~(255 << h.2)) | ((kind & 255) << h.2));
}

/// The edge kind on `side` of cell (x, y, z), 0 for none.
pub fn edge(w: const World, x: integer, y: integer, z: integer, side: integer) -> integer {
  h = edge_home(x, y, side);
  (word(w, h.0, h.1, z) >> h.2) & 255
}

// The edge crossed stepping from (x, y) by (dx, dy), one of the four directions.
fn between(w: const World, x: integer, y: integer, z: integer, dx: integer, dy: integer) -> integer {
  side = if dx > 0 { EAST } else if dx < 0 { WEST } else if dy > 0 { SOUTH } else { NORTH };
  edge(w, x, y, z, side)
}

/// Can a walker step from (x, y, z) by (dx, dy) — one of the four directions — across the edge
/// between: no edge, or one declared walkable.  Whether the cell it lands on can be stood on is
/// `passable`'s question.
pub fn crossable(w: const World, x: integer, y: integer, z: integer, dx: integer, dy: integer) -> boolean {
  e = between(w, x, y, z, dx, dy);
  if e == 0 { return true; }
  k = w.edges[e];
  if !k { return false; }
  k.walk
}

/// Do sight and light pass from (x, y, z) by (dx, dy) across the edge between.
pub fn see_through(w: const World, x: integer, y: integer, z: integer, dx: integer, dy: integer) -> boolean {
  e = between(w, x, y, z, dx, dy);
  if e == 0 { return true; }
  k = w.edges[e];
  if !k { return false; }
  !k.opaque
}

/// Give cell (x, y, z) a rise of `steps` (0..15): its own height, under whatever stands on it.
/// A hillside is cells rising one step at a time, so a walker that steps up one at a time walks
/// up it.
pub fn set_rise(w: World, x: integer, y: integer, z: integer, steps: integer) {
  v = word(w, x, y, z);
  set_word(w, x, y, z, (v & ~(15 << 16)) | ((steps & 15) << 16));
}

/// A cell's rise in steps, 0 for a flat one.
pub fn rise(w: const World, x: integer, y: integer, z: integer) -> integer {
  (word(w, x, y, z) >> 16) & 15
}

/// Is (x, y, z) under the open sky: nothing set on the layer above it.
pub fn open_sky(w: const World, x: integer, y: integer, z: integer) -> boolean {
  terrain(w, x, y, z + 1) == VOID
}

// --- Things ---

/// Put a thing on top of a cell's stack.
pub fn place(w: World, x: integer, y: integer, z: integer, thing: integer) {
  if !w.stacks[x, y, z] {
    w.stacks += [Stack { x: x, y: y, z: z, things: [] }];
  }
  s = w.stacks[x, y, z];
  s.things += [thing];
}

/// Take the top thing off a cell's stack and answer it; 0 when the cell has none.  The stack
/// goes when it empties.
pub fn take(w: World, x: integer, y: integer, z: integer) -> integer {
  s = w.stacks[x, y, z];
  if !s { return 0; }
  n = len(s.things);
  if n == 0 { return 0; }
  t = s.things[n - 1] ?? 0;
  if n == 1 {
    w.stacks[x, y, z] = null;
  } else {
    s.things = s.things[0..n - 1];
  }
  t
}

/// The things on a cell, bottom first.
pub fn things(w: const World, x: integer, y: integer, z: integer) -> vector<integer> {
  s = w.stacks[x, y, z];
  if !s { return []; }
  out: vector<integer> = [];
  for t in s.things { out += [t]; }
  out
}

/// The top thing on a cell, 0 when there is none.
pub fn top(w: const World, x: integer, y: integer, z: integer) -> integer {
  s = w.stacks[x, y, z];
  if !s { return 0; }
  n = len(s.things);
  if n == 0 { 0 } else { s.things[n - 1] ?? 0 }
}

/// The height of a cell in steps: its rise and the declared heights of the things on it.
pub fn height(w: const World, x: integer, y: integer, z: integer) -> integer {
  s = w.stacks[x, y, z];
  if !s { return rise(w, x, y, z); }
  h = rise(w, x, y, z);
  for t in s.things { h += thing_height(w, t); }
  h
}

// --- Questions about cells ---

/// Can a cell be walked into: walkable terrain and nothing on it that blocks.
pub fn passable(w: const World, x: integer, y: integer, z: integer) -> boolean {
  if !walkable_kind(w, terrain(w, x, y, z)) { return false; }
  for t in things(w, x, y, z) {
    k = w.objects[t];
    if !k || !k.walk { return false; }
  }
  true
}

/// Does a cell stop sight and light: opaque terrain, or an opaque thing on it.  An
/// undeclared terrain kind (not void) is opaque.
pub fn opaque(w: const World, x: integer, y: integer, z: integer) -> boolean {
  kind = terrain(w, x, y, z);
  if kind != VOID {
    k = w.terrains[kind];
    if !k || k.opaque { return true; }
  }
  for t in things(w, x, y, z) {
    o = w.objects[t];
    if o && o.opaque { return true; }
  }
  false
}

/// Does a cell give light: under the open sky (by day: `daylight`, the default), lit terrain, or a
/// lit thing on it.  A void cell gives none.
pub fn lit(w: const World, x: integer, y: integer, z: integer, daylight: boolean = true) -> boolean {
  kind = terrain(w, x, y, z);
  if kind == VOID { return false; }
  if daylight && open_sky(w, x, y, z) { return true; }
  k = w.terrains[kind];
  if k && k.light { return true; }
  for t in things(w, x, y, z) {
    o = w.objects[t];
    if o && o.light { return true; }
  }
  false
}

// --- Routines ---

/// The shortest walk from (x0, y0) to (x1, y1) on layer z, in steps to the four neighbours over
/// `passable` cells and `crossable` edges: the cells passed through, the goal last, the start left out.  Empty when
/// the goal cannot be reached within `limit` steps (or is the start).  The goal itself need
/// not be passable, so a path can lead up to a thing to act on.
pub fn shortest_path(w: const World, x0: integer, y0: integer, x1: integer, y1: integer,
                     z: integer, limit: integer = 64) -> vector<(integer, integer)> {
  out: vector<(integer, integer)> = [];
  if (x0 == x1 && y0 == y1) || limit <= 0 { return out; }
  // A square of `limit` round the start holds every cell such a walk can reach.
  gx = x0 - limit;
  gy = y0 - limit;
  gw = 2 * limit + 1;
  from: vector<integer> = [-1; gw * gw];
  start = limit * gw + limit;
  from[start] = start;
  queue: vector<integer> = [start];
  at = 0;
  found = -1;
  while at < len(queue) && found < 0 {
    c = queue[at] ?? 0;
    at += 1;
    cx = c % gw ?? 0;
    cy = c / gw ?? 0;
    for n in [(cx + 1, cy), (cx - 1, cy), (cx, cy + 1), (cx, cy - 1)] {
      nx = n.0;
      ny = n.1;
      if nx < 0 || nx >= gw || ny < 0 || ny >= gw || (from[ny * gw + nx] ?? 0) >= 0 { continue; }
      if !crossable(w, gx + cx, gy + cy, z, nx - cx, ny - cy) { continue; }
      goal = gx + nx == x1 && gy + ny == y1;
      if !goal && !passable(w, gx + nx, gy + ny, z) { continue; }
      from[ny * gw + nx] = c;
      if goal {
        found = ny * gw + nx;
        break;
      }
      queue += [ny * gw + nx];
    }
  }
  if found < 0 { return out; }
  back: vector<integer> = [];
  c = found;
  while c != start {
    back += [c];
    c = from[c] ?? start;
  }
  if len(back) > limit { return out; }
  i = len(back) - 1;
  while i >= 0 {
    b = back[i] ?? 0;
    out += [(gx + (b % gw ?? 0), gy + (b / gw ?? 0))];
    i -= 1;
  }
  out
}

/// Can (x1, y1) be seen from (x0, y0) on layer z: no opaque cell on the straight line between
/// them (Bresenham's), and no opaque edge it crosses.  The two ends themselves do not count, so a wall is seen, and an eye
/// in a doorway sees out.
pub fn line_of_sight(w: const World, x0: integer, y0: integer, x1: integer, y1: integer,
                     z: integer) -> boolean {
  dx = if x1 > x0 { x1 - x0 } else { x0 - x1 };
  dy = if y1 > y0 { y0 - y1 } else { y1 - y0 };
  sx = if x0 < x1 { 1 } else { -1 };
  sy = if y0 < y1 { 1 } else { -1 };
  err = dx + dy;
  x = x0;
  y = y0;
  while x != x1 || y != y1 {
    if (x != x0 || y != y0) && opaque(w, x, y, z) { return false; }
    e2 = 2 * err;
    mx = e2 >= dy;
    my = e2 <= dx;
    // The edges the step crosses: a diagonal step sees past a corner when either way round it
    // is open.
    if mx && my {
      round_x = see_through(w, x, y, z, sx, 0) && see_through(w, x + sx, y, z, 0, sy);
      round_y = see_through(w, x, y, z, 0, sy) && see_through(w, x, y + sy, z, sx, 0);
      if !round_x && !round_y { return false; }
    } else if mx {
      if !see_through(w, x, y, z, sx, 0) { return false; }
    } else if my {
      if !see_through(w, x, y, z, 0, sy) { return false; }
    }
    if mx {
      err += dy;
      x += sx;
    }
    if my {
      err += dx;
      y += sy;
    }
  }
  true
}

/// How dark each cell of (x0, y0) .. (x1, y1) on layer z is, row by row: 0 at a light, one
/// shade darker for each step from the nearest one, up to `DARKEST`, never black.  A light is
/// a `lit` cell or one of `lights` (cells the game says give light now: what a figure carries,
/// a torch).  A light of `lights` starts at its shade in `start` (0, the default, is full; a
/// dimmer light — a torch burning down — reaches less far), the k-th entry for the k-th light.
/// Light spreads over every cell but an opaque one, which is lit but passes no light on unless
/// it is itself a light, and never across an opaque edge.  A light up to `DARKEST` cells outside the rectangle counts.
/// `daylight` false: the open sky is no light — night, when only what gives light of itself (a
/// lamp, a lit terrain) and `lights` light the world.
pub fn shades(w: const World, x0: integer, y0: integer, x1: integer, y1: integer, z: integer,
              lights: const vector<(integer, integer)> = [],
              start: const vector<integer> = [], daylight: boolean = true) -> vector<integer> {
  gx = x0 - DARKEST;
  gy = y0 - DARKEST;
  gw = x1 - x0 + 1 + 2 * DARKEST;
  gh = y1 - y0 + 1 + 2 * DARKEST;
  dist: vector<integer> = [DARKEST; gw * gh];
  source: vector<boolean> = [false; gw * gh];
  for j in 0..gh {
    for i in 0..gw {
      if lit(w, gx + i, gy + j, z, daylight) {
        dist[j * gw + i] = 0;
        source[j * gw + i] = true;
      }
    }
  }
  for k in 0..len(lights) {
    l = lights[k];
    i = l.0 - gx;
    j = l.1 - gy;
    from = start[k] ?? 0;
    if from < 0 { from = 0; }
    if i >= 0 && i < gw && j >= 0 && j < gh && from < (dist[j * gw + i] ?? 0) {
      dist[j * gw + i] = from;
      source[j * gw + i] = true;
    }
  }
  // Shade by shade, so a cell takes its light from the brightest way first: the cells a shade
  // reached, with the lights that start at that shade, spread one shade darker.
  frontier: vector<integer> = [];
  for s in 0..DARKEST - 1 {
    for c in 0..gw * gh {
      if (source[c] ?? false) && (dist[c] ?? DARKEST) == s { frontier += [c]; }
    }
    next: vector<integer> = [];
    for c in frontier {
      if (dist[c] ?? DARKEST) != s { continue; }
      cx = c % gw ?? 0;
      cy = c / gw ?? 0;
      if !(source[c] ?? false) && opaque(w, gx + cx, gy + cy, z) { continue; }
      for n in [(cx - 1, cy), (cx + 1, cy), (cx, cy - 1), (cx, cy + 1)] {
        nx = n.0;
        ny = n.1;
        if nx >= 0 && nx < gw && ny >= 0 && ny < gh && (dist[ny * gw + nx] ?? 0) > s + 1
            && see_through(w, gx + cx, gy + cy, z, nx - cx, ny - cy) {
          dist[ny * gw + nx] = s + 1;
          next += [ny * gw + nx];
        }
      }
    }
    frontier = next;
  }
  out: vector<integer> = [];
  for y in y0..=y1 {
    for x in x0..=x1 {
      out += [dist[(y - gy) * gw + (x - gx)] ?? DARKEST];
    }
  }
  out
}

/// A layer as text, one line per row: each cell's top thing's glyph, or its terrain's, or a
/// space for void.  `marks` puts a glyph of the game's at a cell (a figure): (x, y, glyph).
pub fn view(w: const World, x0: integer, y0: integer, x1: integer, y1: integer, z: integer,
            marks: const vector<(integer, integer, text)> = []) -> text {
  out = "";
  for y in y0..=y1 {
    for x in x0..=x1 {
      g = "";
      for m in marks {
        if m.0 == x && m.1 == y { g = m.2; }
      }
      if g == "" {
        t = top(w, x, y, z);
        o = w.objects[t];
        if t != 0 && o {
          g = o.glyph;
        } else if t != 0 {
          g = "?";
        } else {
          kind = terrain(w, x, y, z);
          k = w.terrains[kind];
          g = if kind == VOID { " " } else if k { k.glyph } else { "#" };
        }
      }
      out += g;
    }
    out += "\n";
  }
  out
}

struct Seen {
  const sx: integer,
  const sy: integer
}

/// The cells of layer z connected to (x, y) that hold the same terrain as it — the same kind
/// and tile, set or read from the ground — four ways, never across an edge between two cells:
/// what a fill paints.  Nearest first (breadth first), (x, y) itself first, and at most `limit`
/// of them, because a layer's ground has no end: a region exactly `limit` long may go on.
pub fn region(w: const World, x: integer, y: integer, z: integer, limit: integer) -> vector<(integer, integer)> {
  out: vector<(integer, integer)> = [];
  if limit <= 0 { return out; }
  kind = terrain(w, x, y, z);
  t = tile(w, x, y, z);
  seen: hash<Seen[sx, sy]> = [Seen { sx: x, sy: y }];
  out += [(x, y)];
  at = 0;
  while at < len(out) && len(out) < limit {
    c = out[at] ?? (x, y);
    at += 1;
    for d in [(1, 0), (-1, 0), (0, 1), (0, -1)] {
      nx = c.0 + d.0;
      ny = c.1 + d.1;
      if len(out) >= limit || seen[nx, ny] { continue; }
      if between(w, c.0, c.1, z, d.0, d.1) != 0 { continue; }
      if terrain(w, nx, ny, z) != kind || tile(w, nx, ny, z) != t { continue; }
      seen += [Seen { sx: nx, sy: ny }];
      out += [(nx, ny)];
    }
  }
  out
}

// Set a cell's raw terrain word (as `raw` reads it), making its chunk when needed.
fn put_raw(w: World, x: integer, y: integer, z: integer, r: integer) {
  ensure_chunk(w, x, y, z);
  c = w.chunks[chunk_of(x), chunk_of(y), z];
  if !c { return; }
  c.cells[inside(y) * CHUNK + inside(x)] = (r & 65535) as u16;
}

/// A stamp: the cells of the box x0..x1, y0..y1 on layers z0..z1, corners included — terrain
/// as set (an unset cell stays unset), rise, the north and west edges each cell holds, and the
/// stacks — moved so the corner (x0, y0, z0) is (0, 0, 0).  The kinds come along: the stamp is
/// a world of its own, to `paste` back or keep.
pub fn copy(w: const World, x0: integer, y0: integer, x1: integer, y1: integer, z0: integer,
            z1: integer) -> World {
  out = world();
  out.terrains = w.terrains;
  out.objects = w.objects;
  out.edges = w.edges;
  for z in z0..=z1 {
    for y in y0..=y1 {
      for x in x0..=x1 {
        r = raw(w, x, y, z);
        if r != UNSET { put_raw(out, x - x0, y - y0, z - z0, r); }
        v = word(w, x, y, z);
        if v != 0 { set_word(out, x - x0, y - y0, z - z0, v); }
        for t in things(w, x, y, z) { place(out, x - x0, y - y0, z - z0, t); }
      }
    }
  }
  out
}

/// Write stamp `from` (a `copy`) into `w` with its (0, 0, 0) at (x, y, z).  Each cell the stamp
/// holds anything at replaces the cell there: its terrain when the stamp set it, its rise and
/// north and west edges, and its stack (none when the stamp's cell has none).  A cell the
/// stamp holds nothing at leaves `w` as it was.
pub fn paste(w: World, from: const World, x: integer, y: integer, z: integer) {
  for c in from.chunks {
    for j in 0..CHUNK {
      for i in 0..CHUNK {
        fx = c.cx * CHUNK + i;
        fy = c.cy * CHUNK + j;
        r = raw(from, fx, fy, c.z);
        v = word(from, fx, fy, c.z);
        if r == UNSET && v == 0 { continue; }
        if r != UNSET { put_raw(w, x + fx, y + fy, z + c.z, r); }
        if v != 0 || word(w, x + fx, y + fy, z + c.z) != 0 { set_word(w, x + fx, y + fy, z + c.z, v); }
        w.stacks[x + fx, y + fy, z + c.z] = null;
      }
    }
  }
  for s in from.stacks {
    w.stacks[x + s.x, y + s.y, z + s.z] = null;
    for t in s.things { place(w, x + s.x, y + s.y, z + s.z, t); }
  }
}

// A cell's east edge is its east neighbour's west edge; its south edge is the cell below's north.
fn test_an_edge_is_shared_by_its_two_cells() {

  assert(tile_world::edge(w, 2, 1, 0, tile_world::EAST) == 0, "taken away from either side");

// Crossing and seeing across each edge kind, both ways.
fn test_crossing_and_seeing() {
  w = room();
  tile_world::set_edge(w, 1, 0, 0, tile_world::EAST, E_WALL);
  tile_world::set_edge(w, 1, 1, 0, tile_world::EAST, E_DOOR);
  tile_world::set_edge(w, 1, 2, 0, tile_world::EAST, E_WINDOW);
  assert(!tile_world::crossable(w, 1, 0, 0, 1, 0) && !tile_world::crossable(w, 2, 0, 0, -1, 0), "a wall, both ways");
  assert(!tile_world::see_through(w, 1, 0, 0, 1, 0), "a wall stops sight");
  assert(tile_world::crossable(w, 2, 1, 0, -1, 0) && !tile_world::see_through(w, 2, 1, 0, -1, 0), "a door: crossed, not seen through");
  assert(!tile_world::crossable(w, 1, 2, 0, 1, 0) && tile_world::see_through(w, 1, 2, 0, 1, 0), "a window: seen through, not crossed");

// Wall edges between x 1 and x 2 on rows 0 and 1 leave a gap on row 2: from (0, 0) to (3, 0) the
// walk goes down to row 2, across, and back up: 3 across + 2 down + 2 up = 7 steps.
fn test_a_path_goes_round_wall_edges() {
  w = room();
  assert(len(tile_world::shortest_path(w, 0, 0, 3, 0, 0)) == 3, "3 steps with no edges");
  tile_world::set_edge(w, 1, 0, 0, tile_world::EAST, E_WALL);
  tile_world::set_edge(w, 1, 1, 0, tile_world::EAST, E_WALL);
  p = tile_world::shortest_path(w, 0, 0, 3, 0, 0);
  assert(len(p) == 7, "7 steps, got {len(p)}");
  px = 0;
  crossed_at = -1;
  for c in p {
    if px == 1 && c.0 == 2 { crossed_at = c.1; }
    px = c.0;
  }
  assert(crossed_at == 2, "crossed from x 1 to x 2 on row 2, got {crossed_at}");

  assert(len(tile_world::shortest_path(w, 0, 0, 3, 0, 0)) == 7, "a door on the gap is still a way through");
  tile_world::set_edge(w, 1, 2, 0, tile_world::EAST, E_WINDOW);
  assert(len(tile_world::shortest_path(w, 0, 0, 3, 0, 0)) == 0, "a window is not");
}

// Floor x 0..4, y 0..2 on layer 0, with a wall at (2, 0) and (2, 1): the west part (x 0..1)
// meets the east part (x 3..4) only through (2, 2).

//   . . # . .

fn room() -> tile_world::World {

  tile_world::fill(w, 0, 0, 4, 2, 0, FLOOR);
  tile_world::set_terrain(w, 2, 0, 0, WALL);

// The room's corner x 0..1, y 0..1, with a rise 2 at (0, 0), a box at (1, 1) and a fence west
// of (1, 0), stamped at (10, 10, 0) into an empty world: the four cells, the rise, the box and
// the fence come along; (12, 10), outside the stamp, stays unset.
fn test_a_stamp_is_copied_and_pasted() {
  w = room();
  tile_world::set_rise(w, 0, 0, 0, 2);
  tile_world::place(w, 1, 1, 0, BOX);
  tile_world::set_edge(w, 1, 0, 0, tile_world::WEST, FENCE);
  stamp = tile_world::copy(w, 0, 0, 1, 1, 0, 0);
  assert(tile_world::terrain(stamp, 1, 1, 0) == FLOOR && tile_world::rise(stamp, 0, 0, 0) == 2, "the stamp holds them");
  t = tile_world::world();
  kinds(t);
  tile_world::paste(t, stamp, 10, 10, 0);
  assert(tile_world::terrain(t, 10, 10, 0) == FLOOR && tile_world::terrain(t, 11, 11, 0) == FLOOR, "the floor");
  assert(tile_world::rise(t, 10, 10, 0) == 2, "the rise");
  assert(tile_world::top(t, 11, 11, 0) == BOX, "the box");
  assert(tile_world::edge(t, 11, 10, 0, tile_world::WEST) == FENCE, "the fence");
  assert(!tile_world::is_set(t, 12, 10, 0), "outside the stamp");
}

// Pasting over a stack replaces it: a lamp at (11, 11) under the stamp's box leaves the box
// alone, and a lamp under a stamp cell with no stack (10, 11) is gone.  A cell the stamp left
// unset keeps what was there: (2, 0) of a stamp of x 2..2 copied from an unset world.
fn test_a_paste_replaces_what_it_covers() {
  w = room();
  tile_world::place(w, 1, 1, 0, BOX);
  stamp = tile_world::copy(w, 0, 0, 1, 1, 0, 0);
  t = room();
  tile_world::place(t, 11, 11, 0, LAMP);
  tile_world::place(t, 10, 11, 0, LAMP);
  tile_world::paste(t, stamp, 10, 10, 0);
  assert(len(tile_world::things(t, 11, 11, 0)) == 1 && tile_world::top(t, 11, 11, 0) == BOX, "the box alone");
  assert(tile_world::top(t, 10, 11, 0) == 0, "the lamp gone");
  empty = tile_world::world();
  tile_world::paste(t, tile_world::copy(empty, 0, 0, 0, 0, 0, 0), 2, 0, 0);
  assert(tile_world::terrain(t, 2, 0, 0) == WALL, "an unset stamp cell changes nothing");
}

use tile_world;

const FLOOR = 1;
const WALL = 2;
const WINDOW = 3;
const BOX = 1;
const LAMP = 2;
const CURTAIN = 3;

fn kinds(w: tile_world::World) {
  tile_world::terrain_kind(w, FLOOR, true, false, false, ".");
  tile_world::terrain_kind(w, WALL, false, true, false, "#");
  tile_world::terrain_kind(w, WINDOW, false, false, false, "=");
  tile_world::thing_kind(w, BOX, 1, true, false, false, "b");
  tile_world::thing_kind(w, LAMP, 0, false, false, true, "*");
  tile_world::thing_kind(w, CURTAIN, 0, false, true, false, "|");
}

// A room x 0 .. 4, y 0 .. 2 on layer 0, walled round, under a ceiling (layer 1 set over it),
// so nothing is open sky.
fn room() -> tile_world::World {
  w = tile_world::world();
  kinds(w);
  tile_world::fill(w, -1, -1, 5, 3, 0, WALL);
  tile_world::fill(w, 0, 0, 4, 2, 0, FLOOR);
  tile_world::fill(w, -1, -1, 5, 3, 1, FLOOR);
  w
}

fn main() {

}

fn test_addressing() {

fn test_sparse_layers() {

// A wall down x 2 for y 0 .. 1 leaves a gap at y 2.  From (0, 0) to (4, 0) every shortest walk
// goes through the gap: 4 steps east, 2 down and 2 back up, 8.
fn test_shortest_path_goes_round() {
  w = room();
  tile_world::fill(w, 2, 0, 2, 1, 0, WALL);
  p = tile_world::shortest_path(w, 0, 0, 4, 0, 0);
  assert(len(p) == 8, "8 steps, got {len(p)}");
  assert(p[7].0 == 4 && p[7].1 == 0, "ends at the goal");
  px = 0;
  py = 0;
  through_gap = false;
  for c in p {
    d = (c.0 - px) * (c.0 - px) + (c.1 - py) * (c.1 - py);
    assert(d == 1, "each step to a neighbour: {px},{py} -> {c.0},{c.1}");
    if c.0 == 2 && c.1 == 2 { through_gap = true; }
    px = c.0;
    py = c.1;
  }
  assert(through_gap, "through the gap at (2, 2)");
}

fn test_shortest_path_limits() {
  w = room();
  tile_world::fill(w, 2, 0, 2, 2, 0, WALL);

  assert(len(tile_world::shortest_path(v, 0, 0, 4, 2, 0, 6)) == 6, "limit 6 reaches it");
  assert(len(tile_world::shortest_path(v, 0, 0, 0, 0, 0)) == 0, "the start itself");
}

// Light along the room's row y 0, read from x -1 (the west wall) to x 5 (the east wall).
fn row(w: const tile_world::World, lights: const vector<(integer, integer)>) -> text {
  t = "";
  for s in tile_world::shades(w, -1, 0, 5, 0, 0, lights) { t += "{s},"; }
  t
}

fn test_shades() {
  w = room();
  none: vector<(integer, integer)> = [];
  assert(row(w, none) == "7,7,7,7,7,7,7,", "dark, never black");
  tile_world::place(w, 0, 0, 0, LAMP);
  assert(row(w, none) == "1,0,1,2,3,4,5,", "a lamp at x 0: {row(w, none)}");
  // A single wall cell at (2, 0) is walked round through row 1: (3, 0) is 5 steps from the lamp.
  tile_world::set_terrain(w, 2, 0, 0, WALL);
  assert(row(w, none) == "1,0,1,2,5,6,7,", "light goes round one wall cell: {row(w, none)}");
  // A whole wall down x 2 is lit and stops it.
  tile_world::fill(w, 2, 0, 2, 2, 0, WALL);
  assert(row(w, none) == "1,0,1,2,7,7,7,", "a wall at x 2 is lit and stops it: {row(w, none)}");
  carried: vector<(integer, integer)> = [(4, 0)];
  assert(row(w, carried) == "1,0,1,2,1,0,1,", "a light the game passes in, at x 4: {row(w, carried)}");
  tile_world::set_terrain(w, 4, 0, 1, tile_world::VOID);
  assert(row(w, none) == "1,0,1,2,1,0,1,", "the open sky over x 4 is a light: {row(w, none)}");
}

// Night: `daylight` false, the open sky gives no light — the cells open to it are as dark as any,
// and only a lamp or a light the game passes in lights them.
fn test_night() {
  w = room();
  none: vector<(integer, integer)> = [];
  tile_world::set_terrain(w, 4, 0, 1, tile_world::VOID);
  assert(row(w, none) == "5,4,3,2,1,0,1,", "by day the open sky over x 4 is a light: {row(w, none)}");
  t = "";
  for s in tile_world::shades(w, -1, 0, 5, 0, 0, none, [], false) { t += "{s},"; }
  assert(t == "7,7,7,7,7,7,7,", "by night it is not: {t}");
  tile_world::place(w, 0, 0, 0, LAMP);
  t = "";
  for s in tile_world::shades(w, -1, 0, 5, 0, 0, none, [], false) { t += "{s},"; }
  assert(t == "1,0,1,2,3,4,5,", "a lamp still lights by night: {t}");
}

// Light along row y 0 from lights that start dimmer: the k-th light at shade `start[k]` (a torch
// burning down), a missing entry full.
fn dim_row(w: const tile_world::World, lights: const vector<(integer, integer)>,
           start: const vector<integer>) -> text {
  t = "";
  for s in tile_world::shades(w, -1, 0, 5, 0, 0, lights, start) { t += "{s},"; }
  t
}

fn test_dimmed_lights() {
  w = room();
  at0: vector<(integer, integer)> = [(0, 0)];
  assert(dim_row(w, at0, [3]) == "4,3,4,5,6,7,7,", "a light at x 0 from shade 3: {dim_row(w, at0, [3])}");
  // The dim one at x 0 keeps its 3 (the bright one at x 4 reaches it at 4); x 1 takes 3 from the
  // bright side.  The second light has no entry: full.
  both: vector<(integer, integer)> = [(0, 0), (4, 0)];
  assert(dim_row(w, both, [3]) == "4,3,3,2,1,0,1,", "the brighter way wins: {dim_row(w, both, [3])}");
  // A dim light on the wall itself (opaque) is a light: it spreads.
  wall: vector<(integer, integer)> = [(-1, 0)];
  assert(dim_row(w, wall, [2]) == "2,3,4,5,6,7,7,", "a wall light from shade 2: {dim_row(w, wall, [2])}");
  assert(dim_row(w, at0, [7]) == "7,7,7,7,7,7,7,", "as dark as the dark: nothing");
}

// What a world holds, listed: layers lowest first, a layer's chunks in reading order, the
// cells with things layer by layer in reading order.  Chunks: (-1, 0) holds x -32 .. -1.
