// cellturn — A STRUCTURE TURNED BY ANY NUMBER OF 30-DEGREE STEPS.
//
// `hex_field::stencil_rotate` turns a stencil by 60 degrees, exactly: a 60-degree turn is a
// symmetry of the lattice.  `stencil_rotate_deg` REFUSES anything else, and it is right to: an
// exact routine must not round.  But a house is drawn at twelve angles, not six, and half of
// them are no symmetry of the lattice at all.  This is the routine for those — approximate, and
// named as such:
//
//   an even step   every cell turned about the pivot (`lattice_turned`).  Exact.
//   an odd step    not exact.  The structure's drawn WALLS (`hex_draw::cells_built`) are turned
//                  30 degrees, each moved to the nearest line a wall can stand on in the turned
//                  lattice — midway between two rows of cell centres — and every cell inside
//                  joins (`TURN_BY_WALLS`).  Then the remaining 60-degree steps.
//
// What an odd step cannot keep exactly it REPORTS: the structure may move by half a row along
// either axis to bring its walls nearest their lines, and that offset comes back with the cells
// (`Turned`), so anything standing on the structure — a door, a yard, a bush on its roof — can
// be carried through the same turn (`turned_point`).
//
// What it asks of a structure: one piece without a courtyard, its walls along two axes at a
// right angle, big enough for its cells to carry its walls.  Turn from the drawing each time,
// never from an earlier turn: two odd steps are two approximations.
//
// It reads nothing but the cells, so it does not matter which way the structure was drawn, or
// how.  Extracted from tile-world-game, whose rules these routines keep and cite by name
// (the HT and HO rules, its `formal/hex-houses.md`).
use hex_field::*;
use hex_form::*;
use hex_draw::*;

// The ways an odd step of 30 degrees is made.  `TURN_BY_WALLS` is the one to use; the other moves
// the outline as a whole until it holds as many cells as the drawing, and is kept to compare.
pub const TURN_BY_WALLS = 0;
pub const TURN_BY_OUTLINE = 1;

// The spacing of the lines a wall can stand on, across the rows of hex centres, in world units:
// rows across an odd heading lie 1.5 units apart, across an even heading sqrt(3)/2 units; a wall
// stands midway between two rows.
pub const TURN_ROW_ODD = 1.5;
pub const TURN_ROW_EVEN = 0.8660254037844386;

// A structure turned: its cells (a new set round the pivot), the pivot's centre in doubled
// coordinates, the steps turned (0..11), and how far an odd step moved the structure — its
// offset in world units, taken after its own 30 degrees and before the remaining 60-degree
// steps.  Zero for an even number of steps.
pub struct Turned {
  pub tn_cells: HexSet,
  pub tn_pk:    integer,
  pub tn_pm:    integer,
  pub tn_steps: integer,
  pub tn_sx:    float,
  pub tn_sy:    float,
}

// A cell as an offset from the pivot cell, in doubled coordinates.
struct TurnOffset {
  dk: integer,
  dm: integer,
}

// @FR-HT-Pivot, @FR-HT-Turn60, @FR-HT-Drawing: whole 30-degree steps about a cell; 60-degree steps cell for cell
// `cells` turned by `steps` x 30 degrees anticlockwise about cell (pq, pr).
pub fn cells_turn(cells: HexSet, pq: integer, pr: integer, steps: integer,
                  way: integer = TURN_BY_WALLS) -> Turned {
  s = ((steps % 12) + 12) % 12 ?? 0;
  r = turn_reach(cells, pq, pr);
  pk = lattice_k(pq, pr);
  pm = lattice_m(pq, pr);
  sx = 0.0;
  sy = 0.0;
  base: vector<TurnOffset> = [];
  if s % 2 == 1 {
    t = turn_thirty(cells, pq, pr, r, way);
    base = t.offsets;
    sx = t.sx;
    sy = t.sy;
  } else {
    base = turn_offsets(cells, pk, pm);
  }
  out = hexset_chunk(pq - r, pr - r, 2 * r + 1, 2 * r + 1);
  for o in base {
    (k, m) = lattice_turned(o.dk, o.dm, s / 2 ?? 0);
    (q, rr) = lattice_to_cell(pk + k, pm + m);
    hexset_set(out, q, rr, true);
  }
  Turned { tn_cells: out, tn_pk: pk, tn_pm: pm, tn_steps: s, tn_sx: sx, tn_sy: sy }
}

// Where the point (x, y) of the drawn structure lands in the turn `t`: turned about the pivot,
// and moved by the offset an odd step moved the structure by.  For anything placed on a
// structure that turns with it.  A few multiplications: the turn itself is not done again.
pub fn turned_point(t: const Turned, x: float, y: float) -> (float, float) {
  px = (t.tn_pk as float) * 0.8660254037844386;
  py = (t.tn_pm as float) * 0.5;
  a = (t.tn_steps as float) * PI / 6.0;
  rest = ((t.tn_steps / 2 ?? 0) as float) * PI / 3.0;
  // the odd step's offset is taken after its own 30 degrees and before the remaining 60-degree steps
  dx = (x - px) * cos(a) - (y - py) * sin(a) + (t.tn_sx * cos(rest) - t.tn_sy * sin(rest));
  dy = (x - px) * sin(a) + (y - py) * cos(a) + (t.tn_sx * sin(rest) + t.tn_sy * cos(rest));
  (px + dx, py + dy)
}

// The structure's cell nearest its middle (the mean of its centres): a pivot to turn it about
// when none is given.
pub fn cells_middle(cells: HexSet) -> (integer, integer) {
  all = turn_offsets(cells, 0, 0);
  n = len(all);
  if n == 0 { return (0, 0); }
  sk = 0;
  sm = 0;
  for c in all {
    sk += c.dk;
    sm += c.dm;
  }
  best = 0;
  best_d = -1;
  for i in 0..n {
    // 4n^2 x the squared world distance to the mean, exact
    ek = all[i].dk * n - sk;
    em = all[i].dm * n - sm;
    d = 3 * ek * ek + em * em;
    if best_d < 0 || d < best_d { best = i; best_d = d; }
  }
  lattice_to_cell(all[best]!.dk, all[best]!.dm)
}

// The structure's cells as offsets from doubled coordinates (pk, pm).
fn turn_offsets(cells: HexSet, pk: integer, pm: integer) -> vector<TurnOffset> {
  out: vector<TurnOffset> = [];
  for fr in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for fq in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if hexset_get(cells, fq, fr) {
        out += [TurnOffset { dk: lattice_k(fq, fr) - pk, dm: lattice_m(fq, fr) - pm }];
      }
    }
  }
  out
}

// How far round the pivot the structure, turned any way, can reach: a window for it.
fn turn_reach(cells: HexSet, pq: integer, pr: integer) -> integer {
  r = 0;
  for fr in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for fq in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if hexset_get(cells, fq, fr) {
        dq = if fq > pq { fq - pq } else { pq - fq };
        dr = if fr > pr { fr - pr } else { pr - fr };
        if dq + dr > r { r = dq + dr; }
      }
    }
  }
  r + 3
}

// One odd step of 30 degrees: the structure's offsets from the pivot once turned 30 degrees,
// and the offset it was moved by.
struct TurnThirty {
  offsets: vector<TurnOffset>,
  sx: float,
  sy: float,
  // the frames it was made in: the drawing's axes (heading `c` and three on) and the turned
  // ones, the rows the walls stand between along each, and the offset the walls were moved by
  c: integer,
  co0: float,
  si0: float,
  co1: float,
  si1: float,
  row_x: float,
  row_y: float,
  off_x: float,
  off_y: float,
}

// @FR-HT-Turn30: one odd step, from the drawn walls
fn turn_thirty(cells: HexSet, pq: integer, pr: integer, r: integer, way: integer) -> TurnThirty {
  b = cells_built(cells, 1);
  c = cells_axes(b.cw_boundary);
  if c < 0 { c = 0; }
  pk = lattice_k(pq, pr);
  pm = lattice_m(pq, pr);
  px = (pk as float) * 0.8660254037844386;
  py = (pm as float) * 0.5;
  // The drawn walls in the structure's own frame (along heading c and c + 3).  Turned 30 degrees,
  // the outline keeps these coordinates in the frame along c + 1 and c + 4.
  a0 = (c as float) * PI / 6.0;
  co0 = cos(a0);
  si0 = sin(a0);
  corners: vector<(float, float)> = [for p in b.cw_corners { ((p.0 - px) * co0 + (p.1 - py) * si0, -(p.0 - px) * si0 + (p.1 - py) * co0) }];
  row_x = if (c + 1) % 2 == 1 { TURN_ROW_ODD } else { TURN_ROW_EVEN };
  row_y = if (c + 4) % 2 == 1 { TURN_ROW_ODD } else { TURN_ROW_EVEN };
  // Every cell near the pivot, in the turned frame: one cosine and sine for them all.
  a1 = ((c + 1) as float) * PI / 6.0;
  co1 = cos(a1);
  si1 = sin(a1);
  ks: vector<integer> = [];
  ms: vector<integer> = [];
  us: vector<float> = [];
  vs: vector<float> = [];
  for fr in pr - r..pr + r + 1 {
    for fq in pq - r..pq + r + 1 {
      dk = lattice_k(fq, fr) - pk;
      dm = lattice_m(fq, fr) - pm;
      dx = (dk as float) * 0.8660254037844386;
      dy = (dm as float) * 0.5;
      ks += [dk];
      ms += [dm];
      us += [dx * co1 + dy * si1];
      vs += [-dx * si1 + dy * co1];
    }
  }
  poly: vector<(float, float)> = [];
  sx = 0.0;
  sy = 0.0;
  off_x = 0.0;
  off_y = 0.0;
  if way == TURN_BY_OUTLINE {
    poly = outline_moved(corners, us, vs, hexset_count(cells));
  } else {
    snap = walls_snapped(corners, row_x, row_y);
    poly = snap.poly;
    // the offset, along headings c + 1 and c + 4, in world units
    off_x = snap.off_x;
    off_y = snap.off_y;
    sx = snap.off_x * co1 - snap.off_y * si1;
    sy = snap.off_x * si1 + snap.off_y * co1;
  }
  out: vector<TurnOffset> = [];
  for i in 0..len(ks) {
    if turn_in_poly(poly, us[i] ?? 0.0, vs[i] ?? 0.0) { out += [TurnOffset { dk: ks[i] ?? 0, dm: ms[i] ?? 0 }]; }
  }
  TurnThirty { offsets: out, sx: sx, sy: sy, c: c, co0: co0, si0: si0, co1: co1, si1: si1, row_x: row_x, row_y: row_y,
               off_x: off_x, off_y: off_y }
}

// ── each wall to the nearest line ───────────────────────────────────────────

// `x` moved to the nearest line midway between rows `row` apart, with the structure moved `off`.
fn to_line(x: float, row: float, off: float) -> float {
  k = floor(((x + off) / row) ?? 0.0);
  k * row + row * 0.5
}

// @FR-HT-Turn30: the structure on its pivot or half a row beside it, per axis
// The structure's offset along one axis, 0 or half a row: the one that moves the walls `lines`
// least to their nearest lines.  A room centred on a row can only be an odd number of rows
// wide, and half a row over an even number.
fn best_offset(lines: const vector<float>, row: float) -> float {
  best = 0.0;
  best_err = -1.0;
  for o in [0.0, row * 0.5] {
    err = 0.0;
    for x in lines { err += fabs(to_line(x, row, o) - o - x); }
    if best_err < 0.0 || err < best_err - 0.000001 {
      best = o;
      best_err = err;
    }
  }
  best
}

// The outline with each wall moved to the nearest line a wall can stand on (the structure moved
// by half a row along an axis where that moves the walls less).
struct TurnSnapped {
  poly: vector<(float, float)>,
  off_x: float,
  off_y: float,
}

// @FR-HT-Near, @FR-HW-Clear: each wall to the nearest line midway between rows
fn walls_snapped(corners: const vector<(float, float)>, row_x: float, row_y: float) -> TurnSnapped {
  n = len(corners);
  xs: vector<float> = [];
  ys: vector<float> = [];
  for i in 0..n {
    (ax, ay) = corners[(i + n - 1) % n ?? 0] ?? (0.0, 0.0);
    (bx, by) = corners[i] ?? (0.0, 0.0);
    if fabs(bx - ax) > fabs(by - ay) { ys += [(ay + by) * 0.5]; } else { xs += [(ax + bx) * 0.5]; }
  }
  off_x = best_offset(xs, row_x);
  off_y = best_offset(ys, row_y);
  TurnSnapped { poly: [for p in corners { (to_line(p.0, row_x, off_x), to_line(p.1, row_y, off_y)) }],
                off_x: off_x, off_y: off_y }
}

// The outline with its walls along each axis moved out or in by one amount (up to half a unit
// each, in sixteenths): the pair that holds as many cells as `want`, or the nearest, moving the
// walls least.
fn outline_moved(corners: const vector<(float, float)>, us: const vector<float>, vs: const vector<float>,
                 want: integer) -> vector<(float, float)> {
  n = len(corners);
  area2 = 0.0;
  for i in 0..n {
    (ax, ay) = corners[(i + n - 1) % n ?? 0] ?? (0.0, 0.0);
    (bx, by) = corners[i] ?? (0.0, 0.0);
    area2 += ax * by - bx * ay;
  }
  out_sign = if area2 > 0.0 { 1.0 } else { -1.0 };
  best_miss = -1;
  best_cost = 0.0;
  best: vector<(float, float)> = [];
  for a in 0..17 {
    for b in 0..17 {
      sx = ((a - 8) as float) * 0.0625 * out_sign;
      sy = ((b - 8) as float) * 0.0625 * out_sign;
      poly: vector<(float, float)> = [];
      for i in 0..n {
        (px, py) = shifted_line(corners, i, sx, sy);
        (qx, qy) = shifted_line(corners, (i + 1) % n ?? 0, sx, sy);
        poly += [if px > -999.0 { (px, qy) } else { (qx, py) }];
      }
      count = 0;
      for c in 0..len(us) { if turn_in_poly(poly, us[c] ?? 0.0, vs[c] ?? 0.0) { count += 1; } }
      miss = count - want;
      if miss < 0 { miss = -miss; }
      cost = fabs(sx) + fabs(sy);
      if best_miss < 0 || miss < best_miss || miss == best_miss && cost < best_cost {
        best_miss = miss;
        best_cost = cost;
        best = poly;
      }
    }
  }
  best
}

// Wall i of the outline moved outward: (x, -1000) for a wall along y (the line x = c), or
// (-1000, y) for a wall along x (the line y = c).  Wall i runs from corner i - 1 to corner i,
// the structure on its left.
fn shifted_line(corners: const vector<(float, float)>, i: integer, sx: float, sy: float) -> (float, float) {
  n = len(corners);
  (ax, ay) = corners[(i + n - 1) % n ?? 0] ?? (0.0, 0.0);
  (bx, by) = corners[i] ?? (0.0, 0.0);
  dx = bx - ax;
  dy = by - ay;
  if fabs(dx) > fabs(dy) {
    y = (ay + by) * 0.5 + (if dx > 0.0 { -sx } else { sx });
    return (-1000.0, y);
  }
  x = (ax + bx) * 0.5 + (if dy > 0.0 { sy } else { -sy });
  (x, -1000.0)
}

// Is (x, y) inside the polygon (crossings of a ray towards +x)?
fn turn_in_poly(poly: const vector<(float, float)>, x: float, y: float) -> boolean {
  n = len(poly);
  inside = false;
  for i in 0..n {
    (x1, y1) = poly[(i + n - 1) % n ?? 0] ?? (0.0, 0.0);
    (x2, y2) = poly[i] ?? (0.0, 0.0);
    if (y1 > y) != (y2 > y) {
      cx = x1 + (y - y1) * (x2 - x1) / (y2 - y1) ?? x1;
      if x < cx { inside = !inside; }
    }
  }
  inside
}

// ── doors and windows through a turn ────────────────────────────────────────
//
// A door or a window is marked on wall edges (`hex_draw::cellopen`).  A 60-degree step turns its
// edges with the cells; an odd step turns its centre with the structure, offset and all, and
// places it again on the turned wall, so it moves a little along the wall and the distances
// between openings change a little.

// A structure with the doors and windows marked on its edges.
pub struct Marked {
  pub mk_cells: HexSet,
  pub mk_marks: EdgeSet,
}

// @FR-HO-Turn60, @FR-HO-Turn30: the openings carried through a turn
// The structure turned `steps` x 30 degrees about cell (pq, pr), its doors and windows with it.
// `hint` (a heading, -1 for none) is the drawn structure's, asked only when its cells show no
// direction.  The walls are solved once for the drawing and once for the turned structure,
// however many openings there are.
pub fn marked_turn(d: Marked, pq: integer, pr: integer, steps: integer, way: integer = TURN_BY_WALLS,
                   hint: integer = -1, sz: OpeningSizes = opening_sizes_default()) -> Marked {
  s = ((steps % 12) + 12) % 12 ?? 0;
  t = cells_turn(d.mk_cells, pq, pr, s, way);
  out = Marked { mk_cells: t.tn_cells, mk_marks: cells_no_marks(t.tn_cells) };
  drawn = cells_built(d.mk_cells, 1, hint);
  ops = cells_openings(drawn.cw_boundary, drawn.cw_runs, drawn.cw_walls, d.mk_marks, sz);
  if s % 2 == 0 {
    // 60-degree steps: every marked edge turned with its two cells
    for o in ops {
      for i in o.op_edges {
        e = drawn.cw_boundary[i]!;
        (ak, am) = lattice_turned(lattice_k(e.we_qa, e.we_ra) - t.tn_pk, lattice_m(e.we_qa, e.we_ra) - t.tn_pm, s / 2 ?? 0);
        (bk, bm) = lattice_turned(lattice_k(e.we_qb, e.we_rb) - t.tn_pk, lattice_m(e.we_qb, e.we_rb) - t.tn_pm, s / 2 ?? 0);
        (qa, ra) = lattice_to_cell(t.tn_pk + ak, t.tn_pm + am);
        (qb, rb) = lattice_to_cell(t.tn_pk + bk, t.tn_pm + bm);
        edge_set_mat(out.mk_marks, qa, ra, qb, rb, o.op_kind);
      }
    }
    return out;
  }
  // an odd step: each centre turned with the structure, offset and all, and placed again on the
  // turned walls
  next_hint = if hint >= 0 { hint + s } else { -1 };
  turned = cells_built(out.mk_cells, 1, next_hint);
  for o in ops {
    (x, y) = turned_point(t, o.op_cx, o.op_cy);
    cells_place_opening(turned, out.mk_marks, o.op_kind, x, y, sz);
  }
  out
}

// ── a stencil through a turn ────────────────────────────────────────────────
//
// A stencil is more than its cells: a height and a label per cell, named layers, and edges — the
// outer walls with their doors and windows, and the INNER walls between its rooms.  Through a
// 60-degree step all of it turns exactly (`hex_field::stencil_rotate`).  Through an odd step:
//
//   the cells         as `cells_turn` turns them: the drawn walls, each to the nearest line;
//   the outer walls   each turned wall takes the material of the drawn wall it is, and the doors
//                     and windows are placed again at their centres (as `marked_turn`);
//   the inner walls   are the boundaries between ROOMS.  A room is the cells an inner wall does
//                     not let you leave; each room is a structure of its own, with straight walls
//                     of its own (`cells_outlines`).  Its outline turns with the stencil and its
//                     walls go to the nearest lines, the same lines the outer walls go to — so an
//                     inner wall comes out straight, meets the outer wall it met, and closes what
//                     it closed.  The turned cells inside a room's turned outline are the room;
//                     an inner wall is wherever two rooms meet.  A wall is the boundary of a
//                     filled region here too, never a line buffered: it cannot leak;
//   what a cell holds a turned cell reads the height, label and layers of the drawn cell nearest
//                     where it came from, in its own room.
//
// What cannot come through is said: a marked edge with the same room on both sides — a wall that
// closes nothing — bounds no room, and is counted in `tu_lost`.

// A stencil turned: the stencil, the turn (for `turned_point`), how many rooms its inner walls
// make, and how many marked edges bounded no room and were not carried (0 for an even step, which
// carries every edge).
pub struct StencilTurn {
  pub tu_stencil: Stencil,
  pub tu_turn:    Turned,
  pub tu_rooms:   integer,
  pub tu_lost:    integer,
}

fn turn_opening(m: integer) -> boolean { m == OPEN_DOOR || m == OPEN_WINDOW }

// The most common of `mats` that is a wall's own (not a door's or a window's); the most common of
// all when there is no other; 0 for none.
fn wall_material(mats: const vector<integer>) -> integer {
  best = 0;
  best_n = 0;
  for pass in 0..2 {
    for m in mats {
      if m == 0 || (pass == 0 && turn_opening(m)) { continue; }
      c = 0;
      for o in mats { if o == m { c += 1; } }
      if c > best_n {
        best = m;
        best_n = c;
      }
    }
    if best_n > 0 { return best; }
  }
  best
}

// A marked edge between two rooms of the drawing: the rooms (the lower first), its material and
// its midpoint in world units.
struct RoomEdge {
  a: integer,
  b: integer,
  mat: integer,
  x: float,
  y: float,
}

// One odd step of a stencil about its cell (0, 0), by walls.
// The parts of the rooms `of` gives the cells: a part is cells of one room joined cell to cell.
// A number per cell, and per part its size and its room.
struct TurnParts {
  part: Labels,
  size: vector<integer>,
  room: vector<integer>,
}

fn room_parts(cells: HexSet, of: Labels) -> TurnParts {
  part = labels_new(hexset_q0(cells), hexset_r0(cells), hexset_w(cells), hexset_h(cells));
  size: vector<integer> = [];
  room: vector<integer> = [];
  for r in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for q in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if !hexset_get(cells, q, r) || label_get(part, q, r) >= 0 { continue; }
      id = len(size);
      mine = label_get(of, q, r);
      label_set(part, q, r, id);
      sq: vector<integer> = [q];
      sr: vector<integer> = [r];
      at = 0;
      while at < len(sq) {
        cq = sq[at] ?? 0;
        cr = sr[at] ?? 0;
        at += 1;
        for d in 0..6 {
          nq = nb_q(cq, cr, d);
          nr = nb_r(cq, cr, d);
          if !hexset_get(cells, nq, nr) || label_get(part, nq, nr) >= 0 || label_get(of, nq, nr) != mine { continue; }
          label_set(part, nq, nr, id);
          sq += [nq];
          sr += [nr];
        }
      }
      size += [len(sq)];
      room += [mine];
    }
  }
  TurnParts { part: part, size: size, room: room }
}

// Make every room one piece.  A room one cell wide comes through a turn of 30 degrees as cells
// that do not touch — along an odd heading a single row's cells stand a step and a half apart —
// and each loose cell would be a room of its own, walled in.  So while a room is in parts, its
// smallest part is joined to the rest of it along the shortest way through other rooms' cells
// (at most three, which take its room: the corridor is widened where it must be), and a part
// with no such way is given to the room it touches most.  Answers how many cells changed room.
fn rooms_whole(cells: HexSet, of: Labels) -> integer {
  changed = 0;
  for _round in 0..256 {
    ps = room_parts(cells, of);
    pick = -1;
    for i in 0..len(ps.size) {
      second = false;
      for j in 0..len(ps.size) {
        if j != i && ps.room[j] == ps.room[i] && ((ps.size[j] ?? 0) > (ps.size[i] ?? 0) || ((ps.size[j] ?? 0) == (ps.size[i] ?? 0) && j < i)) { second = true; }
      }
      if second && (pick < 0 || (ps.size[i] ?? 0) < (ps.size[pick] ?? 0)) { pick = i; }
    }
    if pick < 0 { break; }
    mine = ps.room[pick] ?? -1;
    // the shortest way from the part to another part of its room
    seen = labels_new(hexset_q0(cells), hexset_r0(cells), hexset_w(cells), hexset_h(cells));
    bq: vector<integer> = [];
    br: vector<integer> = [];
    from: vector<integer> = [];
    deep: vector<integer> = [];
    for r in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
      for q in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
        if label_get(ps.part, q, r) != pick { continue; }
        label_set(seen, q, r, 1);
        bq += [q];
        br += [r];
        from += [-1];
        deep += [0];
      }
    }
    at = 0;
    reached = -1;
    while at < len(bq) && reached < 0 {
      cq = bq[at] ?? 0;
      cr = br[at] ?? 0;
      for d in 0..6 {
        nq = nb_q(cq, cr, d);
        nr = nb_r(cq, cr, d);
        if !hexset_get(cells, nq, nr) || label_get(seen, nq, nr) >= 0 { continue; }
        if label_get(of, nq, nr) == mine {
          reached = at;
          break;
        }
        if (deep[at] ?? 0) >= 3 { continue; }
        label_set(seen, nq, nr, 1);
        bq += [nq];
        br += [nr];
        from += [at];
        deep += [(deep[at] ?? 0) + 1];
      }
      at += 1;
    }
    if reached >= 0 {
      // the cells of the way take the room
      while reached >= 0 && (deep[reached] ?? 0) > 0 {
        label_set(of, bq[reached] ?? 0, br[reached] ?? 0, mine);
        changed += 1;
        reached = from[reached] ?? -1;
      }
      continue;
    }
    // no way: the part goes to the room it touches most
    touch: vector<integer> = [];
    count: vector<integer> = [];
    for r in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
      for q in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
        if label_get(ps.part, q, r) != pick { continue; }
        for d in 0..6 {
          nq = nb_q(q, r, d);
          nr = nb_r(q, r, d);
          if !hexset_get(cells, nq, nr) || label_get(ps.part, nq, nr) == pick { continue; }
          o = label_get(of, nq, nr);
          at_o = -1;
          for i in 0..len(touch) { if touch[i] == o { at_o = i; } }
          if at_o < 0 {
            touch += [o];
            count += [1];
          } else { count[at_o] += 1; }
        }
      }
    }
    best = -1;
    for i in 0..len(touch) { if best < 0 || (count[i] ?? 0) > (count[best] ?? 0) { best = i; } }
    if best < 0 { break; }
    for r in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
      for q in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
        if label_get(ps.part, q, r) == pick {
          label_set(of, q, r, touch[best] ?? -1);
          changed += 1;
        }
      }
    }
  }
  changed
}

fn stencil_thirty(st: Stencil, sz: OpeningSizes) -> StencilTurn {
  cells = st.sn_cells;
  has_e = stencil_has_edges(st);
  reach = turn_reach(cells, 0, 0);
  th = turn_thirty(cells, 0, 0, reach, TURN_BY_WALLS);
  win = 2 * reach + 1;
  cells30 = hexset_chunk(0 - reach, 0 - reach, win, win);
  for o in th.offsets {
    (q, r) = lattice_to_cell(o.dk, o.dm);
    hexset_set(cells30, q, r, true);
  }
  t30 = Turned { tn_cells: cells30, tn_pk: 0, tn_pm: 0, tn_steps: 1, tn_sx: th.sx, tn_sy: th.sy };
  // the rooms, and each room's outline in the turned frame, its walls on the lines
  rooms = cells_rooms(cells, if has_e { st.sn_edges } else { edgeset_new(hexset_q0(cells), hexset_r0(cells), hexset_w(cells), hexset_h(cells)) });
  polys: vector<vector<(float, float)>> = [];
  areas: vector<float> = [];
  for ri in 0..rooms.cr_n {
    room = room_cells(cells, rooms, ri);
    poly: vector<(float, float)> = [];
    os = if rooms.cr_n > 1 { cells_outlines(room, th.c) } else { [] };
    if len(os) > 0 {
      for p in os[0]!.co_corners {
        poly += [(to_line(p.0 * th.co0 + p.1 * th.si0, th.row_x, th.off_x), to_line(-p.0 * th.si0 + p.1 * th.co0, th.row_y, th.off_y))];
      }
    }
    area = 0.0;
    for i in 0..len(poly) {
      a = poly[(i + len(poly) - 1) % len(poly) ?? 0] ?? (0.0, 0.0);
      b = poly[i] ?? (0.0, 0.0);
      area += a.0 * b.1 - b.0 * a.1;
    }
    polys += [poly];
    areas += [fabs(area)];
  }
  // each turned cell's room: the smallest room whose turned outline holds it; outside them all,
  // the room of the drawn cell nearest where it came from.  Then every room made one piece.
  hz = heights_new(0 - reach, 0 - reach, win, win);
  lb = labels_new(0 - reach, 0 - reach, win, win);
  ly = layers_new(0 - reach, 0 - reach, win, win);
  for li in 0..layer_count(st.sn_layers) { layer_add(ly, layer_name(st.sn_layers, li)); }
  room30 = labels_new(0 - reach, 0 - reach, win, win);
  for pass in 0..2 {
    if pass == 1 && rooms.cr_n > 1 { rooms_whole(cells30, room30); }
    for o in th.offsets {
      (q, r) = lattice_to_cell(o.dk, o.dm);
      dx = (o.dk as float) * 0.8660254037844386;
      dy = (o.dm as float) * 0.5;
      u = dx * th.co1 + dy * th.si1;
      v = -dx * th.si1 + dy * th.co1;
      mine = if pass == 1 { label_get(room30, q, r) } else { -1 };
      if pass == 0 {
        for ri in 0..rooms.cr_n {
          if len(polys[ri]!) < 3 || !turn_in_poly(polys[ri]!, u, v) { continue; }
          if mine < 0 || (areas[ri] ?? 0.0) < (areas[mine] ?? 0.0) { mine = ri; }
        }
      }
      // where it came from in the drawing: the frame's coordinates less the offset, in the
      // drawing's own frame
      fx = u - th.off_x;
      fy = v - th.off_y;
      wx = fx * th.co0 - fy * th.si0;
      wy = fx * th.si0 + fy * th.co0;
      (hq, hr) = hex_at(wx, wy);
      // the drawn cell nearest that: of its room among the cells round it, of its room anywhere,
      // then any
      sq = 0;
      sr = 0;
      found = false;
      best = 0.0;
      for way in 0..3 {
        if found { break; }
        if way == 1 {
          if pass == 0 || mine < 0 { continue; }
          for cr in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
            for cq in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
              if label_get(rooms.cr_of, cq, cr) != mine { continue; }
              ex = cell_x(cq, cr) - wx;
              ey = cell_y(cq, cr) - wy;
              dist = ex * ex + ey * ey;
              if !found || dist < best {
                found = true;
                best = dist;
                sq = cq;
                sr = cr;
              }
            }
          }
          continue;
        }
        for d in -1..6 {
          cq = if d < 0 { hq } else { nb_q(hq, hr, d) };
          cr = if d < 0 { hr } else { nb_r(hq, hr, d) };
          if !hexset_get(cells, cq, cr) { continue; }
          if way == 0 && mine >= 0 && label_get(rooms.cr_of, cq, cr) != mine { continue; }
          ex = cell_x(cq, cr) - wx;
          ey = cell_y(cq, cr) - wy;
          dist = ex * ex + ey * ey;
          if !found || dist < best {
            found = true;
            best = dist;
            sq = cq;
            sr = cr;
          }
        }
      }
      if pass == 0 {
        if mine < 0 && found { mine = label_get(rooms.cr_of, sq, sr); }
        label_set(room30, q, r, mine);
        continue;
      }
      if !found { continue; }
      if stencil_has_heights(st) { height_set(hz, q, r, height_get(st.sn_heights, sq, sr)); }
      if stencil_has_labels(st) { label_set(lb, q, r, label_get(st.sn_labels, sq, sr)); }
      for li in 0..layer_count(st.sn_layers) { layer_set_at(ly, li, q, r, layer_get_at(st.sn_layers, li, sq, sr)); }
    }
  }
  lost = 0;
  edges30 = edgeset_new(0 - reach, 0 - reach, win, win);
  if has_e {
    // the outer walls: each turned wall the material of the drawn wall it is
    drawn = cells_built(cells, 1, th.c);
    turned = cells_built(cells30, 1, th.c + 1);
    for j in 0..len(turned.cw_walls) {
      wj = turned.cw_walls[j]!;
      mate = -1;
      near = 0.0;
      for i in 0..len(drawn.cw_walls) {
        wi = drawn.cw_walls[i]!;
        if head_norm(surface_heading(wi) + 1) != surface_heading(wj) { continue; }
        (tx, ty) = turned_point(t30, surface_mean_x(wi), surface_mean_y(wi));
        ex = tx - surface_mean_x(wj);
        ey = ty - surface_mean_y(wj);
        if mate < 0 || ex * ex + ey * ey < near {
          mate = i;
          near = ex * ex + ey * ey;
        }
      }
      if mate < 0 { continue; }
      mats: vector<integer> = [];
      for i in drawn.cw_runs[mate] ?? [] {
        e = drawn.cw_boundary[i]!;
        mats += [edge_mat(st.sn_edges, e.we_qa, e.we_ra, e.we_qb, e.we_rb)];
      }
      mat = wall_material(mats);
      if mat == 0 { continue; }
      for i in turned.cw_runs[j] ?? [] {
        e = turned.cw_boundary[i]!;
        edge_set_mat(edges30, e.we_qa, e.we_ra, e.we_qb, e.we_rb, mat);
      }
    }
    for o in cells_openings(drawn.cw_boundary, drawn.cw_runs, drawn.cw_walls, st.sn_edges, sz) {
      (x, y) = turned_point(t30, o.op_cx, o.op_cy);
      cells_place_opening(turned, edges30, o.op_kind, x, y, sz);
    }
    // the inner walls of the drawing, by the two rooms each lies between
    between: vector<RoomEdge> = [];
    for r in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
      for q in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
        if !hexset_get(cells, q, r) { continue; }
        for d in 0..6 {
          nq = nb_q(q, r, d);
          nr = nb_r(q, r, d);
          // each edge once: from the cell that comes first, row by row
          if !hexset_get(cells, nq, nr) || nr < r || (nr == r && nq < q) { continue; }
          m = edge_mat(st.sn_edges, q, r, nq, nr);
          if m == 0 { continue; }
          ra = label_get(rooms.cr_of, q, r);
          rb = label_get(rooms.cr_of, nq, nr);
          if ra == rb {
            lost += 1;
            continue;
          }
          between += [RoomEdge { a: if ra < rb { ra } else { rb }, b: if ra < rb { rb } else { ra }, mat: m,
                                 x: (cell_x(q, r) + cell_x(nq, nr)) * 0.5, y: (cell_y(q, r) + cell_y(nq, nr)) * 0.5 }];
        }
      }
    }
    every: vector<integer> = [for e in between { e.mat }];
    // the turned inner walls: wherever two rooms meet
    inner: vector<RoomEdge> = [];
    iq: vector<integer> = [];
    ir: vector<integer> = [];
    jq: vector<integer> = [];
    jr: vector<integer> = [];
    for o in th.offsets {
      (q, r) = lattice_to_cell(o.dk, o.dm);
      for d in 0..6 {
        nq = nb_q(q, r, d);
        nr = nb_r(q, r, d);
        if !hexset_get(cells30, nq, nr) || nr < r || (nr == r && nq < q) { continue; }
        ra = label_get(room30, q, r);
        rb = label_get(room30, nq, nr);
        if ra == rb { continue; }
        lo = if ra < rb { ra } else { rb };
        hi = if ra < rb { rb } else { ra };
        mats: vector<integer> = [];
        for e in between { if e.a == lo && e.b == hi { mats += [e.mat]; } }
        mat = wall_material(if len(mats) > 0 { mats } else { every });
        if mat == 0 { continue; }
        edge_set_mat(edges30, q, r, nq, nr, mat);
        inner += [RoomEdge { a: lo, b: hi, mat: mat, x: (cell_x(q, r) + cell_x(nq, nr)) * 0.5, y: (cell_y(q, r) + cell_y(nq, nr)) * 0.5 }];
        iq += [q];
        ir += [r];
        jq += [nq];
        jr += [nr];
      }
    }
    // the doors and windows in them: each stretch of one kind between two rooms, its centre
    // carried through the turn, on the edges of the turned wall within half its width
    taken: vector<boolean> = [for _ in 0..len(between) { false }];
    for bi in 0..len(between) {
      first = between[bi]!;
      if (taken[bi] ?? true) || !turn_opening(first.mat) { continue; }
      // the stretch: every edge of that kind between those rooms reached edge by edge
      group: vector<integer> = [bi];
      taken[bi] = true;
      at = 0;
      while at < len(group) {
        g = between[group[at] ?? 0]!;
        at += 1;
        for oi in 0..len(between) {
          e = between[oi]!;
          if (taken[oi] ?? true) || e.a != first.a || e.b != first.b || e.mat != first.mat { continue; }
          if (e.x - g.x) * (e.x - g.x) + (e.y - g.y) * (e.y - g.y) > 1.1 { continue; }
          taken[oi] = true;
          group += [oi];
        }
      }
      cx = 0.0;
      cy = 0.0;
      for gi in group {
        cx += between[gi]!.x;
        cy += between[gi]!.y;
      }
      (x, y) = turned_point(t30, (cx / (len(group) as float)) ?? 0.0, (cy / (len(group) as float)) ?? 0.0);
      half = opening_size(first.mat, sz) * 0.5;
      nearest = -1;
      near = 0.0;
      put = 0;
      for ii in 0..len(inner) {
        e = inner[ii]!;
        if e.a != first.a || e.b != first.b { continue; }
        dist = sqrt((e.x - x) * (e.x - x) + (e.y - y) * (e.y - y)) ?? 0.0;
        if dist <= half + 0.000000001 {
          edge_set_mat(edges30, iq[ii] ?? 0, ir[ii] ?? 0, jq[ii] ?? 0, jr[ii] ?? 0, first.mat);
          put += 1;
        }
        if nearest < 0 || dist < near {
          nearest = ii;
          near = dist;
        }
      }
      if put == 0 && nearest >= 0 {
        edge_set_mat(edges30, iq[nearest] ?? 0, ir[nearest] ?? 0, jq[nearest] ?? 0, jr[nearest] ?? 0, first.mat);
      }
    }
  }
  out = if has_e { stencil_with_edges(cells30, hz, lb, edges30, stencil_has_heights(st), stencil_has_labels(st)) }
        else { stencil_from(cells30, hz, lb, stencil_has_heights(st), stencil_has_labels(st)) };
  out.sn_layers = ly;
  StencilTurn { tu_stencil: out, tu_turn: t30, tu_rooms: rooms.cr_n, tu_lost: lost }
}

// @FR-HT-Pivot, @FR-HT-Turn60, @FR-HT-Turn30: a stencil turned whole
// Stencil `st` turned `steps` x 30 degrees anticlockwise about its own cell (0, 0), with all it
// carries.  An even step is `hex_field::stencil_rotate`, exact.  An odd step is one step of 30
// degrees — the cells by their drawn walls, the outer walls with their materials and openings,
// the inner walls as the straight boundaries of its rooms, what each cell holds read from the
// drawing — and then the 60-degree steps that remain, exact.  `sz` is the width a door and a
// window keep.  The window of the result is fitted to its cells.
pub fn stencil_turn(st: Stencil, steps: integer, sz: OpeningSizes = opening_sizes_default()) -> StencilTurn {
  s = ((steps % 12) + 12) % 12 ?? 0;
  whole = cells_turn(st.sn_cells, 0, 0, s);
  if s % 2 == 0 {
    return StencilTurn { tu_stencil: stencil_rotate(st, s / 2 ?? 0), tu_turn: whole,
                         tu_rooms: cells_rooms(st.sn_cells, st.sn_edges).cr_n, tu_lost: 0 };
  }
  one = stencil_thirty(st, sz);
  StencilTurn { tu_stencil: stencil_rotate(one.tu_stencil, s / 2 ?? 0), tu_turn: whole, tu_rooms: one.tu_rooms, tu_lost: one.tu_lost }
}

// hexcombine — TWO STENCILS ADJACENT: WHO OWNS THE SHARED EDGE? (rung A8, the combination rung).
//
// Where things that work ALONE stop working TOGETHER.  The mechanism is crawler's `cut_arb`,
// float-free.  The design in one line:
//
//   "MARK EVERY PART, THEN CUT ONCE."
//
// Do NOT draw each stencil's walls and overlay them.  Union the footprints first, then cut the
// boundary of the UNION once, tagging each boundary edge with the material of the source stencil
// its cell came from.  Two consequences, both the point of the rung:
//
//   ORDER-FREE BY CONSTRUCTION.  The result reads only the finished union and the fixed source
//   map — never which stencil was stamped when — so `combine_cut(a, b) == combine_cut(b, a)`.  The
//   naive per-body overlay is last-writer, hence order-DEPENDENT: it works for one stencil and
//   breaks for two.  (This is `X4` / law K₂: same-level overlap arbitrates deterministically and
//   order-freely.)
//
//   THE SHARED EDGE FUSES.  Between two adjacent stencils the shared edge is INTERIOR to the union,
//   so cutting the union's boundary never marks it — nobody owns it, and there is no seam wall.
//   Two adjacent stencils become one fabric (crawler's FORMS.md, DESIGN §8.0.1: "seated fabric may
//   merge").  Which two parts were authored is not recoverable from the fabric, and by law P1 need
//   not be — `rebuild` returns the canonical representative.
//
// Exact and float-free: the "nearest source" of a boundary edge is simply its own bordering cell's
// stencil, so no distance and no float enter.  This is the κ ≤ 2 case; N stencils are the same
// construction (union all, cut once, nearest source).  Different LEVELS never contend at all
// (crawler's bridge guarantee) — that axis waits on the layer machinery and is not gated here.
use hex_field::*;
use hex_form::*;

// The union footprint of two stencils sharing `a`'s chunk — a fresh HexSet, `a ∪ b`.
pub fn field_union(a: HexSet, b: HexSet) -> HexSet {
  u = hexset_chunk(hexset_q0(a), hexset_r0(a), hexset_w(a), hexset_h(a));
  for r in hexset_r0(a)..hexset_r0(a) + hexset_h(a) {
    for q in hexset_q0(a)..hexset_q0(a) + hexset_w(a) {
      if hexset_get(a, q, r) || hexset_get(b, q, r) { hexset_set(u, q, r, true); }
    }
  }
  u
}

// Combine two same-level stencils: union the footprints, cut the boundary of the union ONCE, and
// tag each boundary edge with the material of the source stencil its bordering cell came from.
// Order-free by construction (reads the finished union + source map, never stamp order); the shared
// interior edge is never cut, so adjacent stencils fuse.  Returns the number of edges cut.
//
// The tie-break is INTRINSIC, not positional: a cell in BOTH stencils (an overlap) takes the LOWER
// material id, so `combine_cut(a,b) == combine_cut(b,a)` for overlapping stencils too, not only
// disjoint ones — otherwise "order-free" would quietly hold only for the adjacency case gated here.
// (crawler's full `cut_arb` breaks the same tie by nearest SURFACE then lower id; with no surfaces
// in play the lower id is the whole rule.  Overlap itself is a separate, ungated A8 axis.)
//
// Example: @HXP-001
// Example: @HXP-002
pub fn combine_cut(a: HexSet, ma: integer, b: HexSet, mb: integer, e: EdgeSet) -> integer {
  u = field_union(a, b);
  n = 0;
  for r in hexset_r0(u)..hexset_r0(u) + hexset_h(u) {
    for q in hexset_q0(u)..hexset_q0(u) + hexset_w(u) {
      if hexset_get(u, q, r) {
        for d in 0..6 {
          nq = nb_q(q, r, d);
          nr = nb_r(q, r, d);
          if !hexset_get(u, nq, nr) {
            ina = hexset_get(a, q, r);
            inb = hexset_get(b, q, r);
            m = ma;
            if inb && !ina { m = mb; }
            if ina && inb && mb < ma { m = mb; }   // overlap: lower id, order-free
            edge_set_mat(e, q, r, nq, nr, m);
            n = n + 1;
          }
        }
      }
    }
  }
  n
}

// ── LEVELS — THE BRIDGE GUARANTEE ──────────────────────────────────────────
//
// A **level** is the topological *sheet* a thing sits on — exactly OSM's `layer` tag, and the
// `layer*` axis the foxel already has at the top of its schema (`ROUNDTRIP` §2.4; moros's `cy`).
// crawler settled what it buys (`bridgetest.loft`), and the sharpest statement of it is a contrast:
//
//   > the SAME two ways, at the same level, are a level crossing; at different levels, a bridge.
//   > Nothing about the ways changes — only the sheet.
//
// **A LEVEL IS NOT A HEIGHT.** The actual `z` comes from the surface/feature interval; the level is
// a discrete sheet index and nothing else. Two things can share a `z` on different levels, or sit
// at different `z` on the same one.
//
// The load-bearing consequence, and why this is not just bookkeeping: **different levels never
// contend.** So the level is a **filter applied BEFORE the cut**, not an arbitration rule after it —
// it does work the contention degree `κ` would otherwise have to do (`X4`, and `DESIGN` §6.1). Most
// of a world exists only at level 0, where this costs exactly nothing.

// Combine two stencils **as seen at level `at`**: only those whose own level matches take part, and
// the survivors are then cut exactly as `combine_cut` does.  Two stencils on different levels can
// therefore never fuse, never arbitrate and never contend — at either level, one of them simply is
// not there.  Returns the number of edges cut at this level.
//
// Example: @HXP-003
pub fn combine_cut_level(a: HexSet, la: integer, ma: integer,
                         b: HexSet, lb: integer, mb: integer,
                         at: integer, e: EdgeSet) -> integer {
  ina = la == at;
  inb = lb == at;
  u = hexset_chunk(hexset_q0(a), hexset_r0(a), hexset_w(a), hexset_h(a));
  for r in hexset_r0(a)..hexset_r0(a) + hexset_h(a) {
    for q in hexset_q0(a)..hexset_q0(a) + hexset_w(a) {
      if (ina && hexset_get(a, q, r)) || (inb && hexset_get(b, q, r)) {
        hexset_set(u, q, r, true);
      }
    }
  }
  n = 0;
  for r2 in hexset_r0(u)..hexset_r0(u) + hexset_h(u) {
    for q2 in hexset_q0(u)..hexset_q0(u) + hexset_w(u) {
      if hexset_get(u, q2, r2) {
        for d in 0..6 {
          nq = nb_q(q2, r2, d);
          nr = nb_r(q2, r2, d);
          if !hexset_get(u, nq, nr) {
            inaa = ina && hexset_get(a, q2, r2);
            inbb = inb && hexset_get(b, q2, r2);
            m = ma;
            if inbb && !inaa { m = mb; }
            if inaa && inbb && mb < ma { m = mb; }   // overlap: lower id, order-free (X52)
            edge_set_mat(e, q2, r2, nq, nr, m);
            n = n + 1;
          }
        }
      }
    }
  }
  n
}

// The contention degree at one cell, counted **per level**: how many of the two stencils claim it
// *at level `at`*.  This is `κ` (`X53`) made level-aware — two stencils that overlap in `(q,r)` but
// sit on different sheets give `κ = 1` at each level, never `κ = 2` at either.
//
// Example: @HXP-003
pub fn kappa_at_level(a: HexSet, la: integer, b: HexSet, lb: integer,
                      at: integer, q: integer, r: integer) -> integer {
  k = 0;
  if la == at && hexset_get(a, q, r) { k = k + 1; }
  if lb == at && hexset_get(b, q, r) { k = k + 1; }
  k
}

// How many marked edges separate an `a`-cell from a `b`-cell — the "shared wall" run.  Zero after
// `combine_cut` (the seam fuses); non-zero after a per-body overlay (a spurious seam wall).  This
// is the measurable form of "who owns the shared edge?": nobody, if the parts were combined right.
//
// Example: @HXP-001
pub fn shared_marked(a: HexSet, b: HexSet, e: EdgeSet) -> integer {
  n = 0;
  for r in hexset_r0(a)..hexset_r0(a) + hexset_h(a) {
    for q in hexset_q0(a)..hexset_q0(a) + hexset_w(a) {
      if hexset_get(a, q, r) {
        for d in 0..6 {
          nq = nb_q(q, r, d);
          nr = nb_r(q, r, d);
          if hexset_get(b, nq, nr) && edge_mat(e, q, r, nq, nr) != 0 { n = n + 1; }
        }
      }
    }
  }
  n
}

// SPDX-License-Identifier: LGPL-3.0-or-later
// hexframe — THE FRAME SEAM: a POSED body against the world (DESIGN §6; laws K₁, K₂).
//
// Collision reads several presentations at once — the world (identity frame) plus every posed body.
// Inside a frame geometry is exact; ACROSS frames it cannot be, because a pose is continuous.  The
// whole discipline of this module is to make the error live in exactly ONE place:
//
//   THE POSE TRANSFORM IS THE SOLE FLOATING-POINT STEP.  Every cross-frame query routes through it
//   once — world → p⁻¹ → exact local test → (p) → world — so ε > 0 is permitted ONLY on the seam
//   (`ε_seam`), and is a DETERMINISTIC function of its inputs (`L7`/`I9`: byte-identical replay).
//   Inside a frame the error is exactly 0.  (crawler's chunk seam is a different thing, and exact.)
//
// The FORBIDDEN fix is closing a crack by moving geometry — snapping a body's wall onto the world
// lattice.  That relocates the machine-ε seam error into a frame INTERIOR, where it becomes a real
// displacement, and voids law D for that body.  `tests/seam.loft` §3 measures both.
//
// This is the one place `hexframe` is float, on purpose.  Everything the pose acts on is otherwise
// integer/exact (X1–X52), so `ε_seam` is the entire error budget of the stack.
use hex_form::*;

// A pose: a rotation (given as its cos/sin, so a Pythagorean angle can be represented EXACTLY) then
// a translation.  Stored as cos/sin rather than θ so a gate can build a rational rotation and check
// the float pipeline against an exact integer oracle.
pub struct Pose {
  ps_cs: float,
  ps_sn: float,
  ps_tx: float,
  ps_ty: float,
}

// A placement: a rotation given as cos/sin and a translation.  Storing the pair rather than
// an angle keeps `pose_fwd` and `pose_inv` exact inverses with no trig at use time.
pub fn pose_new(cs: float, sn: float, tx: float, ty: float) -> Pose {
  Pose { ps_cs: cs, ps_sn: sn, ps_tx: tx, ps_ty: ty }
}

// world → local.  THE sole float step (with `pose_fwd_*`).  A cross-frame query calls this once and
// then tests in the body's own exact frame.
pub fn pose_inv_x(p: Pose, wx: float, wy: float) -> float {
  p.ps_cs * (wx - p.ps_tx) + p.ps_sn * (wy - p.ps_ty)
}
// A world point's y in the pose's frame — the companion of `pose_inv_x`.
pub fn pose_inv_y(p: Pose, wx: float, wy: float) -> float {
  0.0 - p.ps_sn * (wx - p.ps_tx) + p.ps_cs * (wy - p.ps_ty)
}

// local → world.
pub fn pose_fwd_x(p: Pose, lx: float, ly: float) -> float {
  p.ps_cs * lx - p.ps_sn * ly + p.ps_tx
}
// A local point's y in the world — the companion of `pose_fwd_x`, and the inverse of
// `pose_inv_y`.
pub fn pose_fwd_y(p: Pose, lx: float, ly: float) -> float {
  p.ps_sn * lx + p.ps_cs * ly + p.ps_ty
}

// The round-trip residual |p(p⁻¹(W)) − W|∞ at one world point — the transform's numerical error,
// which is what `ε_seam` bounds.  Exactly 0 in exact arithmetic; ~machine ε in float.
//
// AT ONE WORLD POINT is the load-bearing half: this is a jagged, deterministic function of the
// point, not a constant.  Near the frame origin it commonly reads exactly 0, and reading it there
// and adopting that as ε_seam is the mistake the number exists to prevent.
//
// Example: @HXP-005
pub fn pose_residual(p: Pose, wx: float, wy: float) -> float {
  lx = pose_inv_x(p, wx, wy);
  ly = pose_inv_y(p, wx, wy);
  bx = pose_fwd_x(p, lx, ly);
  by = pose_fwd_y(p, lx, ly);
  ex = bx - wx; if ex < 0.0 { ex = 0.0 - ex; }
  ey = by - wy; if ey < 0.0 { ey = 0.0 - ey; }
  if ey > ex { return ey; }
  ex
}

// Is world point (wx,wy) inside the disk body posed by `p`?  The body is a disk of squared-radius
// `rr` about its LOCAL centre (cx,cy) — a posed tower (its radius is a shell, `X49`).  Routes through
// the transform, then the exact local test |l − c|² ≤ rr.  This is the general body test in
// miniature: transform once, then decide in the body's own frame.
pub fn disk_hit(p: Pose, cx: float, cy: float, rr: float, wx: float, wy: float) -> boolean {
  lx = pose_inv_x(p, wx, wy) - cx;
  ly = pose_inv_y(p, wx, wy) - cy;
  lx * lx + ly * ly <= rr
}

// ── ARBITRATION WHERE FRAMES DISAGREE (law K₂) ─────────────────────────────
//
// Two requirements the design flags as easy to omit silently: a TOTAL ORDER on frames (a
// "whichever is nearer" tie broken by iteration order breaks replay), and a FAIL-SAFE DIRECTION —
// prefer *solid* over *gap*, so a crack yields a spurious contact rather than a body falling through
// the world (composes with `I4`).  Both are captured below and are order-free by construction.

// The fail-safe verdict at a point claimed by two frames: solid if EITHER is solid.  Order-free
// (symmetric) and conservative — a gap only where both agree it is a gap.
//
// Example: @HXP-006
pub fn arb_solid(a_solid: boolean, b_solid: boolean) -> boolean { a_solid || b_solid }

// Which frame OWNS a contested point — its material/surface wins.  The lowest id among the frames
// claiming *solid* (a total order on frames), or -1 if none is solid.  Intrinsic, so it does not
// depend on the order the frames are visited.
//
// Example: @HXP-006
pub fn arb_owner(a_solid: boolean, a_id: integer, b_solid: boolean, b_id: integer) -> integer {
  if a_solid && b_solid {
    if a_id < b_id { return a_id; }
    return b_id;
  }
  if a_solid { return a_id; }
  if b_solid { return b_id; }
  -1
}

// Copyright (c) 2026 Jurjen Stellingwerff
// SPDX-License-Identifier: LGPL-3.0-or-later
// hexseat — SEATING A STENCIL ON TERRAIN.
//
// Terrain is stored in the `height` slot, the same slot as roofs.  Seating is the question of
// where a stencil goes on it: a stencil is authored flat in its own local frame, and the ground
// it is placed on is not flat.  The answer is *"a building … **seated on terrain**, residual
// flagged"* (`SPEC` **G5**).
//
// THE CONSTRUCTION, AND WHY IT COSTS THE ROUND TRIP NOTHING:
//
//   **Seating writes the `height` slot.  It does not touch cells or edges.**
//
// So a stencil's footprint is byte-identical on flat ground and on a slope, and `rebuild` — which
// reads the CELLS — recovers the same text either way.  Terrain and stencil occupy **orthogonal
// slots** in the foxel, so they compose without interacting at all.
//
// The same shape of answer recurs across the family: a **door** is a
// material, not a hole (`X51`); a **level** is a filter before the cut, not an arbitration after it
// (`X58`); **terrain** is a height, not a change of footprint.  Each time the round trip survives
// because the new thing lands in a slot the recovery does not read.
//
// WHAT THE SLOPE DOES COST is a **residual**: a stencil seated at ONE height cannot match ground
// that varies under it.  That residual is real, it is a function of the seat policy, and by `G5` it
// must be **returned and flagged — never silently absorbed**.  Which height to seat at (lowest,
// mean, highest) is a **policy**; it changes the residual and changes nothing about recoverability —
// the same split `A6` found between `Sep` (recoverability) and `X7` (a choice policy).
use hex_field::*;
use hex_form::*;

pub const SEAT_LOW  = 0;    // seat at the lowest ground under the footprint — the building perches
pub const SEAT_MEAN = 1;    // seat at the mean — the residual is smallest, split above and below
pub const SEAT_HIGH = 2;    // seat at the highest — the building is buried into the slope

// The height to seat at, under `policy`.  A pure read of the terrain under the footprint.
//
// Example: @HXP-004
pub fn seat_height(cells: HexSet, terr: Heights, policy: integer) -> float {
  lo = 0.0;
  hi = 0.0;
  sum = 0.0;
  n = 0;
  for r in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for q in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if hexset_get(cells, q, r) {
        z = height_get(terr, q, r);
        if n == 0 { lo = z; hi = z; }
        if z < lo { lo = z; }
        if z > hi { hi = z; }
        sum = sum + z;
        n = n + 1;
      }
    }
  }
  if n == 0 { return 0.0; }
  if policy == SEAT_LOW { return lo; }
  if policy == SEAT_HIGH { return hi; }
  (sum / (n as float)) ?? 0.0
}

// THE RESIDUAL, and it is returned rather than swallowed (`G5`): the largest gap between the seat
// height and the ground under any footprint cell.  Zero on flat ground; on a slope it is the
// cut-and-fill the placement owes, and the editor must show it.
//
// Example: @HXP-004
pub fn seat_residual(cells: HexSet, terr: Heights, z: float) -> float {
  worst = 0.0;
  for r in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for q in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if hexset_get(cells, q, r) {
        d = height_get(terr, q, r) - z;
        if d < 0.0 { d = 0.0 - d; }
        if d > worst { worst = d; }
      }
    }
  }
  worst
}

// SEAT THE STENCIL: write the seat height into the `height` slot for every footprint cell.  Cells
// and edges are untouched — that is the whole point, and it is what keeps `rebuild` exact.  Returns
// the number of cells whose height was written.
//
// Example: @HXP-004
pub fn seat_write(cells: HexSet, terr: Heights, z: float) -> integer {
  n = 0;
  for r in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for q in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if hexset_get(cells, q, r) {
        height_set(terr, q, r, z);
        n = n + 1;
      }
    }
  }
  n
}

// ── SEATED, AND THE GROUND ROUND IT BLENDED BACK ───────────────────────────
//
// `seat_write` levels the footprint and leaves a step all round it: right for a stencil whose
// own walls go down to the ground, wrong for a house on a hillside, which stands on a levelled
// plot the hill runs smoothly into.  `seat_blend` writes the footprint as `seat_write` does and
// then the ground round it: from the seat height at the footprint's edge back to the ground's
// own height `blend` away, along a smoothstep, so there is neither a step nor a crease.
//
// It still touches the `height` slot alone — the footprint's cells and edges are as they were —
// and it still returns what the slope cost: the LOWEST the ground was under the footprint
// (`seat_height(.., SEAT_LOW)`), which is where the structure's walls have to go down to.
//
// Distances are in the caller's unit: `step` is the distance between two neighbouring cell
// centres in it (sqrt(3) in world units; 1.5 where a hex step is 1.5 m and heights are metres).
// The footprint's edge is taken half a `step` out from the nearest footprint centre.
//
// From tile-world-game (rule HG-Seat of its `formal/hex-houses.md`).

fn seat_smooth(t: float) -> float {
  c = if t < 0.0 { 0.0 } else if t > 1.0 { 1.0 } else { t };
  c * c * (3.0 - 2.0 * c)
}

// @FR-HG-Seat: a structure's footprint level, the ground round it blended back
// Seat `cells` at height `z` on `terr`, the ground round them blended from `z` back to its own
// height over `blend` from the footprint's edge.  Answers the lowest the ground was under the
// footprint (`z` itself for an empty footprint, where nothing is written).
pub fn seat_blend(cells: HexSet, terr: Heights, z: float, blend: float, step: float) -> float {
  kx = step * 0.5;
  my = step * 0.28867513459481287;
  fx: vector<float> = [];
  fy: vector<float> = [];
  for r in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for q in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if hexset_get(cells, q, r) {
        fx += [(lattice_k(q, r) as float) * kx];
        fy += [(lattice_m(q, r) as float) * my];
      }
    }
  }
  if len(fx) == 0 { return z; }
  low = seat_height(cells, terr, SEAT_LOW);
  seat_write(cells, terr, z);
  // the window round the footprint the blend reaches: rows lie 3 * my apart
  pad = (ceil((blend / (3.0 * my)) ?? 0.0) as integer) + 2;
  for r in hexset_r0(cells) - pad..hexset_r0(cells) + hexset_h(cells) + pad {
    for q in hexset_q0(cells) - pad..hexset_q0(cells) + hexset_w(cells) + pad {
      if hexset_get(cells, q, r) { continue; }
      x = (lattice_k(q, r) as float) * kx;
      y = (lattice_m(q, r) as float) * my;
      near = 1000000.0;
      for i in 0..len(fx) {
        dx = (fx[i] ?? 0.0) - x;
        dy = (fy[i] ?? 0.0) - y;
        fd = dx * dx + dy * dy;
        if fd < near { near = fd; }
      }
      d = (sqrt(near) ?? 0.0) - kx;
      if d >= blend { continue; }
      t = seat_smooth((d / blend) ?? 1.0);
      own = height_get(terr, q, r);
      height_set(terr, q, r, z + (own - z) * t);
    }
  }
  low
}

// Copyright (c) 2026 Jurjen Stellingwerff
// SPDX-License-Identifier: LGPL-3.0-or-later
// stenciltext — A STENCIL DRAWN AS TEXT: a place template you can read and write by hand.
//
// A template of hex cells — a room, a tower, a house's floor — is a `hex_field::Stencil`: cells,
// a label a cell, and marks on the edges between cells (walls, doors, windows).  This is its
// text: lines of characters, each character a cell or an edge, laid out as the cells lie.
//
//       #   #
//     # a : b #          a cell every four columns, the rows shifted as hex rows are;
//      # D   #           between two cells of a row, the edge between them;
//       # c #            on the line between two rows, the edges between a cell and the two
//        # #             cells it touches in the other row.
//
// Lines alternate: a line of cells (and the edges between neighbours in that row), then a line of
// the edges between that row and the next.  North is up.  A space is nothing: no cell, no mark.
// An edge may be marked between a cell and no cell — the template's outer wall, a door in it.
//
// What a character means is the caller's (`StencilLegend`): each cell character a label, each
// edge character a material.  One position is given as cell (0, 0) — the cell the template is
// turned about (`stencil_turn`) and placed by.
//
// `stencil_read` answers what it could not read, a line each, instead of guessing.
// `stencil_rows` writes a stencil back; what it writes, `stencil_read` reads as the same stencil.
use hex_field::*;

// What the characters mean: `sl_cells`' i-th character is a cell of label `sl_labels[i]`,
// `sl_edges`' i-th an edge of material `sl_mats[i]` (several characters may give one material,
// so a wall can be drawn `|`, `/` and `\`; written back, a material is its first character).
pub struct StencilLegend {
  pub sl_cells:  text,
  pub sl_labels: vector<integer>,
  pub sl_edges:  text,
  pub sl_mats:   vector<integer>,
}

// A stencil read from text, and what could not be read (none: all of it was).
pub struct StencilRead {
  pub sr_stencil: Stencil,
  pub sr_errors:  vector<text>,
}

// A stencil as text: its lines, top first, and the position (line, column) of cell (0, 0).
pub struct StencilRows {
  pub sw_rows: vector<text>,
  pub sw_line: integer,
  pub sw_col:  integer,
}

// What a position of the text is: nothing (kind 0), cell a (1), or the edge between a and b (2).
struct TextSpot {
  kind: integer,
  qa: integer,
  ra: integer,
  qb: integer,
  rb: integer,
}

// The position `up` lines above and `right` columns right of cell (0, 0).  Two lines are a row;
// four columns are a cell, two a lattice step (`lattice_k`).
fn text_spot(up: integer, right: integer) -> TextSpot {
  none = TextSpot { kind: 0, qa: 0, ra: 0, qb: 0, rb: 0 };
  if (up & 1) == 0 {
    // a line of cells
    if (right & 1) != 0 { return none; }
    r = up / 2 ?? 0;
    k = right / 2 ?? 0;
    if ((k - r) & 1) == 0 {
      (q, _r) = lattice_to_cell(k, 3 * r);
      return TextSpot { kind: 1, qa: q, ra: r, qb: q, rb: r };
    }
    (qa, _ra) = lattice_to_cell(k - 1, 3 * r);
    (qb, _rb) = lattice_to_cell(k + 1, 3 * r);
    return TextSpot { kind: 2, qa: qa, ra: r, qb: qb, rb: r };
  }
  // a line of edges, between row `lo` and the row above it
  if (right & 1) == 0 { return none; }
  lo = (up - 1) / 2 ?? 0;
  a = (right - 1) / 2 ?? 0;
  // of the two lattice columns either side, the one of the lower row's parity is its cell's
  kl = if ((a - lo) & 1) == 0 { a } else { a + 1 };
  ku = if kl == a { a + 1 } else { a };
  (qa, _ra) = lattice_to_cell(kl, 3 * lo);
  (qb, _rb) = lattice_to_cell(ku, 3 * (lo + 1));
  TextSpot { kind: 2, qa: qa, ra: lo, qb: qb, rb: lo + 1 }
}

// Where `c` stands in `chars`; -1 when it does not.
fn text_index(chars: text, c: character) -> integer {
  i = 0;
  for o in chars {
    if o == c { return i; }
    i += 1;
  }
  -1
}

fn text_char(chars: text, at: integer) -> text {
  i = 0;
  for o in chars {
    if i == at { return "{o}"; }
    i += 1;
  }
  " "
}

// Read a stencil from its text: `rows` top line first, the character at line `line`, column
// `col` (both from 0) being cell (0, 0).  The stencil has labels and edges; its window is its
// cells' with room for the marks round them.
pub fn stencil_read(rows: const vector<text>, lg: const StencilLegend, line: integer, col: integer) -> StencilRead {
  errors: vector<text> = [];
  cq: vector<integer> = [];
  cr: vector<integer> = [];
  cl: vector<integer> = [];
  ea: vector<TextSpot> = [];
  em: vector<integer> = [];
  for li in 0..len(rows) {
    ci = 0;
    for c in rows[li] ?? "" {
      at = ci;
      ci += 1;
      if c == ' ' { continue; }
      // floor halves of numbers that may be negative: shift both into the positive
      s = text_spot(line - li + 4096, at - col + 8192);
      if s.kind == 0 {
        errors += ["line {li + 1} column {at + 1}: '{c}' stands between the places of the grid"];
      } else if s.kind == 1 {
        i = text_index(lg.sl_cells, c);
        if i < 0 {
          errors += ["line {li + 1} column {at + 1}: '{c}' stands where a cell does and is no cell of the legend"];
          continue;
        }
        cq += [s.qa - 2048];
        cr += [s.ra - 2048];
        cl += [lg.sl_labels[i] ?? 0];
      } else {
        i = text_index(lg.sl_edges, c);
        if i < 0 {
          errors += ["line {li + 1} column {at + 1}: '{c}' stands where an edge does and is no edge of the legend"];
          continue;
        }
        ea += [TextSpot { kind: 2, qa: s.qa - 2048, ra: s.ra - 2048, qb: s.qb - 2048, rb: s.rb - 2048 }];
        em += [lg.sl_mats[i] ?? 0];
      }
    }
  }
  q0 = 0;
  q1 = 0;
  r0 = 0;
  r1 = 0;
  for i in 0..len(cq) {
    if i == 0 || cq[i] < q0 { q0 = cq[i] ?? 0; }
    if i == 0 || cq[i] > q1 { q1 = cq[i] ?? 0; }
    if i == 0 || cr[i] < r0 { r0 = cr[i] ?? 0; }
    if i == 0 || cr[i] > r1 { r1 = cr[i] ?? 0; }
  }
  if len(cq) == 0 { errors += ["no cell in the text"]; }
  w = q1 - q0 + 5;
  h = r1 - r0 + 5;
  cells = hexset_chunk(q0 - 2, r0 - 2, w, h);
  lb = labels_new(q0 - 2, r0 - 2, w, h);
  edges = edgeset_new(q0 - 2, r0 - 2, w, h);
  for i in 0..len(cq) {
    hexset_set(cells, cq[i] ?? 0, cr[i] ?? 0, true);
    label_set(lb, cq[i] ?? 0, cr[i] ?? 0, cl[i] ?? 0);
  }
  for i in 0..len(ea) {
    e = ea[i]!;
    if !hexset_get(cells, e.qa, e.ra) && !hexset_get(cells, e.qb, e.rb) {
      errors += ["an edge between cells ({e.qa}, {e.ra}) and ({e.qb}, {e.rb}) touches no cell of the template"];
      continue;
    }
    edge_set_mat(edges, e.qa, e.ra, e.qb, e.rb, em[i] ?? 0);
  }
  StencilRead { sr_stencil: stencil_with_edges(cells, heights_new(q0 - 2, r0 - 2, w, h), lb, edges, false, true), sr_errors: errors }
}

// A stencil's text: its cells by their labels, and every mark on an edge of one of its cells.  A
// label or a material the legend has no character for is written `?`, which does not read back.
pub fn stencil_rows(st: Stencil, lg: const StencilLegend) -> StencilRows {
  cells = st.sn_cells;
  k0 = 0;
  k1 = 0;
  r0 = 0;
  r1 = 0;
  any = false;
  for r in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for q in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if !hexset_get(cells, q, r) { continue; }
      k = lattice_k(q, r);
      if !any || k < k0 { k0 = k; }
      if !any || k > k1 { k1 = k; }
      if !any || r < r0 { r0 = r; }
      if !any || r > r1 { r1 = r; }
      any = true;
    }
  }
  if !any { return StencilRows { sw_rows: [], sw_line: 0, sw_col: 0 }; }
  // a line above the top row and one below the bottom row for their outer edges; a cell's width
  // left of the leftmost and right of the rightmost for theirs
  line = 2 * r1 + 1;
  col = 2 - 2 * k0;
  rows: vector<text> = [];
  for li in 0..2 * (r1 - r0) + 3 {
    out = "";
    gap = "";
    for at in 0..2 * (k1 - k0) + 5 {
      s = text_spot(line - li + 4096, at - col + 8192);
      ch = " ";
      if s.kind == 1 && hexset_get(cells, s.qa - 2048, s.ra - 2048) {
        l = label_get(st.sn_labels, s.qa - 2048, s.ra - 2048);
        i = -1;
        for j in 0..len(lg.sl_labels) { if i < 0 && lg.sl_labels[j] == l { i = j; } }
        ch = if i < 0 { "?" } else { text_char(lg.sl_cells, i) };
      } else if s.kind == 2 && stencil_has_edges(st) {
        m = edge_mat(st.sn_edges, s.qa - 2048, s.ra - 2048, s.qb - 2048, s.rb - 2048);
        if m != 0 && (hexset_get(cells, s.qa - 2048, s.ra - 2048) || hexset_get(cells, s.qb - 2048, s.rb - 2048)) {
          i = -1;
          for j in 0..len(lg.sl_mats) { if i < 0 && lg.sl_mats[j] == m { i = j; } }
          ch = if i < 0 { "?" } else { text_char(lg.sl_edges, i) };
        }
      }
      if ch == " " { gap += " "; } else {
        out += gap + ch;
        gap = "";
      }
    }
    rows += [out];
  }
  StencilRows { sw_rows: rows, sw_line: line, sw_col: col }
}

// ── cellturn: a structure turned by any number of 30-degree steps ────────────

// Is `p` (a coordinate along heading `head`, scaled) between the lines `lo` and `hi` half steps out?
fn turn_between(head: integer, p: integer, lo: integer, hi: integer) -> boolean {
  if hex_form::head_is_edge(head) { return p >= 12 * lo && p <= 12 * hi; }
  hex_form::sign_root3(p, -12 * lo) >= 0 && hex_form::sign_root3(p, -12 * hi) <= 0
}

// A structure drawn at 0 degrees, by exact tests: a body 2w by 2d half hex steps about cell
// (0, 0) and, when `wing` is not 0, a wing `wing` hex steps square on the west end of its north side.
fn turn_drawn(w: integer, d: integer, wing: integer, reach: integer) -> HexSet {
  cells = hexset_chunk(0 - reach, 0 - reach, 2 * reach + 1, 2 * reach + 1);
  us = hex_form::head_step(0);
  vs = hex_form::head_step(3);
  for r in 0 - reach..reach + 1 {
    for q in 0 - reach..reach + 1 {
      u = hex_form::vec_dot_4(2 * lattice_k(q, r), 2 * lattice_m(q, r), us.0, us.1);
      v = hex_form::vec_dot_4(2 * lattice_k(q, r), 2 * lattice_m(q, r), vs.0, vs.1);
      body = turn_between(0, u, 0 - w, w) && turn_between(3, v, 0 - d, d);
      side = wing > 0 && turn_between(0, u, 0 - w, 0 - w + 2 * wing) && turn_between(3, v, d, d + 2 * wing);
      if body || side { hexset_set(cells, q, r, true); }
    }
  }
  cells
}

// The rules a turn keeps, on one drawn structure at all twelve steps: an even step is the
// lattice's own turn cell for cell (@FR-HT-Turn60) with no offset; an odd step keeps the walls —
// as many, each a quarter turn from the next, each 30 degrees on from the drawing's
// (@FR-HT-Shape) — every cell centre clearly inside or outside them (@FR-HW-Clear), and the
// cells it gains or loses few against the length of the walls.
fn turn_keeps(name: text, cells: HexSet, walls: integer) {
  drawn = hex_draw::cells_built(cells, 9, 0);
  assert(len(drawn.cw_walls) == walls, "{name}: drawn with {walls} walls, got {len(drawn.cw_walls)}");
  n = hexset_count(cells);
  edges = len(drawn.cw_boundary);
  for s in 0..12 {
    t = cells_turn(cells, 0, 0, s);
    if s % 2 == 0 {
      assert(t.tn_sx == 0.0 && t.tn_sy == 0.0, "{name} at {30 * s}: an even step moves nothing");
      assert(hexset_count(t.tn_cells) == n, "{name} at {30 * s}: as many cells");
      for r in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
        for q in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
          if !hexset_get(cells, q, r) { continue; }
          (tq, tr) = cell_rot(q, r, s / 2 ?? 0);
          assert(hexset_get(t.tn_cells, tq, tr), "{name} at {30 * s}: cell {q},{r} turned as the lattice turns it");
        }
      }
    }
    b = hex_draw::cells_built(t.tn_cells, 9, s);
    assert(len(b.cw_walls) == walls, "{name} at {30 * s}: its {walls} walls kept, got {len(b.cw_walls)}");
    for i in 0..walls {
      w = b.cw_walls[i]!;
      nx = b.cw_walls[(i + 1) % walls ?? 0]!;
      assert(hex_form::vec_dot_4(w.ws_dk, w.ws_dm, nx.ws_dk, nx.ws_dm) == 0, "{name} at {30 * s} wall {i}: square to the next");
      assert(hex_form::head_norm(hex_draw::surface_heading(w) - s) % 3 == 0, "{name} at {30 * s} wall {i}: along the turned axes");
    }
    gained = hexset_count(t.tn_cells) - n;
    if gained < 0 { gained = 0 - gained; }
    assert(gained <= edges / 2 ?? 0, "{name} at {30 * s}: {gained} cells gained or lost, the boundary {edges} edges");
    // every centre clearly in or out: no centre within 0.2 of a wall's line between its corners
    for r in hexset_r0(t.tn_cells)..hexset_r0(t.tn_cells) + hexset_h(t.tn_cells) {
      for q in hexset_q0(t.tn_cells)..hexset_q0(t.tn_cells) + hexset_w(t.tn_cells) {
        x = hex_form::cell_x(q, r);
        y = hex_form::cell_y(q, r);
        for i in 0..walls {
          w = b.cw_walls[i]!;
          (dx, dy) = hex_draw::surface_dir(w);
          a = b.cw_corners[(i + walls - 1) % walls ?? 0] ?? (0.0, 0.0);
          c = b.cw_corners[i] ?? (0.0, 0.0);
          along = (x - a.0) * dx + (y - a.1) * dy;
          len_w = (c.0 - a.0) * dx + (c.1 - a.1) * dy;
          across = (x - a.0) * (0.0 - dy) + (y - a.1) * dx;
          if across < 0.0 { across = 0.0 - across; }
          assert(along < 0.0 || along > len_w || across > 0.2, "{name} at {30 * s}: cell {q},{r} lies on wall {i} ({across})");
        }
      }
    }
    // a point carried through the turn: the pivot stays but for the odd step's offset, and a
    // point's distance from the pivot's image is kept
    px = hex_form::cell_x(0, 0);
    py = hex_form::cell_y(0, 0);
    (ox, oy) = turned_point(t, px, py);
    (qx, qy) = turned_point(t, px + 3.0, py + 4.0);
    d = sqrt((qx - ox) * (qx - ox) + (qy - oy) * (qy - oy));
    assert(d > 4.999999 && d < 5.000001, "{name} at {30 * s}: a point 5 from the pivot stays 5 from it ({d})");
    moved = sqrt((ox - px) * (ox - px) + (oy - py) * (oy - py));
    assert(moved < 1.0, "{name} at {30 * s}: the structure moved by under a row ({moved})");
  }
}

// ── a stencil through a turn ────────────────────────────────────────────────

const OUTER = 7;
const INNER = 5;

// A hall as a stencil: 24 hex steps by 16 about cell (0, 0), its outer walls of material 7, a
// door in the middle of its first wall, and inner walls of material 5: one across it where the
// doubled column passes `split`, with a door in it — and, when `tee`, one more down the east part
// where the doubled row passes 0.  Each cell's label and its layer "room" are its room's number
// (0 west; 1 east, or north-east; 2 south-east), its height its room's number and a half.
fn hall_stencil(split: integer, tee: boolean) -> Stencil {
  reach = 30;
  cells = turn_drawn(24, 16, 0, reach);
  edges = edgeset_new(0 - reach, 0 - reach, 2 * reach + 1, 2 * reach + 1);
  hex_draw::draw_walls(cells, edges, OUTER);
  hz = heights_new(0 - reach, 0 - reach, 2 * reach + 1, 2 * reach + 1);
  lb = labels_new(0 - reach, 0 - reach, 2 * reach + 1, 2 * reach + 1);
  for r in 0 - reach..reach + 1 {
    for q in 0 - reach..reach + 1 {
      if !hexset_get(cells, q, r) { continue; }
      room = if lattice_k(q, r) < split { 0 } else if tee && lattice_m(q, r) < 0 { 2 } else { 1 };
      label_set(lb, q, r, room);
      height_set(hz, q, r, (room as float) + 0.5);
    }
  }
  // the inner walls: wherever two cells of different rooms meet
  for r in 0 - reach..reach + 1 {
    for q in 0 - reach..reach + 1 {
      if !hexset_get(cells, q, r) { continue; }
      for d in 0..6 {
        nq = nb_q(q, r, d);
        nr = nb_r(q, r, d);
        if hexset_get(cells, nq, nr) && label_get(lb, q, r) != label_get(lb, nq, nr) {
          // a door in the wall across the hall, two edges of it just north of the middle
          door = label_get(lb, q, r) + label_get(lb, nq, nr) == 1 && lattice_m(q, r) == 6 && lattice_m(nq, nr) == 6;
          edge_set_mat(edges, q, r, nq, nr, if door { hex_draw::OPEN_DOOR } else { INNER });
        }
      }
    }
  }
  built = hex_draw::cells_built(cells, OUTER, 0);
  a = built.cw_corners[len(built.cw_corners) - 1] ?? (0.0, 0.0);
  b = built.cw_corners[0] ?? (0.0, 0.0);
  hex_draw::cells_place_opening(built, edges, hex_draw::OPEN_DOOR, (a.0 + b.0) * 0.5, (a.1 + b.1) * 0.5, hex_draw::opening_sizes_default());
  st = stencil_with_edges(cells, hz, lb, edges, true, true);
  layer_add(st.sn_layers, "room");
  for r in 0 - reach..reach + 1 {
    for q in 0 - reach..reach + 1 {
      if hexset_get(cells, q, r) { layer_set(st.sn_layers, "room", q, r, label_get(lb, q, r)); }
    }
  }
  st
}

// What a turned hall keeps.  Its rooms, counted afresh by walking from cell to cell wherever no
// edge is marked, are `rooms`: no inner wall leaks and none is doubled.  In each such room every
// cell has one label, one height and one "room" layer — the drawn room's — so what a cell holds
// came through with its room.  The inner walls are there in their material with their door; the
// outer walls in theirs with theirs.
fn stencil_keeps(name: text, st: Stencil, t: StencilTurn, rooms: integer, step: integer) {
  cells = t.tu_stencil.sn_cells;
  edges = t.tu_stencil.sn_edges;
  seen = labels_new(hexset_q0(cells), hexset_r0(cells), hexset_w(cells), hexset_h(cells));
  found = 0;
  inner = 0;
  inner_doors = 0;
  outer_doors = 0;
  outer = 0;
  for r in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
    for q in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
      if !hexset_get(cells, q, r) { continue; }
      for d in 0..6 {
        nq = nb_q(q, r, d);
        nr = nb_r(q, r, d);
        m = edge_mat(edges, q, r, nq, nr);
        if !hexset_get(cells, nq, nr) {
          assert(m == OUTER || m == hex_draw::OPEN_DOOR, "{name} at {30 * step}: an outer edge of material {m}");
          outer += 1;
          if m == hex_draw::OPEN_DOOR { outer_doors += 1; }
        } else if m != 0 && (nr > r || (nr == r && nq > q)) {
          assert(m == INNER || m == hex_draw::OPEN_DOOR, "{name} at {30 * step}: an inner edge of material {m}");
          inner += 1;
          if m == hex_draw::OPEN_DOOR { inner_doors += 1; }
        }
      }
      if label_get(seen, q, r) >= 0 { continue; }
      // a room: from this cell to every cell reached across no marked edge
      label_set(seen, q, r, found);
      want = label_get(t.tu_stencil.sn_labels, q, r);
      sq: vector<integer> = [q];
      sr: vector<integer> = [r];
      at = 0;
      while at < len(sq) {
        cq = sq[at] ?? 0;
        cr = sr[at] ?? 0;
        at += 1;
        assert(label_get(t.tu_stencil.sn_labels, cq, cr) == want, "{name} at {30 * step}: one label a room");
        assert(height_get(t.tu_stencil.sn_heights, cq, cr) == (want as float) + 0.5, "{name} at {30 * step}: its room's height");
        assert(layer_get(t.tu_stencil.sn_layers, "room", cq, cr) == want, "{name} at {30 * step}: its room's layer");
        for d in 0..6 {
          nq = nb_q(cq, cr, d);
          nr = nb_r(cq, cr, d);
          if !hexset_get(cells, nq, nr) || label_get(seen, nq, nr) >= 0 || edge_mat(edges, cq, cr, nq, nr) != 0 { continue; }
          label_set(seen, nq, nr, found);
          sq += [nq];
          sr += [nr];
        }
      }
      assert(len(sq) > 20, "{name} at {30 * step}: a room of {len(sq)} cells");
      found += 1;
    }
  }
  assert(found == rooms && t.tu_rooms == rooms, "{name} at {30 * step}: {rooms} rooms, walked {found}, told {t.tu_rooms}");
  assert(t.tu_lost == 0, "{name} at {30 * step}: no wall lost");
  assert(inner >= 14 && inner_doors >= 1 && inner_doors <= 3, "{name} at {30 * step}: the inner wall ({inner} edges) and its door ({inner_doors})");
  assert(outer_doors >= 1 && outer_doors <= 3 && outer > 60, "{name} at {30 * step}: the outer wall ({outer} edges) and its door ({outer_doors})");
  _ = st;
}

// The inner wall comes out straight.  Drawn, the wall across the hall runs north (heading 3);
// turned `s` steps it runs along heading 3 + s, and every one of its edges' midpoints lies within
// half a hex of one line along that heading.
fn test_an_inner_wall_stays_straight() {
  st = hall_stencil(5, false);
  for s in 0..12 {
    t = stencil_turn(st, s);
    cells = t.tu_stencil.sn_cells;
    a = ((3 + s) as float) * PI / 6.0;
    lo = 1000000.0;
    hi = -1000000.0;
    n = 0;
    for r in hexset_r0(cells)..hexset_r0(cells) + hexset_h(cells) {
      for q in hexset_q0(cells)..hexset_q0(cells) + hexset_w(cells) {
        if !hexset_get(cells, q, r) { continue; }
        for d in 0..6 {
          nq = nb_q(q, r, d);
          nr = nb_r(q, r, d);
          if !hexset_get(cells, nq, nr) || edge_mat(t.tu_stencil.sn_edges, q, r, nq, nr) == 0 { continue; }
          mx = (hex_form::cell_x(q, r) + hex_form::cell_x(nq, nr)) * 0.5;
          my = (hex_form::cell_y(q, r) + hex_form::cell_y(nq, nr)) * 0.5;
          across = -mx * sin(a) + my * cos(a);
          if across < lo { lo = across; }
          if across > hi { hi = across; }
          n += 1;
        }
      }
    }
    assert(n >= 28, "at {30 * s}: the inner wall's edges, each seen from both its cells: {n}");
    assert(hi - lo < 0.9, "at {30 * s}: the inner wall within a band of {hi - lo} across heading {(3 + s) % 12 ?? 0}");
  }
}

// How long an inner wall is.
fn inner_length(w: hex_draw::InnerWall) -> float {
  sqrt((w.iw_bx - w.iw_ax) * (w.iw_bx - w.iw_ax) + (w.iw_by - w.iw_ay) * (w.iw_by - w.iw_ay)) ?? 0.0
}

// The inner walls read back as straight walls (`hex_draw::cells_inner_walls`), drawn and at every
// turn.  The hall of two rooms has one, between rooms 0 and 1, of material 5, with its door; the
// hall of three has three — the wall across it is two stretches, one against each room east of it,
// and the wall between those two.  Turned `s` steps there are as many, each along one of the
// turned axes, each as long as it was drawn to within a hex step, and the door is in one of them.
fn test_inner_walls_read_back_straight_at_every_turn() {
  for tee in 0..2 {
    st = hall_stencil(5, tee == 1);
    drawn = hex_draw::cells_inner_walls(st.sn_cells, st.sn_edges, 0);
    want = if tee == 1 { 3 } else { 1 };
    assert(drawn.in_rooms == 2 + tee && len(drawn.in_walls) == want && drawn.in_lost == 0, "drawn, {2 + tee} rooms: {len(drawn.in_walls)} inner walls");
    assert(len(drawn.in_opens) == 1 && drawn.in_opens[0]!.io_kind == hex_draw::OPEN_DOOR, "drawn: its one door");
    lengths: vector<float> = [];
    for w in drawn.in_walls {
      assert(w.iw_mat == INNER && w.iw_a < w.iw_b && w.iw_edges >= 6, "drawn: a wall of material 5 between two rooms, {w.iw_edges} edges");
      lengths += [inner_length(w)];
    }
    lengths.sort();
    for s in 0..12 {
      t = stencil_turn(st, s);
      got = hex_draw::cells_inner_walls(t.tu_stencil.sn_cells, t.tu_stencil.sn_edges, s);
      assert(got.in_rooms == 2 + tee && got.in_lost == 0, "at {30 * s}: {got.in_rooms} rooms, {got.in_lost} lost");
      assert(len(got.in_walls) == want, "at {30 * s}: {want} inner walls, got {len(got.in_walls)}");
      assert(len(got.in_opens) == 1 && got.in_opens[0]!.io_kind == hex_draw::OPEN_DOOR, "at {30 * s}: the door, {len(got.in_opens)}");

      for w in got.in_walls {
        h = hex_draw::surface_heading(w.iw_surface);
        assert(h >= 0 && hex_form::head_norm(h - s) % 3 == 0, "at {30 * s}: an inner wall along a turned axis, heading {h}");
        assert(w.iw_mat == INNER, "at {30 * s}: of material 5");
        turned += [inner_length(w)];
      }
      turned.sort();
      for i in 0..want {
        d = (turned[i] ?? 0.0) - (lengths[i] ?? 0.0);
        assert(d < 1.8 && d > -1.8, "at {30 * s}: wall {i} as long as drawn to within a hex step, {turned[i] ?? 0.0} against {lengths[i] ?? 0.0}");
      }

      assert(off < 0.000001 && off > -0.000001, "at {30 * s}: the door's centre on its wall's line");
    }
  }
}

// A room one cell wide: a corridor down the hall, a room either side.  Along an odd heading a
// single row's cells stand apart, so the corridor's cells would come through 30 degrees loose,
// each a room of its own.  It comes through in one piece instead — three rooms at every step,
// counted afresh — and each side room still has most of its cells.
fn test_a_corridor_one_cell_wide_stays_one_room() {
  reach = 30;
  cells = turn_drawn(24, 16, 0, reach);
  edges = edgeset_new(0 - reach, 0 - reach, 2 * reach + 1, 2 * reach + 1);
  hex_draw::draw_walls(cells, edges, OUTER);
  lb = labels_new(0 - reach, 0 - reach, 2 * reach + 1, 2 * reach + 1);
  sizes = [0, 0, 0];
  for r in 0 - reach..reach + 1 {
    for q in 0 - reach..reach + 1 {
      if !hexset_get(cells, q, r) { continue; }
      // one straight row of cells, each touching the next
      room = if r < 0 { 0 } else if r == 0 { 1 } else { 2 };
      label_set(lb, q, r, room);
      sizes[room] += 1;
    }
  }
  for r in 0 - reach..reach + 1 {
    for q in 0 - reach..reach + 1 {
      if !hexset_get(cells, q, r) { continue; }
      for d in 0..6 {
        nq = nb_q(q, r, d);
        nr = nb_r(q, r, d);
        if hexset_get(cells, nq, nr) && label_get(lb, q, r) != label_get(lb, nq, nr) { edge_set_mat(edges, q, r, nq, nr, INNER); }
      }
    }
  }
  st = stencil_with_edges(cells, heights_new(0 - reach, 0 - reach, 2 * reach + 1, 2 * reach + 1), lb, edges, false, true);
  assert(hex_draw::cells_rooms(cells, edges).cr_n == 3 && sizes[1] > 6 && sizes[1] < sizes[0] / 4, "drawn: a corridor of {sizes[1]} cells between rooms of {sizes[0]} and {sizes[2]}");
  for s in 0..12 {
    t = stencil_turn(st, s);
    rooms = hex_draw::cells_rooms(t.tu_stencil.sn_cells, t.tu_stencil.sn_edges);
    assert(rooms.cr_n == 3 && t.tu_rooms == 3 && t.tu_lost == 0, "at {30 * s}: three rooms, counted {rooms.cr_n}, told {t.tu_rooms}");
    got = [0, 0, 0];
    for r in hexset_r0(t.tu_stencil.sn_cells)..hexset_r0(t.tu_stencil.sn_cells) + hexset_h(t.tu_stencil.sn_cells) {
      for q in hexset_q0(t.tu_stencil.sn_cells)..hexset_q0(t.tu_stencil.sn_cells) + hexset_w(t.tu_stencil.sn_cells) {
        if hexset_get(t.tu_stencil.sn_cells, q, r) { got[label_get(t.tu_stencil.sn_labels, q, r)] += 1; }
      }
    }
    assert(got[1] >= sizes[1] / 2 && got[0] * 10 >= sizes[0] * 8 && got[2] * 10 >= sizes[2] * 8, "at {30 * s}: the corridor {got[1]} of {sizes[1]} cells, the rooms {got[0]} of {sizes[0]} and {got[2]} of {sizes[2]}");
  }
}

// Worked examples for `hex_place` (@PLN141).  Each `test_*` below is a REAL,
// run-every-CI demonstration of a contract the signature cannot state; the
// function documents itself by CITING the tag with `// Example: @HXP-00N`, and
// `check_doc_drift.sh examples` fails if a citation goes dangling.  These are
// teaching examples, distinct from the conformance suite in `01-hex-place.loft`.
//
// WHAT THIS PACKAGE IS ABOUT.  Everything here answers one question — what
// happens where two things MEET — and the answers all have the same shape: the
// contention is removed by CONSTRUCTION rather than settled by a rule applied
// afterwards.  Two stencils fuse because the seam is interior to the union that
// was cut.  Two levels never contend because the level filters before the cut.
// Terrain never moves a footprint because seating writes a different slot.
//
// That shape is exactly what a signature cannot carry: every one of these is a
// call that quietly does the right thing, and a NEARBY call that quietly does
// the wrong one.  The nearby call is what each example shows beside the right
// one, because a demonstration of only the good half teaches nothing about
// which half you are in.
use hex_place::*;
use hex_field::*;

// A 3x3 block of cells, its left column at `off`, in a chunk with room around it.
fn box3(off: integer) -> HexSet {
  s = hexset_chunk(-10, -10, 21, 21);
  for r in 0..3 { for q in 0..3 { hexset_set(s, q + off, r, true); } }
  s
}

// @HXP-001 — TWO ADJACENT STENCILS: WHO OWNS THE SHARED EDGE?  Nobody — and
// that is a property of HOW you cut, not of the stencils.  Union the footprints
// first and cut the boundary of the union once ("mark all, then cut once"), and
// the shared edge is interior to that union, so it is never cut: the two become
// one fabric with no seam wall between them.
//
// Draw each stencil's own walls instead and overlay them, and both sides mark
// the seam — a wall appears down the middle of what the author drew as one
// building, and every cell of it is reachable, so nothing downstream complains.
//
// This is the same question `hex_grid`'s @HXG-003 answers one level down: there,
// ONE wall between two hexes must resolve to one stored slot, or it becomes two
// half-walls that disagree.  Here it is one seam between two STENCILS, and the
// answer is that it should not be stored at all.
fn test_the_shared_edge_belongs_to_nobody_if_you_cut_once() {
  a = box3(0);                    // cells q 0..2
  b = box3(3);                    // cells q 3..5 — adjacent, not overlapping

  // THE WRONG WAY, shown first: cut each stencil's own boundary into one edge
  // set.  `combine_cut(a, 1, a, 1, e)` is exactly "a's own walls".
  naive = edgeset_new(-10, -10, 21, 21);
  na = combine_cut(a, 1, a, 1, naive);
  nb = combine_cut(b, 2, b, 2, naive);
  assert(na == 22 && nb == 22, "each 3x3 block alone has 22 boundary edges");
  seam = shared_marked(a, b, naive);
  assert(seam == 5, "and a wall now runs between them: {seam} marked edges");

  // THE RIGHT WAY: one call, one cut, over the union.
  fused = edgeset_new(-10, -10, 21, 21);
  ng = combine_cut(a, 1, b, 2, fused);
  assert(shared_marked(a, b, fused) == 0, "no wall on the shared interior");

  // And the arithmetic says exactly where the difference went: the five seam
  // edges were each cut twice by the overlay and not at all by the union.
  assert(ng == na + nb - 2 * seam, "{ng} = {na} + {nb} - 2 * {seam}");
}

// @HXP-002 — ORDER-FREE BY CONSTRUCTION, including where the stencils OVERLAP.
// The result reads the finished union and a fixed source map, never which
// stencil was stamped when, so swapping the arguments cannot change a single
// edge.  Where two stencils claim the same cell the tie goes to the LOWER
// MATERIAL ID — an intrinsic property of the pair, not a positional one.
//
// The distinction matters because "last writer wins" is the natural way to
// build this and is indistinguishable from correct on ONE stencil.  It only
// diverges on two, and then only sometimes, which is the worst way for a bug
// to present.
fn test_swapping_the_two_stencils_changes_nothing() {
  c = box3(0);                    // q 0..2
  d = box3(2);                    // q 2..4 — the q=2 column is in BOTH

  one = edgeset_new(-10, -10, 21, 21);
  two = edgeset_new(-10, -10, 21, 21);
  n1 = combine_cut(c, 1, d, 2, one);
  n2 = combine_cut(d, 2, c, 1, two);
  assert(n1 == 30 && n2 == 30, "the same {n1} edges either way round");

  // Not just the same count — the same material on every edge.
  same = true;
  for r in -10..11 {
    for q in -10..11 {
      for dir in 0..6 {
        nq = nb_q(q, r, dir);
        nr = nb_r(q, r, dir);
        if edge_mat(one, q, r, nq, nr) != edge_mat(two, q, r, nq, nr) { same = false; }
      }
    }
  }
  assert(same, "and the same material on every one of them");
}

// @HXP-003 — A LEVEL IS A FILTER APPLIED BEFORE THE CUT, not an arbitration
// rule applied after it.  Two stencils on different levels do not fuse, do not
// arbitrate and do not contend: at either level, one of them simply is not
// there.  So the contention degree is 1 at each level even where the two
// overlap in (q,r) — a bridge over a road contends with nothing.
//
// A LEVEL IS NOT A HEIGHT.  It is a discrete sheet index; the actual z comes
// from the height slot (@HXP-004).  Two things can share a z on different
// levels, or sit at different z on the same one.
fn test_two_levels_never_contend_however_much_they_overlap() {
  c = box3(0);                    // q 0..2
  d = box3(2);                    // q 2..4 — overlaps c at q=2

  // Seen at level 0, with c on level 0 and d on level 1, only c is there —
  // so the cut is c's own boundary, exactly as if d did not exist.
  lv = edgeset_new(-10, -10, 21, 21);
  n0 = combine_cut_level(c, 0, 1, d, 1, 2, 0, lv);
  solo = edgeset_new(-10, -10, 21, 21);
  nsolo = combine_cut(c, 1, c, 1, solo);
  assert(n0 == nsolo, "at level 0 the cut is c alone: {n0} edges");

  // The overlap cell is claimed once at each level, never twice at either.
  assert(kappa_at_level(c, 0, d, 1, 0, 2, 0) == 1, "one claim at level 0");
  assert(kappa_at_level(c, 0, d, 1, 1, 2, 0) == 1, "one claim at level 1");

  // Put them on the SAME level and that same cell is contended by two.
  assert(kappa_at_level(c, 0, d, 0, 0, 2, 0) == 2, "same level: two claims, one cell");
}

// @HXP-004 — SEATING WRITES THE HEIGHT SLOT AND NOTHING ELSE.  A stencil is
// authored flat and the ground under it is not, so seating has to give somewhere
// — and where it gives is a number RETURNED to you, never a change to the
// footprint.  The cells are byte-identical on flat ground and on a slope, which
// is what lets the model be recovered from the cells either way.
//
// The policy (low / mean / high) chooses the residual, not the shape: a building
// perches, splits the difference, or is buried, and the footprint is the same
// three by three cells in all three cases.  Read `seat_residual` and show it —
// it is the cut and fill the placement owes, and swallowing it is the one thing
// this design refuses to do for you.
fn test_seating_moves_the_height_never_the_footprint() {
  a = box3(0);
  terr = heights_new(-10, -10, 21, 21);
  for r in 0..3 { for q in 0..3 { height_set(terr, q, r, q as float); } }   // a slope, 0..2

  // Three policies, three seat heights, over the same nine cells.
  assert(seat_height(a, terr, SEAT_LOW) == 0.0, "perch on the lowest ground");
  assert(seat_height(a, terr, SEAT_MEAN) == 1.0, "or split the difference");
  assert(seat_height(a, terr, SEAT_HIGH) == 2.0, "or bury into the slope");

  // The residual is the choice's cost, and it is handed back rather than absorbed.
  assert(seat_residual(a, terr, 1.0) == 1.0, "seated at the mean, the worst gap is 1.0");
  assert(seat_residual(a, terr, 0.0) == 2.0, "perched, it is 2.0 — the same slope, a worse fit");

  // Seating writes heights.  The footprint does not move.
  before = hexset_count(a);
  wrote = seat_write(a, terr, 1.0);
  assert(wrote == 9, "nine cells got a height");
  assert(hexset_count(a) == before, "and the footprint is the same {before} cells");
  assert(height_get(terr, 0, 0) == 1.0, "the ground under the stencil now reads the seat height");
}

// @HXP-005 — THE POSE TRANSFORM IS THE ONLY PLACE FLOAT ERROR LIVES, and ONE
// MEASUREMENT OF IT IS NOT A BOUND.  Inside a frame every test is exact
// integer work; across frames it cannot be, because a pose is continuous.  The
// design confines that error to the transform so the whole stack has one error
// budget — but the size of it depends on WHERE you evaluate it.
//
// Measure the residual near the frame origin and you will very reasonably read
// exactly zero, and conclude the transform is exact.  The same pose, at a point
// a hundred times further out, is not zero.  It is deterministic — the same
// inputs always give the same answer, which is what replay needs — but it is a
// jagged function of the point, not a constant you can sample once.
//
// So: never compare cross-frame coordinates with `==`, and never take a
// tolerance from a measurement made next to the origin.
fn test_one_sample_of_the_seam_error_is_not_a_bound() {
  g = pose_new(0.8660254037844386, 0.5, 3.0, 4.0);     // a 30-degree pose

  // Near the origin: exactly zero, and that is a true measurement.
  near = pose_residual(g, 10.0, 7.0);
  assert(near == 0.0, "at (10,7) the round trip is exact");

  // The same pose, further out: not zero.
  far = pose_residual(g, 1000.0, 700.0);
  assert(far > 0.0, "at (1000,700) it is not, though nothing about the pose changed");
  assert(far < 0.000001, "it is still tiny — the seam is tight, it is simply not zero");

  // Deterministic, which is what byte-identical replay rests on.
  assert(pose_residual(g, 1000.0, 700.0) == far, "the same inputs give the same answer");

  // Inside the frame there is no such caveat: the forward map lands a local
  // point exactly on the boundary of a local disk test, radius 5 about (0,0).
  wx = pose_fwd_x(g, 3.0, 4.0);
  wy = pose_fwd_y(g, 3.0, 4.0);
  assert(disk_hit(g, 0.0, 0.0, 25.0, wx, wy), "the local test is exact in its own frame");
}

// @HXP-006 — ARBITRATION FAILS SAFE TOWARD SOLID, and the owner is INTRINSIC.
// Where two frames disagree about a point, `arb_solid` answers solid if either
// does: a crack between presentations yields a spurious contact rather than a
// body falling through the world. It is symmetric, so it cannot depend on which
// frame you happened to visit first.
//
// `arb_owner` breaks the tie by LOWEST ID — a total order on frames, not
// "whichever is nearer" and not iteration order, either of which would make a
// replay diverge. Its no-owner answer is **-1**, and -1 is a sentinel, not an
// id: in loft a negative index reads from the END of a vector rather than
// failing, so handing it on unchecked finds a frame, just not the right one.
fn test_arbitration_prefers_solid_and_names_an_owner_intrinsically() {
  assert(arb_solid(true, false), "solid over gap is solid");
  assert(arb_solid(false, true), "and so is gap over solid — it is symmetric");
  assert(!arb_solid(false, false), "only where both agree is it a gap");

  // Both solid: the lower id owns it, whichever way round you ask.
  assert(arb_owner(true, 7, true, 3) == 3, "lowest id wins");
  assert(arb_owner(true, 3, true, 7) == 3, "and the answer does not depend on the order");

  // One solid: that one owns it, high id or not.
  assert(arb_owner(false, 7, true, 3) == 3, "the solid frame owns it");
  assert(arb_owner(true, 7, false, 3) == 7, "even when its id is the larger");

  // Neither: -1, which is not a frame.
  assert(arb_owner(false, 7, false, 3) == -1, "no owner is -1, and -1 is not an id");
}
