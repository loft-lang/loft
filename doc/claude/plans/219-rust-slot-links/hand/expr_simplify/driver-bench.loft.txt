// Benchmark 22: A recursive enum — arithmetic expressions as a user-built tree (`Expr`), whose
// `Add` and `Mul` variants hold two expressions each OUT OF LINE (`formal/layout.md`
// L-OutOfLine).
//
// Four routines, each ONE OP per repetition on a complete expression of depth 10 (1,024
// numbers, 1,023 operators): its SHAPE is fixed, the salt changes which operator each node is
// and which number each leaf holds, so the work never varies with it.  Folded into `sink`:
//   expr_build     an expression built by recursion, then its size and its value
//   expr_eval      the value of a prebuilt expression (modulo 1,000,000,007), its numbers shifted
//                  by the salt
//   expr_simplify  a prebuilt expression rewritten without `x * 0`, `x * 1` and `x + 0`, then the
//                  size and the value of the result (the value must not change)
//   expr_render    a prebuilt expression rendered as fully parenthesised text, then its length
//                  and a hash of its characters
// The `hash` column is one canonical op's result: every lane must print the same one.

enum Expr {
  Num { v: integer },
  Add { l: Expr, r: Expr },
  Mul { l: Expr, r: Expr },
}

/// A scrambled number: an LCG step of `i` (the 21_tree key).
fn key(i: integer, salt: integer) -> integer { ((i * 1103515245 + 12345 + salt * 7919) % 2147483648) % 100000 }

/// The complete expression of depth `d` whose root is node `i` in heap order.
fn gen(d: integer, i: integer, salt: integer) -> Expr {
  h = key(i, salt);
  if d == 0 { return Num { v: h % 10 }; }
  if (h / 7) % 2 == 0 {
    Add { l: gen(d - 1, 2 * i + 1, salt), r: gen(d - 1, 2 * i + 2, salt) }
  } else {
    Mul { l: gen(d - 1, 2 * i + 1, salt), r: gen(d - 1, 2 * i + 2, salt) }
  }
}

fn size(e: Expr) -> integer {
  match e {
    Num { v: _ } => 1,
    Add { l, r } => 1 + size(l) + size(r),
    Mul { l, r } => 1 + size(l) + size(r),
  }
}

fn eval(e: Expr, shift: integer) -> integer {
  match e {
    Num { v } => v + shift,
    Add { l, r } => (eval(l, shift) + eval(r, shift)) % 1000000007,
    Mul { l, r } => (eval(l, shift) * eval(r, shift)) % 1000000007,
  }
}

fn is_num(e: Expr, n: integer) -> boolean {
  match e {
    Num { v } => v == n,
    _ => false,
  }
}

/// `x * 0` is 0, `x * 1` and `x + 0` are x; applied bottom-up, so a rewritten child can enable
/// its parent's rewrite.
fn simplify(e: Expr) -> Expr {
  match e {
    Num { v } => Num { v: v },
    Add { l, r } => {
      a = simplify(l);
      b = simplify(r);
      if is_num(a, 0) { return b; }
      if is_num(b, 0) { return a; }
      Add { l: a, r: b }
    }
    Mul { l, r } => {
      a = simplify(l);
      b = simplify(r);
      if is_num(a, 0) || is_num(b, 0) { return Num { v: 0 }; }
      if is_num(a, 1) { return b; }
      if is_num(b, 1) { return a; }
      Mul { l: a, r: b }
    }
  }
}

fn render(e: Expr) -> text {
  match e {
    Num { v } => "{v}",
    Add { l, r } => "({render(l)}+{render(r)})",
    Mul { l, r } => "({render(l)}*{render(r)})",
  }
}

fn expr_build(salt: integer) -> integer {
  e = gen(10, 0, salt);
  size(e) * 1000003 + eval(e, 0)
}

fn expr_eval(e: Expr, salt: integer) -> integer { eval(e, salt) }

fn expr_simplify(e: Expr) -> integer {
  s = simplify(e);
  size(s) * 1000003 + eval(s, 0)
}

fn expr_render(e: Expr) -> integer {
  s = render(e);
  h = 0;
  for c in s { h = (h * 31 + c as integer) % 1000000007; }
  len(s) * 1000003 + h
}

/// `--n N`: how many ops the timed loop runs (the harness calibrates it per lane).
fn arg_n(dflt: integer) -> integer {
  an_args = arguments();
  an_n = dflt;
  an_i = 0;
  while an_i < len(an_args) {
    if an_args[an_i]? == "--n" && an_i + 1 < len(an_args) {
      an_n = (an_args[an_i + 1]? as integer?)?;
      an_i += 1;
    }
    an_i += 1;
  }
  if an_n < 2 { an_n = 2; }
  an_n
}

/// One measured routine, in the row format every lane prints (bench/README.md).
fn row(name: text, iters: integer, us: integer, items: integer, result: integer, sink: integer) {
  rw_ns = ((us * 1000) / iters) ?? 0;
  rw_it = (((us * 1000) as float) / ((iters * items) as float)) ?? 0.0;
  println("routine\titers\tus\tns_op\tpx\tns_px\thash");
  println("{name}\t{iters}\t{us}\t{rw_ns}\t{items}\t{rw_it:.3}\t{result:x}");
  println("time: {us / 1000}ms sink={sink}");
}

fn main() {
  n = arg_n(10);
  e0 = gen(10, 0, 0);
  e1 = gen(10, 0, 1);
  t0 = ticks();
  sink = 0;
  for r in 0..n { sink += expr_simplify(if r & 1 == 0 { e0 } else { e1 }); }
  row("expr_simplify", n, ticks() - t0, 2047, expr_simplify(e0), sink);
}
