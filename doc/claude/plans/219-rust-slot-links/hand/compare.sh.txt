# bash compare.sh.txt ROUTINE [LOFT] — the @PLN219 hand forms of ROUTINE against what the current tree
# emits: re-emit the base from the stored driver, restore every form onto it, compile all with the
# C lane's flags (and the `_nt` twins with -DLOFT_NO_THREADS), build the Rust twin, measure.
# A form that no longer applies is named; re-base it from its step*.py.txt.
set -euo pipefail
here=$(cd "$(dirname "$0")" && pwd)
root=$(cd "$here/../../../../.." && pwd)
r=$1
loft=${2:-$root/target/release/loft}
w=$(mktemp -d "${TMPDIR:-/tmp}/pln219-$r.XXXX")
mkdir -p "$w/b"
cp "$here/$r/driver-bench.loft.txt" "$w/driver-bench.loft"
cp "$here/$r/driver-bench.rs.txt" "$w/driver-bench.rs"
LOFT_NO_CACHE=1 "$loft" --native-c --native-emit "$w/b/raw.c" "$w/driver-bench.loft" --n 2 >/dev/null 2>&1
sed "s|\"[^\"]*driver-bench.loft\"|\"DRIVER.loft\"|g" "$w/b/raw.c" > "$w/b/base.c"
python3 "$root/bench/portal/hand_forms.py" restore "$here/$r" "$w/b" "$w/forms" || true
"$root/bench/portal/hand_price_c.sh" "$w/b/base.c" "$w/base"
bins=("$w/rust" "$w/base")
rustc -O --edition 2021 "$w/driver-bench.rs" -o "$w/rust" 2>/dev/null
for f in "$w"/forms/*.c; do
  [ -e "$f" ] || continue
  n=$(basename "$f" .c)
  "$root/bench/portal/hand_price_c.sh" "$f" "$w/$n"
  HAND_PRICE_CFLAGS="-std=c99 -O3 -DLOFT_NO_THREADS" "$root/bench/portal/hand_price_c.sh" "$f" "$w/${n}_nt"
  bins+=("$w/$n" "$w/${n}_nt")
done
bash "$here/m.sh.txt" "${bins[@]}"
echo "work: $w"
