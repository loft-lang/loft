    loft_set_child_rec(loft_types, 101, t10, t11, 0);

b = '''    v2_cur = loft_new_child_rec(loft_types, 101, v3__slot_cur_1, loft_zero16, 16);
    loft_set_int_at(v2_cur, 0, v1_k, "/private/tmp/p219/hand/drv/tree_delete.loft", 40);'''
