# step 1 — the destination is the host's store: `t.left = insert_rec(t.left, k)` hands the
# callee a buffer naming t's STORE (no record yet); a record the callee builds there is ADOPTED by
# the field — its number written, nothing copied — and no store is minted or freed per insert.
import sys
src, dst = sys.argv[1], sys.argv[2]
s = open(src).read()
RT = '''
/* hand: a buffer naming a store and no record — the callee claims its result there. */
static loft_ref loft_store_buf(loft_ref host) { loft_ref b = { host.store_nr, 0, 0, 0 }; return b; }
/* hand: write the result of a call handed `loft_store_buf(host)` into the out-of-line field `f`:
 * a record the callee claimed in f's store is adopted by number; anything else is written as
 * `OpSetChildRec` writes it (its argument handed back: the field already says it). */
static void loft_adopt_child(const loft_tdesc *T, uint16_t tp, loft_ref f, loft_ref v) {
    if (v.store_nr == f.store_nr && v.rec != 0 && v.rec != loft_u32(loft_store_of(f), f.rec, f.pos)) {
        loft_set_u32(loft_store_of(f), f.rec, f.pos, v.rec);
    } else {
        loft_set_child_rec(T, tp, f, v, 0);
    }
}
'''
anchor = 'static void loft_vec_add_array('
s = s.replace(anchor, RT + anchor, 1)
reps = [
