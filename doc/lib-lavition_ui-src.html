// Text metrics for the panel — MEASURED, never assumed.
//
// This file used to say `text_width(s) = len(s) * 8` and expose a GLYPH_W of 8,
// waiting for "the loft_gl text bridge in Step 9".  The bridge landed on
// 2026-08-03, and `probe/b1` measured what it actually gives:
//
//     desktop  advance 19.27   MMMMMMMMMM 192.66 = iiiiiiiiii 192.66
//     browser  advance 26.66   MMMMMMMMMM 266.56 / iiiiiiiiii  71.09
//
// So `8` was wrong on both targets, and the two targets do not even agree with
// each other.  A panel laid out on a made-up advance mis-lays every row, and
// nothing reports it: the text still draws, it is just in the wrong place.
//
// ⚠ THE LIBRARY STAYS PURE.  These tests run under `loft test` with no window
// and no GL context, so nothing here may call `gl_measure_text`.  The consumer
// measures once at startup and hands the numbers over; this file does the
// arithmetic.  That is the whole seam.

// What the consumer measured.  `m_mono` is a FACT about the font that was
// resolved, not a request that was made — see `metrics_measured`.
pub struct Metrics {
  // ⚠ THE ADVANCE IS IN 1/64 PIXEL, and a whole-pixel one is what B1.6 caught.
  //
  // It was `m_advance: integer`. DejaVu Sans Mono at 16px advances 9.6px, which
  // truncates to 9 — and the error is per CHARACTER, so it accumulates: the
  // editor's own subject line measured **31px narrower** than the bridge
  // rasterises it. Under-estimating is the dangerous direction, because
  // `fit_text` then believes more fits than does and the text overflows the box
  // it was just proved to fit, silently.
  //
  // 1/64 is the unit fonts are hinted in, and it makes the residual under a
  // pixel over any line a panel can hold.
  m_adv64:   integer,       // advance in 1/64 px
  m_line_h:  integer,       // line height in pixels
  m_mono:    boolean,       // is it actually fixed-pitch?
}

// The advance in whole pixels, for a caller that wants to show it.
//
// ⚠ NOT A SECOND FIELD. One quantity stored twice is how the hotkey column got
// its collision — two numbers describing one distance drift apart the moment one
// is edited. This is derived.
pub fn metrics_advance(m: Metrics) -> integer { (m.m_adv64 + 32) / 64 ?? 0 }

// Build metrics from two runs the consumer measured through the bridge: `n`
// copies of a WIDE character and `n` copies of a NARROW one.
//
// ⚠ THE TWO RUNS ARE THE WHOLE INSTRUMENT, and one width cannot replace them.
// Ten M's are ten M's in any font, so a single measurement tells you the
// advance of an M and nothing about whether the font is fixed-pitch.  Two
// same-length, opposite-width strings answer it in one comparison — and getting
// this wrong is not hypothetical: asking for `DejaVuSansMono.ttf` gives a
// fixed-pitch font on the desktop and a PROPORTIONAL fallback in the browser,
// because the browser resolves the path's base name to a CSS family it does not
// know (`probe/b1`).
//
// `m_advance` is taken from the wide run: under fixed pitch the two agree, and
// when they do not, the wider one is the safer basis for a fits-in-the-box
// question — it over-estimates rather than overflowing.
pub fn metrics_measured(wide_run: float, narrow_run: float, n: integer,
                        line_h: integer) -> Metrics {
  if n <= 0 {
    return Metrics { m_adv64: 0, m_line_h: line_h, m_mono: false };
  }
  Metrics {
    m_adv64:  ((wide_run * 64.0 / (n as float)) ?? 0.0) as integer,
    m_line_h: line_h,
    m_mono:   wide_run == narrow_run,
  }
}

// Pixel width of a string at these metrics.
//
// Exact under fixed pitch, an OVER-estimate otherwise (see `metrics_measured`),
// which is the direction that keeps a label inside its box rather than spilling
// out of it.
// ⚠ ROUNDED UP. The residual is a fraction of a pixel per character, and the
// safe direction to spend it is outwards: an over-estimate reserves a pixel too
// many and the text fits, an under-estimate overflows a box that was measured as
// fitting. Same reason `metrics_measured` takes the WIDE run.
pub fn text_width(s: text, m: Metrics) -> integer {
  ((len(s) * m.m_adv64 + 63) / 64) ?? 0
}

pub fn text_height(m: Metrics) -> integer { m.m_line_h }

pub fn fits(s: text, px: integer, m: Metrics) -> boolean { text_width(s, m) <= px }

// The longest prefix of `s` that fits in `px`, with ".." marking the cut.
//
// Returns `s` untouched when it already fits, so a caller cannot tell "fitted"
// from "not measured" by looking for the marker — which is why the marker is
// only ever added when something was actually removed.
//
// ".." rather than a single "…": the ellipsis is one code point but several
// bytes, and `text_width` counts characters.  A marker whose own width is
// ambiguous is the wrong thing to measure a truncation with.
// The byte offset just past the `n`th character of `s`, or its end if it has
// fewer than `n`.
//
// ⚠ THIS EXISTS BECAUSE THE TWO UNITS DO NOT MEET. `len` counts characters and a
// slice bound is a byte offset, and loft's stdlib has nothing between them — so a
// cut computed in characters, which is what an advance gives you, has to be
// walked into an offset before it can be used. `c#next` is the byte just past the
// current character, which is exactly the cut point.
//
// ⚠ The failure it prevents is SILENT: loft snaps a byte cut outward to a
// character boundary, so a character count spent as bytes never corrupts the
// text — it just stops early, and the label comes back shorter than the box it
// was measured against. Every ASCII fixture agrees with the bug.
fn byte_after_chars(s: text, n: integer) -> integer {
  if n <= 0 { return 0; }
  bac_seen = 0;
  for bac_c in s {
    bac_seen += 1;
    if bac_seen >= n { return bac_c#next; }
  }
  size(s)
}

pub fn fit_text(s: text, px: integer, m: Metrics) -> text {
  if m.m_adv64 <= 0 { return s; }
  if text_width(s, m) <= px { return s; }
  // Room for the marker itself.  Below that there is no honest answer, so it
  // gives back as much as fits and no marker.
  adv = metrics_advance(m);
  if adv <= 0 { return s; }
  keep = ((px / adv) ?? 0) - 2;
  if keep <= 0 {
    room = (px / adv) ?? 0;
    if room <= 0 { return ""; }
    return s[0..byte_after_chars(s, room)];
  }
  s[0..byte_after_chars(s, keep)] + ".."
}

// The editor's side panel: layout, the spec that fills it, and hit-testing.
//
// A fixed-width strip on the left of the window. Toolbar at the top, scrollable
// list below, a status bar across the bottom of the window.
//
// ⚠ THIS FILE KNOWS NOTHING ABOUT WHAT IT IS SHOWING. It was `moros_ui`'s and
// took a `ToolState`, which is why it dragged `moros_sim`'s player physics and
// collision into a layout library and made the panel unusable by the browser
// client — the one program that needs one. What crosses the seam now is DATA:
// labels, indices and a status string the consumer already has.
//
// See doc/claude/EDITOR_UI.md §U2.
use self::widgets::*;
use self::font::*;

// ── Layout constants ─────────────────────────────────────────────
pub PANEL_WIDTH        = 240;
pub TOOLBAR_ROW_HEIGHT = 32;
pub TOOLBAR_GAP        = 4;
pub TOOLBAR_TOP        = 8;
pub SEPARATOR_HEIGHT   = 2;
pub LIST_ITEM_HEIGHT   = 20;
pub STATUS_HEIGHT      = 24;
// ⚠ THE SUBJECT LINE GETS ITS OWN BAR, ACROSS THE TOP, and the width is what
// decides that. CATALOGUE §C1 wants it top-left and always visible, and its own
// example — `world hollow-hill · AUTO · level ON · run armed · roof hidden ·
// 30 Hz` — is 69 characters, about 620px. It was never going to live in a 240px
// column; putting it there would truncate the answer to "what am I working on"
// down to the first two words.
pub SUBJECT_HEIGHT     = 24;

// Room a label does not get: the padding inside its box, and the column the
// hotkey glyph sits in.
pub LABEL_PAD  = 12;
pub HOTKEY_COL = 20;
// Where the hotkey starts inside its button, and therefore how much room it has:
// from HOTKEY_X to the label's start at HOTKEY_COL, less a gap so the two never
// touch. ⚠ DERIVED, not a second magic number — the first attempt fitted the
// hotkey to `HOTKEY_COL - 4` while drawing it at `+6`, which left it 2px longer
// than the gap and put it straight back under the label it was being moved out
// of. Two numbers describing one distance is how that happens.
pub HOTKEY_X   = 6;
pub HOTKEY_GAP = 2;
pub fn hotkey_room() -> integer { HOTKEY_COL - HOTKEY_X - HOTKEY_GAP }

// ── What a consumer hands over ───────────────────────────────────
//
// One entry per toolbar button. `sb_hotkey` is shown, not bound — binding keys
// is the consumer's, because only it knows what a key should do.
pub struct SpecButton {
  sb_label:  text,
  sb_hotkey: text,
}

// The whole contents of a panel, as plain data.
//
// ⚠ INDICES, NOT OBJECTS. `ps_selected_button` is an index into `ps_buttons`,
// not a `ToolKind`; `ps_selected_item` an index into `ps_items`. That is what
// lets Moros, crawler and a browser client each map their own state to the same
// panel — and it is also what keeps a click from meaning anything here (§U3).
pub struct PanelSpec {
  ps_buttons:         vector<SpecButton>,
  ps_selected_button: integer,
  ps_items:           vector<Entry>,
  ps_selected_item:   integer,
  ps_status:          text,
  // What you are working on and what is switched on — §C1's one line.
  ps_subject:         text,
}

pub fn spec_button(label: text, hotkey: text) -> SpecButton {
  SpecButton { sb_label: label, sb_hotkey: hotkey }
}

// ── Panel struct ─────────────────────────────────────────────────
pub struct Panel {
  p_rect:    UiRect,
  p_toolbar: vector<Button>,
  p_list:    ListBox,
  p_status:  StatusStrip,
  p_subject: StatusStrip,
  // ⚠ THE METRICS IT WAS BUILT WITH, carried rather than passed again.
  // `panel_build` FITS every label to its box with these; `panel_text_list`
  // PLACES the same labels. If the two could be handed different metrics, a
  // panel could be fitted at one advance and drawn at another — text that
  // overflows the box it was just proved to fit in, and nothing to catch it.
  // Carrying them makes that unrepresentable rather than merely unlikely.
  p_metrics: Metrics,
}

// ── Layout helpers ───────────────────────────────────────────────

// The subject bar: full width, at the very top, never covered by anything.
pub fn subject_rect(window_w: integer) -> UiRect {
  rect(0, 0, window_w, SUBJECT_HEIGHT)
}

// ⚠ THE STRIP STARTS BELOW THE SUBJECT BAR rather than under it. An always-visible
// line that something else is drawn over is not always visible, and "never
// hidden" is the one property §C1 asks of it by name.
pub fn panel_rect(window_w: integer, window_h: integer) -> UiRect {
  // window_w is present so callers do not have to special-case the single fixed
  // strip; if the panel grows a right-aligned variant, the dimension is here.
  _ = window_w;
  rect(0, SUBJECT_HEIGHT, PANEL_WIDTH, window_h - SUBJECT_HEIGHT)
}

pub fn toolbar_button_rect(idx: integer) -> UiRect {
  tbr_y = SUBJECT_HEIGHT + TOOLBAR_TOP + idx * (TOOLBAR_ROW_HEIGHT + TOOLBAR_GAP);
  rect(8, tbr_y, PANEL_WIDTH - 16, TOOLBAR_ROW_HEIGHT)
}

// ⚠ THE BUTTON COUNT IS A PARAMETER, not a constant. It was `TOOLBAR_BUTTONS =
// 6`, which is Moros's tool count — a piece of one consumer's content baked into
// the geometry of every consumer's panel. The list starts below however many
// buttons it was given.
pub fn list_rect(window_h: integer, n_buttons: integer) -> UiRect {
  lr_top = SUBJECT_HEIGHT + TOOLBAR_TOP
    + n_buttons * (TOOLBAR_ROW_HEIGHT + TOOLBAR_GAP)
    + SEPARATOR_HEIGHT + 8;
  lr_bottom = window_h - STATUS_HEIGHT - 4;
  rect(8, lr_top, PANEL_WIDTH - 16, lr_bottom - lr_top)
}

// ⚠ THE STATUS BAR SPANS THE WINDOW, and it did not until the metrics were made
// load-bearing. It was `PANEL_WIDTH` — 240 px — while the text it is handed runs
// to 80-odd characters, about 648 px. It overflowed its own box by 2.7x from the
// day it was written, and nothing reported it because nothing measured: the
// renderer was never built, so the text never met its rectangle.
pub fn status_rect(window_w: integer, window_h: integer) -> UiRect {
  rect(0, window_h - STATUS_HEIGHT, window_w, STATUS_HEIGHT)
}

// ── Panel build ──────────────────────────────────────────────────

// ⚠ EVERY LABEL IS FITTED TO ITS BOX HERE, not left for the renderer to overflow.
// How many characters fit is a question about the FONT, which this library
// cannot know — so the consumer measures once and hands over `Metrics`, and the
// fit becomes a pure testable fact about the panel rather than something a
// renderer discovers by drawing past the edge.
pub fn panel_build(spec: PanelSpec, window_w: integer, window_h: integer,
                   metrics: Metrics) -> Panel {
  pb_buttons: vector<Button> = [];
  pb_i = 0;
  for pb_sb in spec.ps_buttons {
    pb_br = toolbar_button_rect(pb_i);
    pb_buttons += [Button {
      btn_id:           pb_i,
      btn_rect:         pb_br,
      btn_label:        fit_text(pb_sb.sb_label, pb_br.r_w - LABEL_PAD - HOTKEY_COL, metrics),
      // ⚠ THE HOTKEY IS FITTED TOO, and it was not until a picture showed why.
      // The label was measured against its box and the hotkey placed beside it
      // UNMEASURED, so a one-character hint fitted and `drag` ran straight under
      // the label: `d` and `Look` overlapping in the first button, legible as
      // neither. A string placed in a box nobody measured is the same class
      // `B1.2` was about, one column over.
      btn_hotkey_label: fit_text(pb_sb.sb_hotkey, hotkey_room(), metrics),
      btn_selected:     pb_i == spec.ps_selected_button,
    }];
    pb_i = pb_i + 1;
  }

  pb_lr = list_rect(window_h, len(spec.ps_buttons));
  pb_items: vector<Entry> = [];
  for pb_it in spec.ps_items {
    // ⚠ THE REASON IS FITTED WITH THE NAME, as one row, because that is what the
    // author reads. At 224px a long reason truncates — `fit_text` marks the cut —
    // and the FULL reason stays on the entry for a consumer with somewhere
    // roomier to put it. A row that showed only the name would grey a thing out
    // and not say why, which is the blank no again.
    pb_txt = pb_it.en_label;
    if !pb_it.en_available { pb_txt = "{pb_it.en_label} — {pb_it.en_reason}"; }
    // ⚠ THE KIND IS CARRIED, AND IT WAS NOT. `B5.1` put `en_kind` on `Entry` so
    // ONE list could hold two families, and this rebuild left it out — so every
    // row in `lb_items` came out `KIND_UNSET` and a renderer reading the built
    // panel could not tell a part from a material. Nothing noticed, because no
    // consumer read the kind until `B5.2` drew an image per row: the field
    // reached the struct, the wire and the parser, and stopped one line short of
    // its only reader. The LABEL is fitted here and the kind is not — a kind is
    // a handle, not something shown, so there is no box to fit it to.
    pb_items += [Entry { en_kind:      pb_it.en_kind,
                         en_label:     fit_text(pb_txt, pb_lr.r_w - LABEL_PAD, metrics),
                         en_available: pb_it.en_available,
                         en_reason:    pb_it.en_reason }];
  }

  pb_sr = status_rect(window_w, window_h);
  pb_qr = subject_rect(window_w);

  Panel {
    p_metrics: metrics,
    p_rect:    panel_rect(window_w, window_h),
    p_toolbar: pb_buttons,
    p_list: ListBox {
      lb_rect:        pb_lr,
      lb_items:       pb_items,
      lb_selected:    spec.ps_selected_item,
      lb_scroll:      0,
      lb_item_height: LIST_ITEM_HEIGHT,
    },
    p_status: StatusStrip {
      ss_rect: pb_sr,
      ss_text: fit_text(spec.ps_status, pb_sr.r_w - LABEL_PAD, metrics),
    },
    p_subject: StatusStrip {
      ss_rect: pb_qr,
      ss_text: fit_text(spec.ps_subject, pb_qr.r_w - LABEL_PAD, metrics),
    },
  }
}

// ── Hit-test ─────────────────────────────────────────────────────
//
// ⚠ A CLICK RETURNS A HIT; IT DOES NOT MEAN ANYTHING.  There used to be a
// `route_click` beside this that also mutated the caller's tool state. Only the
// consumer knows what a click on item 3 means — and for a client talking to a
// server the answer is usually "send a message and wait to be told", not "set a
// local field" (CATALOGUE.md §C1). With the mutation gone, `route_click` was
// this function under another name, so it is not here.
pub enum UiHit {
  UhWorld,
  UhToolButton { tb_id: integer },
  UhListItem   { li_idx: integer },
  UhNone,
}

// Where row `idx` sits on screen — the inverse of `panel_hit_test`'s list arm.
//
// ⚠ IT EXISTS SO A CONSUMER NEED NOT RE-DERIVE THE LAYOUT. Drawing a row,
// clicking one, and driving one from a test all want the same rectangle, and the
// only other way to have it is to write `lb_rect.r_y + idx * lb_item_height`
// out again at each site — a second copy of the layout, which stays green right
// up until the layout moves. `verbbar` has answered this since it shipped, with
// a `vt_rect` on every slot; the list did not, and the list is the half a
// catalogue is clicked in.
//
// ⚠ THE RECTANGLE COMES BACK EVEN WHEN THE ROW IS SCROLLED OUT OF SIGHT.
// Clamping it here would answer a different question — *is this row visible* —
// and a caller that wants that one asks `rect_contains(p.p_list.lb_rect, …)` of
// the centre. An index with no row at all is the one case with no answer, and it
// gets an empty rectangle rather than a plausible one.
pub fn list_row_rect(p: Panel, idx: integer) -> UiRect {
  if idx < 0 || idx >= len(p.p_list.lb_items) { return rect(0, 0, 0, 0); }
  rect(p.p_list.lb_rect.r_x,
       p.p_list.lb_rect.r_y + idx * p.p_list.lb_item_height - p.p_list.lb_scroll,
       p.p_list.lb_rect.r_w,
       p.p_list.lb_item_height)
}

pub fn panel_hit_test(p: Panel, mx: integer, my: integer) -> UiHit {
  // Outside the panel's x-range → the world behind it.
  if mx >= p.p_rect.r_x + p.p_rect.r_w {
    return UhWorld;
  }
  for pht_b in p.p_toolbar {
    if rect_contains(pht_b.btn_rect, mx, my) {
      return UhToolButton { tb_id: pht_b.btn_id };
    }
  }
  if rect_contains(p.p_list.lb_rect, mx, my) {
    pht_local_y = my - p.p_list.lb_rect.r_y + p.p_list.lb_scroll;
    // A zero item height is a mis-built panel, not a hit: -1 falls outside the
    // range check below, so it reports "no item" rather than a null index.
    pht_idx = (pht_local_y / p.p_list.lb_item_height) ?? -1;
    if pht_idx >= 0 && pht_idx < len(p.p_list.lb_items) {
      return UhListItem { li_idx: pht_idx };
    }
  }
  UhNone
}

// What the panel looks like — as a LIST OF RECTANGLES, not as GL calls.
//
// ⚠ WHY A LIST AND NOT A `panel_render(painter, …)`. Taking a `Painter2D` would
// pull `graphics` into this package, and then its tests would need a window: the
// layout suite runs under `loft test` with no GL context at all, which is the
// property that made the whole re-homing worth doing.
//
// It also moves a real claim out of the picture and into a test. "The selected

// The panel as rectangles, back to front.
//
// ⚠ THE ORDER IS THE Z-ORDER, and the consumer draws it as given. Emitting the
// backgrounds after the things standing on them would paint over them, and that
// is a bug a picture shows and a count does not — so `test_the_backgrounds_come
// _first` pins it here instead.
pub fn panel_draw_list(p: Panel, th: Theme) -> vector<DrawRect> {
  dl: vector<DrawRect> = [];

  // the strip itself, then the two wells that sit on it
  dl += [DrawRect { dr_rect: p.p_rect,           dr_colour: th.th_panel_bg }];
  dl += [DrawRect { dr_rect: p.p_list.lb_rect,   dr_colour: th.th_list_bg }];
  dl += [DrawRect { dr_rect: p.p_status.ss_rect, dr_colour: th.th_status_bg }];
  // ⚠ THE SUBJECT BAR IS EMITTED WITH THE BACKGROUNDS, not after them, and that
  // is only safe because nothing overlaps it: the strip starts below it. If it
  // ever floated over the panel this order would paint the panel on top of the
  // one line §C1 calls "never hidden".
  dl += [DrawRect { dr_rect: p.p_subject.ss_rect, dr_colour: th.th_subject_bg }];

  // a hairline above the list, so the toolbar reads as its own group
  lr = p.p_list.lb_rect;
  dl += [DrawRect {
    dr_rect: rect(lr.r_x, lr.r_y - SEPARATOR_HEIGHT - 4, lr.r_w, SEPARATOR_HEIGHT),
    dr_colour: th.th_separator,
  }];

  for b in p.p_toolbar {
    dl += [DrawRect {
      dr_rect:   b.btn_rect,
      dr_colour: if b.btn_selected { th.th_button_sel } else { th.th_button },
    }];
  }

  // the selected row, if there is one and it is in view
  if p.p_list.lb_selected >= 0 && p.p_list.lb_selected < len(p.p_list.lb_items) {
    row_y = lr.r_y + p.p_list.lb_selected * p.p_list.lb_item_height - p.p_list.lb_scroll;
    // ⚠ Clipped to the well. A selection scrolled out of view would otherwise be
    // drawn ABOVE the list, over the toolbar — the failure mode of every
    // hand-rolled list, and invisible until someone scrolls.
    if row_y >= lr.r_y && row_y + p.p_list.lb_item_height <= lr.r_y + lr.r_h {
      dl += [DrawRect {
        dr_rect:   rect(lr.r_x, row_y, lr.r_w, p.p_list.lb_item_height),
        dr_colour: th.th_item_sel,
      }];
    }
  }

  dl
}


// ── the words ────────────────────────────────────────────────────
//
// A second list, for the same reason the first one is a list: turning a string
// into pixels needs a font and a GL texture, and neither belongs in a package
// whose tests run without a window.
//
// ⚠ EVERY STRING HERE IS ALREADY FITTED. `panel_build` truncated each label to
// its box, and this places what came out of that — so a caller cannot draw a
// label the panel never agreed would fit. That is why `Panel` carries its own
// `p_metrics` rather than taking them again here.
pub struct DrawText {
  dt_text:   text,
  dt_x:      integer,
  dt_y:      integer,          // TOP of the line, which is what draw-a-texture wants
  dt_colour: integer,
}

// Centre a line of text vertically in a box of height `h`.
fn baseline_in(box_y: integer, box_h: integer, line_h: integer) -> integer {
  box_y + (box_h - line_h) / 2 ?? box_y
}

pub fn panel_text_list(p: Panel, th: Theme) -> vector<DrawText> {
  tl: vector<DrawText> = [];
  lh = p.p_metrics.m_line_h;

  for b in p.p_toolbar {
    r = b.btn_rect;
    y = baseline_in(r.r_y, r.r_h, lh);
    // the hotkey sits in its own column at the left, dimmer than the label
    tl += [DrawText { dt_text: b.btn_hotkey_label,
                      dt_x: r.r_x + HOTKEY_X, dt_y: y, dt_colour: th.th_text_dim }];
    tl += [DrawText { dt_text: b.btn_label,
                      dt_x: r.r_x + HOTKEY_COL, dt_y: y, dt_colour: th.th_text }];
  }

  lr = p.p_list.lb_rect;
  i = 0;
  for it in p.p_list.lb_items {
    row_y = lr.r_y + i * p.p_list.lb_item_height - p.p_list.lb_scroll;
    // ⚠ Clipped to the well, exactly as the selection highlight is. A row
    // scrolled out of view would otherwise be drawn over the toolbar, and text
    // floating above a list is the tell that nobody scrolled it while looking.
    if row_y >= lr.r_y && row_y + p.p_list.lb_item_height <= lr.r_y + lr.r_h {
      // ⚠ GREYED, NOT HIDDEN. The dim colour is the at-a-glance signal and the
      // reason (already folded into the row's text by `panel_build`) is the
      // sentence — §C3 wants both, because a grey row with no words says only
      // that something is wrong.
      tl += [DrawText { dt_text: it.en_label,
                        dt_x: lr.r_x + 6,
                        dt_y: baseline_in(row_y, p.p_list.lb_item_height, lh),
                        dt_colour: if it.en_available { th.th_text } else { th.th_text_dim } }];
    }
    i += 1;
  }

  sr = p.p_status.ss_rect;
  tl += [DrawText { dt_text: p.p_status.ss_text,
                    dt_x: sr.r_x + 6,
                    dt_y: baseline_in(sr.r_y, sr.r_h, lh),
                    dt_colour: th.th_text }];

  qr = p.p_subject.ss_rect;
  tl += [DrawText { dt_text: p.p_subject.ss_text,
                    dt_x: qr.r_x + 6,
                    dt_y: baseline_in(qr.r_y, qr.r_h, lh),
                    dt_colour: th.th_text }];
  tl
}

// Copyright (c) 2026 Jurjen Stellingwerff
// SPDX-License-Identifier: LGPL-3.0-or-later
//
// THE VERB BAR — what you can press, and what presses it, in one strip.
//
// A row of slots along the bottom of the window. Each slot is a VERB the editor can
// do here, drawn with the KEY the person has bound to it. It is the picture of the
// definition layer ([EDITING_MODES § Phase 6](../../../doc/claude/EDITING_MODES.md)),
// and it is a **view** of that table rather than a second place the binding lives.
//
// ⛔ **WHY IT EXISTS: THE THING IT REPLACES WAS WRONG ON SCREEN.** The side panel's
// toolbar drew a hotkey glyph per button as a literal — `e` beside *Stencil* where `e`
// is `stair_up`, `c` beside *Cart* where `c` is `cellar`, `f` beside *Field* where `f`
// is `fence`. Three of six, for months, because a literal glyph is connected to
// nothing that could disagree with it. Everything below takes its glyph from the
// caller's map, so a rebind moves the label with the binding.
//
// ── ⚠ THE TWO DESIGN CALLS, because both could reasonably have gone the other way ──
//
// **1. Slots are NOT numbered.** There is no `1..9` convention: a slot shows whatever
// key its verb is bound to, and an unbound verb shows a blank. World of Warcraft's
// numbers are a consequence of its bars being PAGES of fixed size; here a verb has a
// letter because a letter is where an author's hand already is. ⚠ Numbering would also
// quietly make the bar a second binding surface — *slot 3 is verb X* — which is the
// duplication the whole phase exists to remove.
//
// **2. One row, and overflow is REPORTED rather than wrapped.** A bar that grows a
// second row is a bar that eats the window it is drawn over, and the ask was for
// something that *doesn't take up too much screen space*. So it keeps what fits and
// says how many it could not show — `vb_hidden`, which a consumer draws as `+3`.
// ⏭ The real answer to overflow is the MODE (`D1`): the verbs available where you are
// standing are a handful, not the whole vocabulary, and this bar shrinks for free the
// day that query narrows.
//
// ⚠ **AND THIS FILE KNOWS NOTHING ABOUT VERBS**, exactly as `panel.loft` knows nothing
// about what it is showing. It takes labels, key glyphs and an armed flag; the words
// `opening` and `cellar` never appear here. That is what lets a different editor with
// a different vocabulary use it unchanged.
use self::widgets::*;
use self::panel::*;
use self::font::*;
// `DrawRect`, `DrawText` and `Theme` — the picture as lists, which is `render.loft`'s
// shape and this file's for its reason: no GL here, so the tests need no window.
use self::render::*;

// ── Layout constants ─────────────────────────────────────────────
//
// ⚠ **SMALL ON PURPOSE, AND THE WIDTH IS A MEASUREMENT RATHER THAN A TASTE.** The
// content is a 4-character short label under a key glyph; the editor's font measures
// **10px per character, mono** (`client: panel … advance 10`), so the label is 40px and
// the slot is that plus 3px each side. Anything bigger is a toolbar, and a toolbar is
// what this replaces.
//
// ⛔ **IT WAS 34, WHICH IS THE WIDTH A DIFFERENT FONT WANTED.** Guessed at an 8px
// advance and shipped: the first real run drew every label truncated to one character
// plus an ellipsis — `fenc` as `f…` — while the layout suite stayed green, because the
// only claim it made was *the text is inside its box* and truncated text always is.
// `test_a_four_character_label_survives_the_editors_own_metrics` is the row that says
// the slot is wide enough for the content it exists to hold.
pub VERB_SLOT_W   = 46;
// ⚠ **TWO LINES, AND THE HEIGHT IS THE SECOND MEASUREMENT THIS WIDGET GOT WRONG BY
// GUESSING.** The editor draws at `UI_SIZE` 16.0, which measures ~16px per line; a slot
// holding a key over a label needs both plus a little air. ⛔ It was 30 — enough for
// one line and four pixels — so the first real run drew `Up` and `rise` ON TOP OF EACH
// OTHER, while the layout suite stayed green because its only claim was that the key's
// y is LESS than the label's. Ordering is not separation.
// `test_the_two_lines_of_a_slot_do_not_overlap_at_the_editors_metrics` is the row that
// says so now.
pub VERB_SLOT_H   = 40;
pub VERB_SLOT_GAP = 3;
// The margin between the bar and the bottom of the window, and between the bar and
// the panel on its left.
pub VERB_BAR_PAD  = 6;
// ⚠ THE BAR'S HEIGHT IS THE SLOT'S, and it is a constant rather than a sum, because a
// consumer needs to reserve the strip before it knows how many slots there are.
pub fn verbbar_height() -> integer { VERB_SLOT_H + VERB_BAR_PAD * 2 }

// ── What a consumer hands over ───────────────────────────────────
//
// ⚠ `sv_verb` IS CARRIED THROUGH UNTOUCHED AND NEVER DRAWN. It is what a click means —
// *rebind THIS verb* — and a consumer that had to map a slot index back to a verb
// would be keeping a second list in step with this one. `panel.loft` uses indices for
// its list because a catalogue row has no name that outlives the list; a verb does.
pub struct SpecVerb {
  sv_verb:  text,        // the published name — what a rebind names
  sv_label: text,        // the short label the slot draws, already fitted by the caller
  sv_key:   text,        // the key glyph, "" when the verb is bound to nothing
  sv_on:    boolean,     // armed / mid-gesture — a run with one end down, say
}

pub fn spec_verb(verb: text, label: text, key: text) -> SpecVerb {
  SpecVerb { sv_verb: verb, sv_label: label, sv_key: key, sv_on: false }
}

pub fn spec_verb_on(verb: text, label: text, key: text, on: boolean) -> SpecVerb {
  SpecVerb { sv_verb: verb, sv_label: label, sv_key: key, sv_on: on }
}

// ── The laid-out bar ─────────────────────────────────────────────
pub struct VerbSlot {
  vt_rect:  UiRect,
  vt_verb:  text,
  vt_label: text,
  vt_key:   text,
  vt_on:    boolean,
}

pub struct VerbBar {
  vb_rect:   UiRect,
  vb_slots:  vector<VerbSlot>,
  // How many the strip could not show. ⚠ **REPORTED, NEVER SILENTLY DROPPED** — a bar
  // that truncated quietly would read as *those verbs do not exist here*, which is
  // exactly the wrong sentence to put in front of an author who cannot find one.
  vb_hidden: integer,
}

// Lay the bar out along the bottom of the window, to the right of the side panel.
//
// ⚠ **IT TAKES THE PANEL'S WIDTH RATHER THAN ASSUMING IT.** A consumer with no side
// panel passes 0 and gets a full-width bar; this package must not decide that an
// editor has a panel at all.
//
// ⚠ **AND THE STATUS STRIP IS ABOVE IT, NOT BELOW** — `panel.loft` puts the status
// across the bottom of the WINDOW, so the bar sits clear of it. Two things claiming
// the last 24 rows of pixels is the overlap `panel_draw_list`'s own comment warns
// about for the subject bar, one edge over.
// ⚠ **IT FITS ITS OWN TEXT, WHICH IS `panel_build`'s CONTRACT ONE WIDGET OVER.** That
// file's `panel_text_list` says *"every string here is already fitted … so a caller
// cannot draw a label the panel never agreed would fit"*, and the same has to hold
// here or a long key glyph runs out of its slot and across its neighbour. ⚠ It is a
// BACKSTOP and not the answer: two names sharing a prefix truncate to the same thing,
// so a caller that wants them distinguishable shortens them itself (`key_glyph`).
pub fn verbbar_build(specs: vector<SpecVerb>, win_w: integer, win_h: integer,
                     panel_w: integer, m: Metrics) -> VerbBar {
  x0 = panel_w + VERB_BAR_PAD;
  avail = win_w - x0 - VERB_BAR_PAD;
  bar_h = verbbar_height();
  y0 = win_h - STATUS_HEIGHT - bar_h;
  bar = rect(x0, y0, if avail > 0 { avail } else { 0 }, bar_h);

  // How many fit: n slots and n-1 gaps.
  //
  // ⚠ **THE ARITHMETIC IS WRITTEN AS A LOOP RATHER THAN A DIVISION**, because the
  // closed form is off by one at the boundary in the direction that OVERFLOWS — a
  // slot half outside the bar is the failure a division silently produces, and the
  // rows in `tests/verbbar.loft` that assert containment are what found it.
  slots: vector<VerbSlot> = [];
  hidden = 0;
  cx = x0 + VERB_BAR_PAD;
  for s in specs {
    right = cx + VERB_SLOT_W;
    if right <= x0 + bar.r_w - VERB_BAR_PAD {
      slots += [VerbSlot {
        vt_rect:  rect(cx, y0 + VERB_BAR_PAD, VERB_SLOT_W, VERB_SLOT_H),
        vt_verb:  s.sv_verb,
        vt_label: fit_text(s.sv_label, VERB_SLOT_W - 2, m),
        vt_key:   fit_text(s.sv_key, VERB_SLOT_W - 2, m),
        vt_on:    s.sv_on,
      }];
      cx = right + VERB_SLOT_GAP;
    } else {
      hidden = hidden + 1;
    }
  }
  VerbBar { vb_rect: bar, vb_slots: slots, vb_hidden: hidden }
}

// Which slot is under the pointer, or -1.
//
// ⚠ **IT ANSWERS AN INDEX AND THE CALLER READS THE VERB OFF IT**, rather than
// answering the verb directly: -1 has to mean *nothing*, and a text return would have
// to spell that as `""`, which is also a legal verb name in the caller's world (it is
// `VB_NONE`). An integer cannot be confused with an answer.
pub fn verbbar_hit(b: VerbBar, mx: integer, my: integer) -> integer {
  i = 0;
  for s in b.vb_slots {
    if rect_contains(s.vt_rect, mx, my) { return i; }
    i += 1;
  }
  0 - 1
}

// The verb a slot names, or "" when the index is not a slot.
pub fn verbbar_verb(b: VerbBar, i: integer) -> text {
  if i < 0 || i >= len(b.vb_slots) { return ""; }
  (b.vb_slots[i] ?? VerbSlot {}).vt_verb
}

// ── The picture, as lists ────────────────────────────────────────
//
// Same shape and same reason as `panel_draw_list`: rectangles and strings, so this
// package needs no GL and its tests need no window.
pub fn verbbar_draw_list(b: VerbBar, th: Theme) -> vector<DrawRect> {
  dl: vector<DrawRect> = [];
  dl += [DrawRect { dr_rect: b.vb_rect, dr_colour: th.th_panel_bg }];
  for s in b.vb_slots {
    dl += [DrawRect {
      dr_rect:   s.vt_rect,
      dr_colour: if s.vt_on { th.th_button_sel } else { th.th_button },
    }];
  }
  dl
}

// ⚠ **TWO LINES PER SLOT, KEY OVER LABEL, AND THE ORDER IS THE POINT.** The question a
// bar answers at a glance is *which key does this*, so the key is the top line and the
// label reads under it. Putting the label first makes the bar a list of features and
// the key an annotation, which is the toolbar it replaces.
//
// ⚠ **AN UNBOUND VERB DRAWS A BLANK TOP LINE RATHER THAN BEING HIDDEN.** The slot says
// *this exists and you have no key for it*, which is the only honest reading after a
// rebind displaces something — `keymap_bind` leaves the displaced verb in the list
// precisely so this slot can appear.
pub fn verbbar_text_list(b: VerbBar, th: Theme, m: Metrics) -> vector<DrawText> {
  tl: vector<DrawText> = [];
  for s in b.vb_slots {
    r = s.vt_rect;
    // ⚠ **BOTH LINES ARE PLACED FROM THE METRICS, NOT FROM TWO MAGIC OFFSETS.** The
    // pair has to be centred as a BLOCK — a key pinned to the top and a label pinned
    // to the bottom overlap the moment the font is taller than the slot expected,
    // which is exactly how the first version drew `Up` through `rise`.
    pad = (r.r_h - m.m_line_h * 2) / 2 ?? 0;
    if pad < 0 { pad = 0; }
    kw = text_width(s.vt_key, m);
    tl += [DrawText {
      dt_text:   s.vt_key,
      dt_x:      r.r_x + (r.r_w - kw) / 2 ?? r.r_x,
      dt_y:      r.r_y + pad,
      dt_colour: th.th_text,
    }];
    // …and the short label directly under it, dimmer, because the key is the answer.
    lw = text_width(s.vt_label, m);
    tl += [DrawText {
      dt_text:   s.vt_label,
      dt_x:      r.r_x + (r.r_w - lw) / 2 ?? r.r_x,
      dt_y:      r.r_y + pad + m.m_line_h,
      dt_colour: th.th_text_dim,
    }];
  }
  // ⚠ THE OVERFLOW COUNT IS DRAWN, which is what makes `vb_hidden` a report rather
  // than a field nobody reads. It sits after the last slot, in the dim colour, so it
  // reads as *there is more* and not as another verb.
  if b.vb_hidden > 0 {
    lastx = b.vb_rect.r_x + VERB_BAR_PAD;
    for s2 in b.vb_slots { lastx = s2.vt_rect.r_x + s2.vt_rect.r_w + VERB_SLOT_GAP; }
    tl += [DrawText {
      dt_text:   "+{b.vb_hidden}",
      dt_x:      lastx + 2,
      dt_y:      b.vb_rect.r_y + VERB_BAR_PAD + (VERB_SLOT_H - m.m_line_h) / 2 ?? 0,
      dt_colour: th.th_text_dim,
    }];
  }
  tl
}

// Pure data + hit-test helpers; no rendering (that lives in

// ⚠ `UiRect`, NOT `Rect` — plan 19 `L6`. The registry's `shapes` publishes a `Rect`
// of four FLOATS; this one is four integer pixels, and a UI that snapped to a float
// rectangle would be the whole reason this package has its own. Nothing imports both
// today, which is exactly when the rename is still free: `lavition_ui` is published at
// `L7`, and a published name cannot be renamed. `tools/names.sh` found it.
pub struct UiRect {
  r_x: integer,
  r_y: integer,
  r_w: integer,
  r_h: integer,
}

pub fn rect(x: integer, y: integer, w: integer, h: integer) -> UiRect {
  UiRect { r_x: x, r_y: y, r_w: w, r_h: h }
}

pub fn rect_contains(r: UiRect, mx: integer, my: integer) -> boolean {
  mx >= r.r_x && mx < r.r_x + r.r_w && my >= r.r_y && my < r.r_y + r.r_h
}

// A clickable button in the toolbar.  `btn_id` is the integer
// identity the panel uses in hit-test results (0..5 = ToolKind).
pub struct Button {
  btn_id:           integer,
  btn_rect:         UiRect,
  btn_label:        text,
  btn_hotkey_label: text,      // e.g. "1" shown left of the label
  btn_selected:     boolean,
}

// One row of a catalogue: what it is called, and whether you can have it.
//
// ⚠ AVAILABILITY IS PART OF THE ENTRY, NOT A SEPARATE DIALOG (CATALOGUE §C3).
// A thing you cannot use is shown GREYED WITH ITS REASON rather than hidden:
// hiding it makes the author think it is missing, while showing it with
// `too wide for frame/2x3` tells them what to change. It is §C2's refusal rule —
// reason, and never a blank no — applied to a list instead of to a gesture.
// ⚠ AND IT CARRIES ITS KIND (§C3). A catalogue entry is *kind · name · image ·
// availability*, and ONE widget serves both families — a material and a part are
// each *a named thing you pick and then place*. The kind is what lets a consumer
// group them, filter them, or send the right message when one is chosen, without
// a second list existing to be kept in step with the first.
//
// It is TEXT rather than an enum because `lavition_ui` must not know what families
// exist: a layout library that enumerated `material | part` would have to be
// edited before a consumer could invent a third.
pub const KIND_UNSET = "";

pub struct Entry {
  en_kind:      text,        // "material", "part", … — the consumer's vocabulary
  en_label:     text,
  en_available: boolean,
  en_reason:    text,        // "" when available; required when not
}

pub fn entry(label: text) -> Entry {
  Entry { en_kind: KIND_UNSET, en_label: label, en_available: true, en_reason: "" }
}

pub fn entry_of(kind: text, label: text) -> Entry {
  Entry { en_kind: kind, en_label: label, en_available: true, en_reason: "" }
}

// ⚠ THE REASON IS NOT OPTIONAL HERE. An entry that is unavailable "because" is
// the blank no §C2 exists to forbid, so a missing reason becomes a visible
// placeholder rather than an empty string nobody notices.
pub fn entry_blocked(label: text, why: text) -> Entry {
  return entry_blocked_of(KIND_UNSET, label, why);
}

pub fn entry_blocked_of(kind: text, label: text, why: text) -> Entry {
  Entry { en_kind: kind, en_label: label, en_available: false,
          en_reason: if why == "" { "no reason given" } else { why } }
}

fn after_differs(a: integer, b: integer) -> boolean { a != b }

// ── The row rectangle, against the hit-test that has to agree with it ────────
//
// ⚠ THE ROUND TRIP IS THE CLAIM, not the arithmetic. `list_row_rect` and
// `panel_hit_test` are one layout read in two directions, and the only way they
// can be wrong is by disagreeing — so the test asks the hit-test where the
// rectangle's own centre lands and requires the row back. Checking the numbers
// against numbers would be the table checked against the table.
fn row_centre_hits_its_own_row(p: Panel, why: text) {
  seen = 0;
  for i in 0..len(p.p_list.lb_items) {
    r = list_row_rect(p, i);
    cx = r.r_x + r.r_w / 2 ?? 0;
    cy = r.r_y + r.r_h / 2 ?? 0;
    // ⚠ A ROW SCROLLED OUT OF THE STRIP HAS A POSITION AND NO HIT, which is the
    // documented split between these two and not a disagreement: `list_row_rect`
    // answers *where is row i*, `panel_hit_test` answers *what is under this
    // pixel*, and a pixel outside the list is under nothing. Asking the hit-test
    // about it would be asking it the other function's question. ⚠ This `continue`
    // is why the count below exists — a filter that quietly removed every row
    // would leave a green test that examined nothing.
    if !rect_contains(p.p_list.lb_rect, cx, cy) { continue; }
    seen += 1;
    hit = panel_hit_test(p, cx, cy);
    assert(is_item(hit), "{why}: row {i}'s own centre ({cx},{cy}) is inside the "
                       + "list and is not an item at all");
    if hit is UhListItem { li_idx } {
      assert(li_idx == i, "{why}: row {i}'s own centre hit-tests as row {li_idx}");
    }
  }
  assert(seen > 0, "{why}: no row's centre landed inside the list — nothing was tested");
}

fn test_every_rows_centre_hit_tests_back_to_that_row() {
  row_centre_hits_its_own_row(six_button_panel(), "unscrolled");
}

// ⚠ SCROLLED IS THE CASE THAT SEPARATES THE TWO. Unscrolled, a row rectangle

fn test_the_row_rectangle_follows_the_scroll() {
  p = six_button_panel();

}

// The metrics seam — plan 18, `B1.2`, re-homed by `B1.2b`.
//
// `font.loft` used to claim `text_width(s) = len(s) * 8`. `probe/b1` measured
// what the bridge actually gives and the 8 was wrong on BOTH targets: 19.27 on
// the desktop, 26.66 in the browser, and the browser's font was not even fixed
// pitch. A panel laid out on a made-up advance mis-lays every row and nothing
// reports it, because the text still draws.
//
// So the library does arithmetic on numbers the CONSUMER measured, and these
// tests pin the arithmetic. They run with no window and no GL context, which is
// the constraint that forced the seam in the first place.
use lavition_ui::*;

fn mono(advance: integer) -> Metrics {
  run = (advance * 10) as float;
  metrics_measured(run, run, 10, advance)
}

// ── the fixed-pitch instrument ───────────────────────────────────

// ⚠ THE CLAIM `len * advance` IS EXACT RESTS ENTIRELY ON THIS. Two runs of the
// SAME LENGTH and OPPOSITE WIDTH is the whole instrument; one width cannot
// answer it, because ten M's are ten M's in any font.
fn test_two_equal_runs_are_fixed_pitch() {
  m = metrics_measured(192.65625, 192.65625, 10, 38);
  assert(m.m_mono, "equal M-run and i-run must read as fixed pitch");
  assert(metrics_advance(m) == 19, "advance from the wide run: {metrics_advance(m)}, want 19");
  // and the fine unit it is actually kept in: 19.265625 * 64 = 1233
  assert(m.m_adv64 == 1233, "adv64 is {m.m_adv64}, want 1233 (19.265625 px)");
}

// ⚠ THE CONTROL, AND IT IS A REAL MEASUREMENT — these are the numbers
// `probe/b1` read out of headless Chrome when the browser resolved
// `DejaVuSansMono.ttf` to a proportional fallback. If this test ever passes as
// mono, the check that protects every layout below has stopped working.
fn test_the_browser_fallback_is_caught_as_proportional() {
  m = metrics_measured(266.5625, 71.09375, 10, 36);

}

// ⚠ THE SAME BOX, THE SAME CHARACTER COUNT, A LABEL THAT IS NOT ASCII.
//
// `fit_text` computes its cut in CHARACTERS — `px / advance` — and loft's slice
// bounds are BYTE offsets, so a two-byte character spends two of them. The cut
// then lands early and the label comes back SHORTER than it had room for: not
// corrupt (loft snaps a byte cut outward to a character boundary), just quietly
// wrong, and invisible to a suite whose every fixture is ASCII.
//
// 'thíck_cürved' is 12 characters and 14 bytes. At advance 8 a 48px box has
// room for 6, so 4 survive the marker: 'thíc..'. Byte-cutting at 4 yields 'thí'
// — three characters — and a panel that had room for a fourth never shows it.
fn test_fit_counts_characters_not_bytes() {
  m = mono(8);
  s = fit_text("thíck_cürved", 48, m);
  assert(s == "thíc..", "want 'thíc..', got '{s}'");
  assert(len(s) == 6, "6 characters of room were paid for, {len(s)} came back");
  assert(text_width(s, m) <= 48, "the truncation must itself fit");
}

// ⚠ THE MARKER ONLY APPEARS WHEN SOMETHING WAS REMOVED. Otherwise a caller
// cannot tell "this fitted" from "this was never measured", and a panel full of
// ".." would read as a rendering bug rather than as narrow boxes.
fn test_the_marker_means_something_was_cut() {
  m = mono(8);
  assert(fit_text("Item", 200, m) == "Item", "no marker on a label that fits");
  cut = fit_text("thick_curved", 48, m);

// ⚠ `fits` AND `fit_text` ARE ONE QUESTION ASKED TWO WAYS, and the test is that
// they agree rather than that each is right on its own. A boolean twin that drifts
// from the function it predicts is worse than no twin: a caller asks `fits` and
// then draws the untruncated string, and the overflow is somewhere else entirely.
//
// ⚠ SWEPT ACROSS THE BOUNDARY, not sampled near it. The interesting pixel is the
// one where the answer changes, and a fixture that happens to sit either side of it
// tests nothing about the edge.
fn test_fits_is_true_exactly_when_nothing_would_be_cut() {
  m = mono(8);
  seen_yes = 0;
  seen_no = 0;
  for px in 0..120 {
    y = fits("Stencil", px, m);
    same = fit_text("Stencil", px, m) == "Stencil";
    assert(y == same, "at {px}px `fits` says {y} and `fit_text` "
                    + "{if same { \"changes nothing\" } else { \"cuts\" }}");
    if y { seen_yes += 1; } else { seen_no += 1; }
  }
  assert(seen_yes > 0 && seen_no > 0,
         "the sweep never crossed the boundary — {seen_yes} fit, {seen_no} did not");
}

fn test_unmeasured_metrics_do_not_mangle_text() {
